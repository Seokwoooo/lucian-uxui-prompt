<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var ls = root.querySelector('.u470-ls'), em = root.querySelector('.u470-em'), undo = root.querySelector('.u470-undo'), fg = root.querySelector('.u470-rg .fg'), num = root.querySelector('.u470-rg b'), live = root.querySelector('.u470-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = {
    del: function (n) { return n + ' deleted. You can undo for 5 seconds'; },
    back: function (n) { return n + ' restored'; }
  };
  var LIFE = 5000, C = 2 * Math.PI * 10, slot = null, raf = 0, last = 0, paused = false, backT = 0;
  function rows() { return [].slice.call(ls.querySelectorAll('.u470-r:not(.is-gone)')); }
  function nameOf(r) { return r.querySelector('.u470-tx b').textContent; }
  function empty() { em.classList.toggle('is-on', !rows().length); }
  function tick(now) {
    raf = 0;
    if (!root.isConnected || !slot) return;
    if (!paused) slot.left -= Math.min(100, now - last);
    last = now;
    var f = Math.max(0, slot.left / LIFE);
    fg.style.strokeDasharray = (C * f).toFixed(2) + ' ' + C.toFixed(2);
    num.textContent = String(Math.max(1, Math.ceil(slot.left / 1000)));
    if (slot.left <= 0) { commit(); return; }
    raf = requestAnimationFrame(tick);
  }
  function showPill(on, keyboard) {
    undo.classList.toggle('is-on', on);
    undo.tabIndex = on ? 0 : -1;
    if (on) undo.removeAttribute('aria-hidden'); else undo.setAttribute('aria-hidden', 'true');
    if (on && keyboard) undo.focus();
  }
  function commit() {
    if (!slot) return;
    var r = slot.row;
    slot = null;
    cancelAnimationFrame(raf); raf = 0;
    if (r.parentNode) r.parentNode.removeChild(r);
    showPill(false);
    empty();
  }
  ls.addEventListener('click', function (e) {
    var b = e.target.closest('.u470-del');
    if (!b) return;
    var r = b.closest('.u470-r');
    if (r.classList.contains('is-gone')) return;
    if (slot) commit();
    var keyboard = e.detail === 0;
    r.classList.remove('is-back');
    r.classList.add('is-gone');
    r.querySelector('.u470-del').tabIndex = -1;
    slot = { row: r, left: LIFE, name: nameOf(r) };
    paused = false;
    last = performance.now();
    fg.style.strokeDasharray = C.toFixed(2) + ' ' + C.toFixed(2);
    num.textContent = '5';
    showPill(true, keyboard);
    live.textContent = T.del(slot.name);
    setTimeout(empty, rm ? 0 : 400);
    raf = requestAnimationFrame(tick);
  });
  undo.addEventListener('click', function (e) {
    if (!slot) return;
    var r = slot.row, n = slot.name, keyboard = e.detail === 0;
    slot = null;
    cancelAnimationFrame(raf); raf = 0;
    clearTimeout(backT);
    r.classList.add('is-back');
    r.classList.remove('is-gone');
    var del = r.querySelector('.u470-del');
    del.tabIndex = 0;
    em.classList.remove('is-on');
    showPill(false);
    live.textContent = T.back(n);
    if (keyboard) del.focus();
    backT = setTimeout(function () { if (root.isConnected) r.classList.remove('is-back'); }, rm ? 0 : 800);
  });
  undo.addEventListener('pointerenter', function () { paused = true; });
  undo.addEventListener('pointerleave', function () { paused = false; });
  undo.addEventListener('focus', function () { paused = true; });
  undo.addEventListener('blur', function () { paused = false; });
})();
</script>
