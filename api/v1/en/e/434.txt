<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var w = q('.u434-w'), f = q('.u434-f'), input = q('.u434-in'), lb = q('.u434-lb'), h = q('.u434-msg .h'), e = q('.u434-msg .e'), s = q('.u434-msg .s');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { bad: 'That doesn’t look like an email', ok: 'Looks good, you can use this one', help: 'A work email joins your team right away' };
  var uid = 'u434' + Math.random().toString(36).slice(2, 7);
  input.id = uid; lb.htmlFor = uid;
  h.textContent = T.help; e.textContent = T.bad; s.textContent = T.ok;
  var msg = q('.u434-msg'); msg.id = uid + 'm'; input.setAttribute('aria-describedby', msg.id);
  var timer = 0;
  function set(st) {
    w.dataset.s = st;
    f.classList.toggle('is-ok', st === 'ok');
    if (st === 'err') { f.classList.remove('is-err'); void f.offsetWidth; f.classList.add('is-err'); } else f.classList.remove('is-err');
    input.setAttribute('aria-invalid', st === 'err' ? 'true' : 'false');
    [h, e, s].forEach(function (x) { x.setAttribute('aria-hidden', 'true'); });
    ({ idle: h, err: e, ok: s })[st].removeAttribute('aria-hidden');
  }
  function check() {
    var v = input.value.trim();
    if (!v) { set('idle'); return; }
    set(/^[^\s@]+@[^\s@]+\.[a-z]{2,}$/i.test(v) ? 'ok' : 'err');
  }
  input.addEventListener('input', function () {
    clearTimeout(timer);
    f.classList.add('is-focus');
    if (w.dataset.s !== 'idle') set('idle');
    timer = setTimeout(function () { if (root.isConnected) check(); }, rm ? 300 : 800);
  });
  input.addEventListener('focus', function () { f.classList.add('is-focus'); });
  input.addEventListener('blur', function () { f.classList.remove('is-focus'); clearTimeout(timer); check(); });
  f.addEventListener('click', function () { f.classList.add('is-focus'); input.focus({ preventScroll: true }); });
  set('idle');
})();
</script>
