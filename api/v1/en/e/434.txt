<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var input = q('.u434-live'), box = q('.u434-lbox'), tag = q('.u434-ltag'), help = q('.u434-lhelp'), msg = q('.u434-lhelp span'), ct = q('.u434-ct'), st = q('.u434-st'), lbl = q('.u434-ll');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { s: { default: 'Default', focus: 'Focus', filled: 'Filled', check: 'Checking', error: 'Error', success: 'Success' }, help: '3–16 lowercase letters, numbers or _', bad: 'Only lowercase letters, numbers and _', short: 'Use at least 3 characters', checking: 'Checking availability…', taken: 'That name is taken', ok: 'That name is available' };
  var TAKEN = ['admin', 'root', 'lucian', 'test', 'support'];
  var ICON = { ok: '<svg class="ok" viewBox="0 0 24 24"><path d="m5 12.5 4.5 4.5L19 7.5"/></svg>', no: '<svg class="no" viewBox="0 0 24 24"><circle cx="12" cy="12" r="8.5"/><path d="M12 8v4.5M12 16h.01"/></svg>' };
  var uid = 'u434' + Math.random().toString(36).slice(2, 7);
  input.id = uid + 'in'; lbl.htmlFor = input.id;
  msg.id = uid + 'h'; input.setAttribute('aria-describedby', msg.id);
  var focused = false, timer = 0, state = '';

  function set(s, text, icon) {
    state = s;
    tag.dataset.s = s;
    tag.textContent = T.s[s];
    box.classList.toggle('is-error', s === 'error');
    box.classList.toggle('is-success', s === 'success');
    help.classList.toggle('is-error', s === 'error');
    help.classList.toggle('is-success', s === 'success');
    if (msg.textContent !== text) msg.textContent = text;
    st.innerHTML = icon === 'spin' ? '<i class="u434-spin"></i>' : icon ? ICON[icon] : '';
    input.setAttribute('aria-invalid', s === 'error' ? 'true' : 'false');
  }
  function check() {
    var v = input.value;
    clearTimeout(timer);
    ct.textContent = v.length + '/16';
    ct.classList.toggle('is-max', v.length >= 16);
    if (!v) { set(focused ? 'focus' : 'default', T.help); return; }
    if (!/^[a-z0-9_]*$/.test(v)) { set('error', T.bad, 'no'); return; }
    if (v.length < 3) { set('filled', T.short); return; }
    set('check', T.checking, 'spin');
    timer = setTimeout(function () {
      if (!root.isConnected || input.value !== v) return;
      if (TAKEN.indexOf(v) >= 0) set('error', T.taken, 'no');
      else set('success', T.ok, 'ok');
    }, rm ? 150 : 650);
  }
  input.addEventListener('input', function () { box.classList.add('is-act'); check(); });
  input.addEventListener('focus', function () { focused = true; if (!input.value) set('focus', T.help); });
  input.addEventListener('blur', function () { focused = false; box.classList.remove('is-act'); if (!input.value) set('default', T.help); });
  box.addEventListener('click', function () { box.classList.add('is-act'); if (!input.value) set('focus', T.help); input.focus({ preventScroll: true }); });
  set('default', T.help);
})();
</script>
