    color = i;
    pr.style.setProperty('--pc', COLORS[i][1]);
    nmC.textContent = '· ' + COLORS[i][0];
    SW.forEach(function (b, k) { b.setAttribute('aria-checked', k === i ? 'true' : 'false'); b.tabIndex = k === i ? 0 : -1; });

  });
