<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var dz = q('.u432-dz'), br = q('.u432-br'), fileIn = q('.u432-file'), ls = q('.u432-ls'), none = q('.u432-none'), cnt = q('.u432-hd small'), live = q('.u432-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { count: '{n} files', one: '1 file', done: 'Done', net: 'Upload interrupted', size: 'Over the 10 MB limit', type: 'Format not supported', retry: 'Retry', remove: 'Remove {n}', up: 'Uploading {n}', ok: '{n} uploaded', bad: '{n}: {e}' };
  var MAX = 10 * 1024 * 1024;
  var EXT = { pdf: '#E5484D', png: '#8E4EC6', jpg: '#0090FF', jpeg: '#0090FF', docx: '#2B63D9' };
  var ICON = {
    ok: '<svg class="ok" viewBox="0 0 24 24" aria-hidden="true"><path d="m5 12.5 4.5 4.5L19 7.5"/></svg>',
    no: '<svg class="no" viewBox="0 0 24 24" aria-hidden="true"><circle cx="12" cy="12" r="8.5"/><path d="M12 8v4.5M12 16h.01"/></svg>',
    retry: '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="M4.5 12a7.5 7.5 0 1 0 2.2-5.3L4.5 9"/><path d="M4.5 4.5V9H9"/></svg>',
    x: '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="M6 6l12 12M18 6 6 18"/></svg>'
  };
  var items = [];
  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function fmt(b) { return b < 1024 * 1024 ? Math.max(1, Math.round(b / 1024)) + 'KB' : (b / 1048576).toFixed(1) + 'MB'; }
  function extOf(n) { var m = /\.([a-z0-9]+)$/i.exec(n); return m ? m[1].toLowerCase() : ''; }

  function count() {
    cnt.textContent = items.length === 1 ? T.one : items.length ? T.count.replace('{n}', items.length) : '';
    none.hidden = !!items.length;
    ls.hidden = !items.length;
  }
  function paint(it) {
    var r = it.row;
    r.dataset.st = it.st;
    if (it.retry) r.dataset.retry = ''; else r.removeAttribute('data-retry');
    var em = r.querySelector('.u432-st em');
    if (it.st === 'up') em.textContent = fmt(it.size * it.p) + ' / ' + fmt(it.size) + ' · ' + Math.floor(it.p * 100) + '%';
    else if (it.st === 'done') em.textContent = fmt(it.size) + ' · ' + T.done;
    else em.textContent = T[it.err];
    r.querySelector('.u432-bar i').style.setProperty('--p', it.p.toFixed(3));
  }
  function add(name, size, opt) {
    opt = opt || {};
    var ext = extOf(name);
    var it = { name: name, size: size, ext: ext, p: opt.p || 0, st: 'up', err: '', retry: false };
    if (!EXT[ext]) { it.st = 'err'; it.err = 'type'; }
    else if (size > MAX) { it.st = 'err'; it.err = 'size'; }
    else if (opt.fail) { it.st = 'err'; it.err = 'net'; it.retry = true; }
    else if (opt.done) { it.st = 'done'; it.p = 1; }
    var r = document.createElement('div');
    r.className = 'u432-row';
    r.setAttribute('role', 'listitem');
    r.innerHTML = '<span class="u432-ty" aria-hidden="true"><i></i></span><div class="u432-md"><b class="u432-nm"></b><span class="u432-st">' + ICON.ok + ICON.no + '<em></em></span></div>' +
      '<button type="button" class="u432-retry">' + ICON.retry + '<span></span></button><button type="button" class="u432-rm">' + ICON.x + '</button><span class="u432-bar" aria-hidden="true"><i></i></span>';
    r.querySelector('.u432-ty i').textContent = (ext || '?').slice(0, 4).toUpperCase();
    if (EXT[ext]) r.querySelector('.u432-ty').style.setProperty('--c', EXT[ext]);
    r.querySelector('.u432-nm').textContent = name;
    r.querySelector('.u432-nm').title = name;
    r.querySelector('.u432-retry span').textContent = T.retry;
    r.querySelector('.u432-rm').setAttribute('aria-label', T.remove.replace('{n}', name));
    if (opt.noAnim) r.style.animation = 'none';
    it.row = r;
    items.push(it);
    ls.appendChild(r);
    paint(it);
    count();
    if (it.st === 'up') start(it);
    else if (!opt.quiet) live.textContent = T.bad.replace('{n}', name).replace('{e}', T[it.err]);
    if (!opt.noAnim) ls.scrollTop = ls.scrollHeight;
    return it;
  }
  function start(it) {
    it.st = 'up'; it.err = ''; it.retry = false;
    paint(it);
    live.textContent = T.up.replace('{n}', it.name);
    var per = 0.045 * Math.max(0.6, Math.min(1.2, 3e6 / Math.max(it.size, 1)));
    clearTimeout(it.t);
    (function tick() {
      if (!root.isConnected || items.indexOf(it) < 0) return;
      it.p = Math.min(1, it.p + per * (0.55 + Math.random() * 0.9));
      if (it.p >= 1) {
        it.st = 'done';
        live.textContent = T.ok.replace('{n}', it.name);
        paint(it);
        return;
      }
      paint(it);
      it.t = setTimeout(tick, rm ? 16 : 90);
    })();
  }
  function remove(it) {
    var i = items.indexOf(it);
    if (i < 0) return;
    items.splice(i, 1);
    clearTimeout(it.t);
    var r = it.row, next = items[i] || items[i - 1];
    var finish = function () { r.remove(); count(); };
    if (rm || !r.animate) finish();
    else r.animate([{ opacity: 1, height: '52px', marginBottom: '0px' }, { opacity: 0, height: '0px', marginBottom: '-8px' }], { duration: 220, easing: 'cubic-bezier(.2,.8,.2,1)' }).onfinish = finish;
    if (r.contains(document.activeElement)) (next ? next.row.querySelector('.u432-rm') : br).focus({ preventScroll: true });
  }
  ls.addEventListener('click', function (e) {
    var b = e.target.closest('button');
    if (!b) return;
    var row = b.closest('.u432-row'), it = items.filter(function (x) { return x.row === row; })[0];
    if (!it) return;
    if (b.classList.contains('u432-rm')) remove(it);
    else if (b.classList.contains('u432-retry')) { it.p = Math.max(0, it.p - 0.02); start(it); }
  });
  function addList(list) { Array.prototype.forEach.call(list || [], function (f) { add(f.name, f.size); }); }
  dz.addEventListener('click', function (e) { if (!e.target.closest('.u432-file')) fileIn.click(); });
  br.addEventListener('click', function (e) { e.stopPropagation(); fileIn.click(); });
  fileIn.addEventListener('change', function () { addList(fileIn.files); fileIn.value = ''; });
  var depth = 0;
  dz.addEventListener('dragenter', function (e) { e.preventDefault(); depth++; dz.classList.add('is-drag'); });
  dz.addEventListener('dragover', function (e) { e.preventDefault(); if (e.dataTransfer) e.dataTransfer.dropEffect = 'copy'; });
  dz.addEventListener('dragleave', function () { depth = Math.max(0, depth - 1); if (!depth) dz.classList.remove('is-drag'); });
  dz.addEventListener('drop', function (e) {
    e.preventDefault(); depth = 0; dz.classList.remove('is-drag');
    if (e.dataTransfer) addList(e.dataTransfer.files);
  });

  add('brand_guide_v3.pdf', 2516582, { done: true, noAnim: true });
  add('main_banner@2x.png', 4299161, { p: 0.38, noAnim: true });
  add('contract_final.docx', 1258291, { p: 0.62, fail: true, noAnim: true, quiet: true });
  ls.scrollTop = 0;
  live.textContent = '';
})();
</script>
