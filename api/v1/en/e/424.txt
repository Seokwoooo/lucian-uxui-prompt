<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var ls = root.querySelector('.u424-ls');
  var chips = Array.prototype.slice.call(root.querySelectorAll('.u424-chip'));
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var uid = 'u424-' + Math.random().toString(36).slice(2, 7);
  var BILL = [
    ['Will I be charged when the trial ends?', 'No. We remind you 3 days before it ends, and you only move to a paid plan once you add a payment method.'],
    ['Do I get a refund if I cancel a yearly plan?', 'We refund the unused days. You can request it right from Settings > Billing.'],
    ['Can I get tax invoices?', 'Yes. Add your business details and an e-invoice is issued automatically on each billing date.']
  ];
  var ACC = [
    ['How do I invite teammates?', 'Enter their email in Settings > Members and an invite link goes out right away.'],
    ['Can I have more than one workspace?', 'Yes. Create as many as you need with one account and switch from the top left.'],
    ['Can I change my email address?', 'Change it in Settings > My account, and we send a confirmation to the new address.']
  ];
  var SEC = [
    ['Where is my data stored?', 'All data is encrypted in a Korean region and backed up daily.'],
    ['Do you support SSO?', 'SAML-based SSO and SCIM provisioning come with the Team plan.'],
    ['Can I turn on two-step verification?', 'Yes. Turn it on with an authenticator app or text messages in Settings > Security.']
  ];
  var CATS = [[BILL[0], ACC[0], SEC[0]], BILL, ACC, SEC];
  var cat = 0, qs = [], swT = 0;
  chips.forEach(function (c, i) { c.id = uid + '-c' + i; });
  ls.id = uid + '-p';
  function render(open) {
    var list = CATS[cat];
    ls.innerHTML = '';
    ls.setAttribute('aria-labelledby', chips[cat].id);
    qs = list.map(function (qa, i) {
      var li = document.createElement('li');
      li.className = 'u424-it';
      li.innerHTML = '<button class="u424-q" type="button" aria-expanded="false"><span></span><svg viewBox="0 0 14 14" aria-hidden="true"><path d="m3.5 5.3 3.5 3.5 3.5-3.5"/></svg></button><div class="u424-a" role="region"><div><p></p></div></div>';
      var q = li.querySelector('button'), a = li.querySelector('.u424-a');
      q.querySelector('span').textContent = qa[0];
      li.querySelector('p').textContent = qa[1];
      q.id = uid + '-q' + cat + i; a.id = uid + '-a' + cat + i;
      q.setAttribute('aria-controls', a.id); a.setAttribute('aria-labelledby', q.id);
      a.inert = true;
      q.addEventListener('click', function () { toggle(i); });
      q.addEventListener('keydown', function (e) {
        var to = e.key === 'ArrowDown' ? i + 1 : e.key === 'ArrowUp' ? i - 1 : e.key === 'Home' ? 0 : e.key === 'End' ? list.length - 1 : null;
        if (to === null) return;
        e.preventDefault(); qs[(to + list.length) % list.length].focus();
      });
      ls.appendChild(li);
      return q;
    });
    if (open >= 0) toggle(open);
  }
  function toggle(i) {
    qs.forEach(function (q, k) {
      var li = q.parentNode, v = k === i && q.getAttribute('aria-expanded') !== 'true';
      li.classList.toggle('open', v);
      q.setAttribute('aria-expanded', String(v));
      li.querySelector('.u424-a').inert = !v;
    });
  }
  function pick(i, focus) {
    if (i === cat) return;
    cat = i;
    chips.forEach(function (c, k) { c.setAttribute('aria-selected', String(k === i)); c.tabIndex = k === i ? 0 : -1; });
    if (focus) chips[i].focus({ preventScroll: true });
    clearTimeout(swT);
    if (rm) { render(-1); return; }
    ls.classList.add('is-sw');
    swT = setTimeout(function () { if (!root.isConnected) return; render(-1); ls.classList.remove('is-sw'); }, 180);
  }
  chips.forEach(function (c, i) {
    c.setAttribute('aria-controls', ls.id);
    c.addEventListener('click', function () { pick(i); });
    c.addEventListener('keydown', function (e) {
      var n = chips.length, to = e.key === 'ArrowRight' ? cat + 1 : e.key === 'ArrowLeft' ? cat - 1 : e.key === 'Home' ? 0 : e.key === 'End' ? n - 1 : null;
      if (to === null) return;
      e.preventDefault(); pick((to + n) % n, true);
    });
  });
  render(0);
})();
</script>
