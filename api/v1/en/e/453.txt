<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var h = root.querySelector('.u453-h'), pn = root.querySelector('.u453-pn'), ic = root.querySelector('.u453-ic svg');
  var a = root.querySelector('.u453-a'), b = root.querySelector('.u453-b'), live = root.querySelector('.u453-live');
  var lns = [].slice.call(root.querySelectorAll('.u453-ln'));
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var uid = 'u453' + Math.random().toString(36).slice(2, 7);
  pn.id = uid;
  h.setAttribute('aria-controls', uid);
  var T = { done: 'Thought for 4 seconds' };
  var SPIN = 2400, THINK = 4000;
  var spin = null;
  function later(fn, ms) { setTimeout(function () { if (root.isConnected) fn(); }, rm ? 0 : ms); }
  function setOpen(v) { h.setAttribute('aria-expanded', v ? 'true' : 'false'); }
  h.addEventListener('click', function () { setOpen(h.getAttribute('aria-expanded') !== 'true'); });
  h.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && h.getAttribute('aria-expanded') === 'true') { e.preventDefault(); e.stopPropagation(); setOpen(false); }
  });
  if (!rm && ic.animate) spin = ic.animate([{ transform: 'rotate(0deg) scale(1)' }, { transform: 'rotate(180deg) scale(.78)' }, { transform: 'rotate(360deg) scale(1)' }], { duration: SPIN, iterations: Infinity });
  [450, 1500, 2650].forEach(function (t, i) { later(function () { lns[i].classList.add('is-on'); }, t); });
  later(function () {
    root.classList.add('is-done');
    a.setAttribute('aria-hidden', 'true');
    b.removeAttribute('aria-hidden');
    setOpen(false);
    live.textContent = T.done;
    if (spin) {
      var p = ((spin.currentTime || 0) % SPIN) / SPIN, deg = p * 360, s = p < .5 ? 1 - .44 * p : .78 + .44 * (p - .5);
      spin.cancel();
      ic.animate([{ transform: 'rotate(' + deg + 'deg) scale(' + s + ')' }, { transform: 'rotate(' + (Math.ceil((deg + 25) / 90) * 90) + 'deg) scale(1)' }], { duration: 750, easing: 'cubic-bezier(.3,1.45,.5,1)', fill: 'forwards' });
    }
  }, THINK);
})();
</script>
