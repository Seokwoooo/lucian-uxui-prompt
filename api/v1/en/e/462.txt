<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var tr = root.querySelector('.u462-tr'), th = root.querySelector('.u462-th'), live = root.querySelector('.u462-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { done: 'Payment complete' };
  var W = 280, TH = 52, PAD = 4, MAX = W - TH - PAD * 2, DONE_W = 172;
  var x = 0, v = 0, target = 0, raf = 0, drag = null, done = false, tBack = 0, tIdle = 0;
  function paint(xr) {
    var px = Math.max(0, Math.min(MAX, xr)), q = Math.min(0.16, Math.max(0, -xr) / 36);
    var p = px / MAX;
    root.style.setProperty('--u462-x', px.toFixed(2) + 'px');
    root.style.setProperty('--u462-p', p.toFixed(3));
    th.style.setProperty('--u462-qx', (1 - q).toFixed(3));
    th.style.setProperty('--u462-qy', (1 + q * 0.5).toFixed(3));
    th.setAttribute('aria-valuenow', String(Math.round(p * 100)));
  }
  function frame() {
    raf = 0;
    if (!root.isConnected) return;
    for (var i = 0; i < 4; i++) { var a = 260 * (target - x) - 15 * v; v += a / 240; x += v / 240; }
    paint(x);
    if (target >= MAX && !drag && !done && x >= MAX - 0.5) { finish(); return; }
    if (Math.abs(target - x) < 0.05 && Math.abs(v) < 0.05) { x = target; v = 0; paint(x); onRest(); return; }
    raf = requestAnimationFrame(frame);
  }
  function go(t) {
    target = t;
    if (rm) { x = t; v = 0; paint(x); onRest(); return; }
    if (!raf) raf = requestAnimationFrame(frame);
  }
  function onRest() { if (target >= MAX && !done && !drag) finish(); }
  function finish() {
    done = true;
    cancelAnimationFrame(raf); raf = 0;
    root.classList.remove('is-run');
    root.classList.add('is-done');
    x = DONE_W - TH - PAD * 2;
    paint(x);
    th.setAttribute('aria-valuetext', 'done');
    live.textContent = T.done;
    clearTimeout(tBack);
    tBack = setTimeout(reset, rm ? 1600 : 2800);
  }
  function reset() {
    if (!root.isConnected) return;
    done = false;
    root.classList.remove('is-done');
    root.classList.add('is-back');
    x = 0; v = 0; target = 0;
    paint(0);
    th.removeAttribute('aria-valuetext');
    live.textContent = '';
    setTimeout(function () { if (root.isConnected) root.classList.remove('is-back'); }, rm ? 0 : 700);
  }
  th.addEventListener('pointerdown', function (e) {
    if (e.button !== 0 || done) return;
    var r = root.getBoundingClientRect();
    drag = { x0: e.clientX, b: x, k: root.offsetWidth / (r.width || 1), lx: e.clientX, lt: performance.now(), vel: 0 };
    cancelAnimationFrame(raf); raf = 0; v = 0;
    root.classList.add('is-run');
    clearTimeout(tIdle);
    try { th.setPointerCapture(e.pointerId); } catch (er) {  }
  });
  th.addEventListener('pointermove', function (e) {
    if (!drag) return;
    var now = performance.now(), dt = Math.max(1, now - drag.lt);
    drag.vel = drag.vel * 0.6 + ((e.clientX - drag.lx) * drag.k / dt) * 0.4;
    drag.lx = e.clientX; drag.lt = now;
    var raw = drag.b + (e.clientX - drag.x0) * drag.k;
    x = raw < 0 ? raw * 0.25 : raw > MAX ? MAX + (raw - MAX) * 0.15 : raw;
    paint(x);
    if (raw >= MAX) { drag = null; target = x; go(MAX); }
  });
  function up() {
    if (!drag) return;
    var fling = drag.vel * 160;
    drag = null;
    root.classList.remove('is-run');
    if (x + fling >= MAX * 0.9) go(MAX); else go(0);
  }
  th.addEventListener('pointerup', up);
  th.addEventListener('pointercancel', up);
  th.addEventListener('keydown', function (e) {
    if (done) return;
    var k = e.key, step = MAX / 4;
    if (k === 'Enter' || k === ' ' || k === 'End') { e.preventDefault(); go(MAX); return; }
    if (k === 'ArrowRight' || k === 'ArrowUp') { e.preventDefault(); go(Math.min(MAX, target + step)); }
    else if (k === 'ArrowLeft' || k === 'ArrowDown' || k === 'Home') { e.preventDefault(); go(k === 'Home' ? 0 : Math.max(0, target - step)); }
    else return;
    clearTimeout(tIdle);
    tIdle = setTimeout(function () { if (root.isConnected && !done && target < MAX) go(0); }, 1600);
  });
  paint(0);
})();
</script>
