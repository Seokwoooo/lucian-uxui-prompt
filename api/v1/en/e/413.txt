(function () {
  var root = document.currentScript.previousElementSibling;
  var tr = root.querySelector('.u413-tr'), pn = root.querySelector('.u413-pn');
  var items = Array.prototype.slice.call(pn.querySelectorAll('a'));
  var uid = 'u413-' + Math.random().toString(36).slice(2, 7);
  var open = false, cur = -1, tOpen = 0, tClose = 0, hoverAt = 0;
  pn.id = uid + '-p'; tr.setAttribute('aria-controls', pn.id);
  pn.inert = true;
  function origin() {
    var k = root.getBoundingClientRect().width / root.offsetWidth || 1;
    var a = tr.getBoundingClientRect(), b = pn.getBoundingClientRect();
    pn.style.setProperty('--ox', Math.round((a.left + a.width / 2 - b.left) / k) + 'px');
  }
  function mark(i) {
    cur = i;
    items.forEach(function (a, k) { a.classList.toggle('is-cur', k === i); });
    if (i >= 0) items[i].focus({ preventScroll: true });
  }
  function set(v, back) {
    clearTimeout(tOpen); clearTimeout(tClose);
    if (v === open) return;
    open = v;
    if (v) origin();
    root.classList.toggle('is-open', v);
    tr.setAttribute('aria-expanded', String(v));
    pn.inert = !v;
    if (!v) { mark(-1); if (back) tr.focus({ preventScroll: true }); }
  }
  function enter() {
    clearTimeout(tClose);
    if (!open) { clearTimeout(tOpen); tOpen = setTimeout(function () { if (root.isConnected) { hoverAt = Date.now(); set(true); } }, 140); }
  }
  function leave() {
    clearTimeout(tOpen);
    if (open) { clearTimeout(tClose); tClose = setTimeout(function () { if (root.isConnected) set(false); }, 260); }
  }
  [tr, pn].forEach(function (el) {
    el.addEventListener('pointerenter', function (e) { if (e.pointerType === 'mouse') enter(); });
    el.addEventListener('pointerleave', function (e) { if (e.pointerType === 'mouse') leave(); });
  });
  pn.addEventListener('pointermove', function () { if (cur >= 0) mark(-1); });
  tr.addEventListener('click', function () {
    if (open && Date.now() - hoverAt < 500) return;
    set(!open);
  });
  tr.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowDown' || e.key === 'ArrowUp') { e.preventDefault(); set(true); mark(e.key === 'ArrowDown' ? 0 : items.length - 1); }
    else if (e.key === 'Escape' && open) { e.preventDefault(); e.stopPropagation(); set(false, true); }
  });
  pn.addEventListener('keydown', function (e) {
    var n = items.length, i = cur < 0 ? items.indexOf(document.activeElement) : cur;
    var to = e.key === 'ArrowDown' ? i + 1 : e.key === 'ArrowUp' ? i - 1 : e.key === 'Home' ? 0 : e.key === 'End' ? n - 1 : null;
    if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); set(false, true); return; }
    if (to === null) return;
    e.preventDefault();
    mark((to + n) % n);
  });
  root.addEventListener('focusout', function (e) {
    if (open && e.relatedTarget && !tr.parentNode.contains(e.relatedTarget)) set(false);
  });
  items.forEach(function (a) { a.addEventListener('click', function (e) { e.preventDefault(); set(false, true); }); });
  root.querySelectorAll('.u413-nl, .u413-logo').forEach(function (a) { a.addEventListener('click', function (e) { e.preventDefault(); }); });
  function outside(e) {
    if (!root.isConnected) { document.removeEventListener('pointerdown', outside, true); return; }
    if (open && !tr.contains(e.target) && !pn.contains(e.target)) set(false);
  }
  document.addEventListener('pointerdown', outside, true);
})();
