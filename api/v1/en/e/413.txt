<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var tr = root.querySelector('.u413-tr'), pn = root.querySelector('.u413-pn'), hl = root.querySelector('.u413-hl');
  var its = Array.prototype.slice.call(pn.querySelectorAll('.u413-it'));
  var uid = 'u413-' + Math.random().toString(36).slice(2, 7);
  var open = false, cur = -1, tOpen = 0, tClose = 0, hoverAt = 0;
  pn.id = uid + '-p'; tr.setAttribute('aria-controls', pn.id);
  pn.inert = true;
  function mark(i, focus) {
    cur = i;
    its.forEach(function (a, k) { a.classList.toggle('is-on', k === i); });
    pn.classList.toggle('is-hl', i >= 0);
    if (i >= 0) { hl.style.setProperty('--hy', (its[i].offsetTop + 6) + 'px'); if (focus) its[i].focus({ preventScroll: true }); }
  }
  function set(v, back) {
    clearTimeout(tOpen); clearTimeout(tClose);
    if (v === open) return;
    open = v;
    root.classList.toggle('is-open', v);
    tr.setAttribute('aria-expanded', String(v));
    pn.inert = !v;
    if (!v) { mark(-1); if (back) tr.focus({ preventScroll: true }); }
  }
  function enter() { clearTimeout(tClose); if (!open) { clearTimeout(tOpen); tOpen = setTimeout(function () { if (root.isConnected) { hoverAt = Date.now(); set(true); } }, 120); } }
  function leave() { clearTimeout(tOpen); if (open) { clearTimeout(tClose); tClose = setTimeout(function () { if (root.isConnected) set(false); }, 240); } }
  [tr, pn].forEach(function (el) {
    el.addEventListener('pointerenter', function (e) { if (e.pointerType === 'mouse') enter(); });
    el.addEventListener('pointerleave', function (e) { if (e.pointerType === 'mouse') leave(); });
  });
  its.forEach(function (a, i) {
    a.addEventListener('pointerenter', function () { mark(i); });
    a.addEventListener('click', function (e) { e.preventDefault(); set(false, true); });
  });
  pn.addEventListener('pointerleave', function () { mark(-1); });
  tr.addEventListener('click', function () { if (open && Date.now() - hoverAt < 500) return; set(!open); });
  tr.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowDown' || e.key === 'ArrowUp') { e.preventDefault(); set(true); mark(e.key === 'ArrowDown' ? 0 : its.length - 1, true); }
    else if (e.key === 'Escape' && open) { e.preventDefault(); e.stopPropagation(); set(false, true); }
  });
  pn.addEventListener('keydown', function (e) {
    var n = its.length, i = cur < 0 ? its.indexOf(document.activeElement) : cur;
    if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); set(false, true); return; }
    var to = e.key === 'ArrowDown' ? i + 1 : e.key === 'ArrowUp' ? i - 1 : e.key === 'Home' ? 0 : e.key === 'End' ? n - 1 : null;
    if (to === null) return;
    e.preventDefault(); mark((to + n) % n, true);
  });
  root.addEventListener('focusout', function (e) { if (open && e.relatedTarget && !root.contains(e.relatedTarget)) set(false); });
  function outside(e) {
    if (!root.isConnected) { document.removeEventListener('pointerdown', outside, true); return; }
    if (open && !tr.contains(e.target) && !pn.contains(e.target)) set(false);
  }
  document.addEventListener('pointerdown', outside, true);
})();
</script>
