<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var uid = 'u433' + Math.random().toString(36).slice(2, 7);
  root.querySelectorAll('input[type="radio"]').forEach(function (r) { r.name = uid; });
  var sw = root.querySelector('.u433-sw-row');
  sw.addEventListener('pointerdown', function (e) { if (e.button === 0) sw.classList.add('is-press'); });
  ['pointerup', 'pointerleave', 'pointercancel'].forEach(function (t) { sw.addEventListener(t, function () { sw.classList.remove('is-press'); }); });
  sw.querySelector('input').addEventListener('keydown', function (e) { if (e.key === ' ') sw.classList.add('is-press'); });
  sw.querySelector('input').addEventListener('keyup', function () { sw.classList.remove('is-press'); });
})();
</script>
