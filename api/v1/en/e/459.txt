(function () {

  var q = function (s) { return root.querySelector(s); }, qa = function (s) { return [].slice.call(root.querySelectorAll(s)); };
  var at = q('.u459-at'), isl = q('.u459-i'), tap = q('.u459-tap'), live = q('.u459-live');
  var layers = qa('.u459-v'), ring = q('.u459-ring .fg'), bar = q('.u459-bar i');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { pause: 'Pause', play: 'Play', resume: 'Resume', timer: 'Timer {t}', music: 'Music, {s}', call: 'Incoming call from Mom', oncall: 'On a call with Mom, {t}', ex: ', expanded' };
  var TRACKS = [['Summer Night', 'Hangyeol', 223], ['Sound of Waves', 'Yunseul', 196]];
  var ACT = ['timer', 'music', 'call'];
  var SIZE = { c: [206, 36, 18], te: [316, 84, 42], ce: [316, 84, 42], oe: [316, 84, 42], me: [316, 160, 46] };
  var act = 0, ex = false, callOn = false;
  var tS = 272, tRun = true, mS = 72, track = 0, playing = true, cS = 0;

  function fmt(s) { s = Math.max(0, s); return Math.floor(s / 60) + ':' + ('0' + (s % 60)).slice(-2); }
  function key() { var a = ACT[act]; return (a === 'call' ? (callOn ? 'o' : 'c') : a.charAt(0)) + (ex ? 'e' : 'c'); }
  function label() {
    var a = ACT[act], s = a === 'timer' ? T.timer.replace('{t}', fmt(tS)) : a === 'music' ? T.music.replace('{s}', TRACKS[track][0]) : callOn ? T.oncall.replace('{t}', fmt(cS)) : T.call;
    return s + (ex ? T.ex : '');
  }
  function paintTexts() {
    qa('.u459-tt').forEach(function (e) { e.textContent = fmt(tS); });
    qa('.u459-ct').forEach(function (e) { e.textContent = fmt(cS); });
    ring.style.strokeDashoffset = (56.55 * (1 - tS / 300)).toFixed(2);
    var len = TRACKS[track][2];
    q('.u459-p0').textContent = fmt(mS);
    q('.u459-p1').textContent = '-' + fmt(len - mS);
    bar.style.setProperty('--u459-p', (mS / len * 100).toFixed(2) + '%');
  }
  function render(announce) {
    var k = key(), sz = ex ? SIZE[k] : SIZE.c;
    layers.forEach(function (l) { l.classList.toggle('is-on', l.dataset.v === k); });
    isl.style.width = sz[0] + 'px';
    isl.style.height = sz[1] + 'px';
    isl.style.borderRadius = sz[2] + 'px';
    tap.setAttribute('aria-expanded', ex ? 'true' : 'false');
    tap.setAttribute('aria-label', label());
    if (announce) live.textContent = label();
  }
  function setEx(v) { ex = v; render(true); }
  function nudge(dir) {
    if (rm || !at.animate) return;
    at.animate([{ transform: 'translateX(' + (dir * 14) + 'px)' }, { transform: 'translateX(' + (-dir * 4) + 'px)', offset: .55 }, { transform: 'none' }], { duration: 520, easing: 'ease-out' });
  }
  function go(dir, quiet) {
    act = (act + dir + ACT.length) % ACT.length;
    ex = false;
    render(true);
    if (!quiet) nudge(dir);
  }
  function syncPause() {
    var tp = q('[data-a="pause"]'), mp = q('[data-a="play"]');
    tp.classList.toggle('is-paused', !tRun);
    tp.setAttribute('aria-label', tRun ? T.pause : T.resume);
    isl.classList.toggle('is-paused', !playing);
    mp.setAttribute('aria-label', playing ? T.pause : T.play);
    mp.querySelector('path').setAttribute('d', playing ? 'M7 5h3.2v14H7zM13.8 5H17v14h-3.2z' : 'M8 5.6v12.8c0 .6.6.9 1.1.6l9.9-6.4c.5-.3.5-.9 0-1.2L9.1 5c-.5-.3-1.1 0-1.1.6Z');
  }
  function setTrack(i) {
    track = (i + TRACKS.length) % TRACKS.length;
    mS = 0;
    var mi = q('.u459-mi');
    if (!rm && mi.animate) mi.animate([{ opacity: 0, transform: 'translateY(6px)', filter: 'blur(3px)' }, { opacity: 1, transform: 'none', filter: 'blur(0)' }], { duration: 360, easing: 'cubic-bezier(.3,1.3,.5,1)' });
    q('.u459-song').textContent = TRACKS[track][0];
    q('.u459-artist').textContent = TRACKS[track][1];
    qa('.u459-art').forEach(function (a) { a.classList.toggle('is-b', track === 1); });
    bar.style.transition = 'none';
    paintTexts();
    void bar.offsetWidth;
    bar.style.transition = '';
  }

    cancel: function () { tS = 300; tRun = true; syncPause(); paintTexts(); go(1, true); },
    play: function () { playing = !playing; syncPause(); },
    prev: function () { setTrack(track - 1); },
    next: function () { setTrack(track + 1); },
    accept: function () { callOn = true; cS = 0; paintTexts(); ex = false; render(true); },
    decline: function () { ex = false; act = 0; render(true); },
    end: function () { callOn = false; ex = false; act = 0; render(true); }
  };
  root.addEventListener('click', function (e) {
    var bt = e.target.closest('.u459-v button');
    if (!bt || !ACTIONS[bt.dataset.a]) return;
    var had = document.activeElement === bt;
    ACTIONS[bt.dataset.a]();
    if (had && !bt.closest('.u459-v').classList.contains('is-on')) tap.focus({ preventScroll: true });
  });
  var swiped = false;
  isl.addEventListener('click', function (e) {
    if (e.target.closest('.u459-v button')) return;
    if (swiped) { swiped = false; return; }
    setEx(!ex);
  });
  tap.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowRight' || e.key === 'ArrowLeft') { e.preventDefault(); go(e.key === 'ArrowRight' ? 1 : -1); }
    else if (e.key === 'Escape' && ex) { e.preventDefault(); e.stopPropagation(); setEx(false); }
  });
  var down = null;
  isl.addEventListener('pointerdown', function (e) {
    if (e.button !== 0 || e.target.closest('.u459-v button')) return;
    var k = root.getBoundingClientRect().width / root.offsetWidth || 1;
    down = { x: e.clientX, k: k, dx: 0, id: e.pointerId, cap: false };
    isl.classList.add('is-press');
    at.classList.add('is-drag');
  });
  isl.addEventListener('pointermove', function (e) {
    if (!down) return;
    down.dx = (e.clientX - down.x) / down.k;
    var d = down.dx, r = Math.sign(d) * Math.min(48, Math.abs(d) * 0.4);
    at.style.transform = 'translateX(' + r.toFixed(1) + 'px)';
    if (Math.abs(d) > 8) {
      isl.classList.remove('is-press');
      if (!down.cap) { down.cap = true; try { isl.setPointerCapture(down.id); } catch (er) {  } }
    }
  });
  function up() {
    if (!down) return;
    var d = down.dx;
    down = null;
    isl.classList.remove('is-press');
    at.classList.remove('is-drag');
    at.style.transform = '';
    if (Math.abs(d) > 36) {
      swiped = true;
      setTimeout(function () { swiped = false; }, 0);
      go(d < 0 ? 1 : -1, true);
    }
  }
  isl.addEventListener('pointerup', up);
  isl.addEventListener('pointercancel', up);
  isl.addEventListener('lostpointercapture', up);
  isl.addEventListener('pointerleave', function () { if (down && !down.cap) up(); });
  var clock = setInterval(function () {
    if (!root.isConnected) { clearInterval(clock); return; }
    if (tRun) { tS -= 1; if (tS < 0) tS = 300; }
    if (playing) { mS += 1; if (mS > TRACKS[track][2]) mS = 0; }
    if (callOn) cS += 1;
    paintTexts();
    tap.setAttribute('aria-label', label());
  }, 1000);
  isl.style.transition = 'none';
  syncPause();
  paintTexts();
  render(false);
  void isl.offsetWidth;
  isl.style.transition = '';
})();
