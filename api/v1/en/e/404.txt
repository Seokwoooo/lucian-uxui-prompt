<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var LV = ['', 'Weak', 'Fair', 'Strong', 'Very strong'];
  var T = { ok: 'Ready to sign up', done: 'Signed up' };
  var f = root.querySelector('.u404-f'), inp = root.querySelector('.u404-in'), eye = root.querySelector('.u404-eye');
  var bars = [].slice.call(root.querySelectorAll('.u404-bars i')), meter = root.querySelector('.u404-bars'), lvl = root.querySelector('.u404-st em');
  var rules = [].slice.call(root.querySelectorAll('.u404-rules li')), go = root.querySelector('.u404-go'), live = root.querySelector('.u404-live');
  var uid = 'u404' + Math.random().toString(36).slice(2, 7), ready = false;
  inp.id = uid; f.querySelector('label').htmlFor = uid;
  rules.forEach(function (r, k) { r.id = uid + 'r' + k; });
  inp.setAttribute('aria-describedby', rules.map(function (r) { return r.id; }).join(' '));
  var CHECK = [function (v) { return v.length >= 8; }, function (v) { return /\d/.test(v); }, function (v) { return /[^A-Za-z0-9\s]/.test(v); }];
  function up() { f.classList.toggle('is-up', f.classList.contains('is-f') || !!inp.value); }
  root.addEventListener('pointerdown', function (e) { f.classList.toggle('is-f', f.contains(e.target) && !eye.contains(e.target)); up(); });
  inp.addEventListener('focus', function () { f.classList.add('is-f'); up(); });
  inp.addEventListener('blur', function () { f.classList.remove('is-f'); up(); });
  eye.addEventListener('click', function () { var s = eye.getAttribute('aria-pressed') !== 'true'; eye.setAttribute('aria-pressed', String(s)); inp.type = s ? 'text' : 'password'; });
  function score(v) { return v ? Math.max(1, (v.length >= 8) + /\d/.test(v) + /[^A-Za-z0-9\s]/.test(v) + (v.length >= 12 || (/[a-z]/.test(v) && /[A-Z]/.test(v)))) : 0; }
  function sync() {
    var v = inp.value, lv = score(v), all = true;
    rules.forEach(function (r, k) { var ok = CHECK[k](v); if (ok !== r.classList.contains('is-ok')) r.classList.toggle('is-ok', ok); all = all && ok; });
    root.setAttribute('data-lv', lv);
    bars.forEach(function (b, k) { b.classList.toggle('is-on', k < lv); });
    meter.setAttribute('aria-valuenow', lv); meter.setAttribute('aria-valuetext', LV[lv] || '');
    lvl.textContent = LV[lv];
    if (all !== ready) { ready = all; go.classList.toggle('is-awake', all); go.setAttribute('aria-disabled', String(!all)); if (all) live.textContent = T.ok; }
    up();
  }
  inp.addEventListener('input', sync);
  go.addEventListener('click', function () {
    if (!ready || go.classList.contains('is-ok')) return;
    go.classList.add('is-ok'); live.textContent = T.done;
    setTimeout(function () { if (!root.isConnected) return; go.classList.remove('is-ok'); inp.value = ''; sync(); }, rm ? 1200 : 2200);
  });
  sync();
})();
</script>
