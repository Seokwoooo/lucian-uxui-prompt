<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var LV = ['', 'Weak', 'Fair', 'Strong', 'Very strong'];
  var T = { bad: 'Check the email format', same: 'Passwords match', diff: 'Passwords don’t match', sent: 'Signed up. Check your verification email' };
  var form = root.querySelector('.u404-form'), ins = root.querySelectorAll('.u404-in');
  var nm = ins[0], em = ins[1], pw = ins[2], cf = ins[3];
  var bxs = [].slice.call(root.querySelectorAll('.u404-bx')), msgs = root.querySelectorAll('.u404-msg');
  var eye = root.querySelector('.u404-eye'), bars = [].slice.call(root.querySelectorAll('.u404-bars i')), meter = root.querySelector('.u404-bars'), lvl = root.querySelector('.u404-st em');
  var rules = [].slice.call(root.querySelectorAll('.u404-rules li')), sub = root.querySelector('.u404-sub'), live = root.querySelector('.u404-live');
  var done = root.querySelector('.u404-done'), back = root.querySelector('.u404-back'), to = root.querySelector('.u404-to');
  var uid = 'u404' + Math.random().toString(36).slice(2, 8);
  var MAIL = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;
  var CHECK = [function (v) { return v.length >= 8; }, function (v) { return /\d/.test(v); }, function (v) { return /[^A-Za-z0-9\s]/.test(v); }];
  [].slice.call(root.querySelectorAll('.u404-f > label')).forEach(function (l, k) { ins[k].id = uid + k; l.htmlFor = ins[k].id; });
  msgs[0].id = uid + 'm0'; msgs[1].id = uid + 'm1';
  em.setAttribute('aria-describedby', msgs[0].id); cf.setAttribute('aria-describedby', msgs[1].id);
  rules.forEach(function (r, k) { r.id = uid + 'r' + k; });
  pw.setAttribute('aria-describedby', rules.map(function (r) { return r.id; }).join(' '));
  function later(fn, ms) { setTimeout(function () { if (root.isConnected) fn(); }, ms); }
  function say(p, text, good) { p.textContent = text || ''; p.classList.toggle('is-on', !!text); p.classList.toggle('is-good', !!good); }
  root.addEventListener('pointerdown', function (e) { bxs.forEach(function (b) { b.classList.toggle('is-f', b.contains(e.target) && !eye.contains(e.target)); }); });
  ins.forEach(function (i, k) { i.addEventListener('blur', function () { bxs[k].classList.remove('is-f'); }); });
  function score(v) {
    if (!v) return 0;
    var s = (v.length >= 8) + /\d/.test(v) + /[^A-Za-z0-9\s]/.test(v) + (v.length >= 12 || (/[a-z]/.test(v) && /[A-Z]/.test(v)));
    return Math.max(1, s);
  }
  function emailState(final) {
    var v = em.value.trim(), ok = MAIL.test(v);
    bxs[1].classList.toggle('is-ok', ok);
    if (ok || !v) { bxs[1].classList.remove('is-err'); em.removeAttribute('aria-invalid'); say(msgs[0], ''); }
    else if (final) { bxs[1].classList.add('is-err'); em.setAttribute('aria-invalid', 'true'); say(msgs[0], T.bad); }
    return ok;
  }
  function matchState(final) {
    var a = pw.value, b = cf.value, same = !!b && a === b, wrong = !!b && (final ? a !== b : a.indexOf(b) !== 0);
    bxs[3].classList.toggle('is-ok', same);
    bxs[3].classList.toggle('is-err', wrong);
    if (wrong) cf.setAttribute('aria-invalid', 'true'); else cf.removeAttribute('aria-invalid');
    say(msgs[1], same ? T.same : wrong ? T.diff : '', same);
    return same;
  }
  function sync() {
    var v = pw.value, lv = score(v), all = true;
    rules.forEach(function (r, k) {
      var ok = CHECK[k](v);
      if (ok !== r.classList.contains('is-ok')) r.classList.toggle('is-ok', ok);
      all = all && ok;
    });
    root.setAttribute('data-lv', lv);
    bars.forEach(function (b, k) { b.classList.toggle('is-on', k < lv); });
    meter.setAttribute('aria-valuenow', lv);
    meter.setAttribute('aria-valuetext', LV[lv] || '');
    lvl.textContent = LV[lv];
    var okMail = emailState(false), okMatch = matchState(false);
    sub.disabled = !(nm.value.trim() && okMail && all && okMatch);
  }
  nm.addEventListener('input', sync);
  em.addEventListener('input', sync);
  em.addEventListener('blur', function () { emailState(true); });
  pw.addEventListener('input', sync);
  cf.addEventListener('input', sync);
  cf.addEventListener('blur', function () { if (cf.value) matchState(true); });
  eye.addEventListener('click', function () {
    var show = eye.getAttribute('aria-pressed') !== 'true';
    eye.setAttribute('aria-pressed', String(show));
    pw.type = show ? 'text' : 'password';
  });
  root.querySelector('.u404-ft a').addEventListener('click', function (e) { e.preventDefault(); });
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    if (sub.disabled || sub.classList.contains('is-busy')) return;
    bxs.forEach(function (b) { b.classList.remove('is-f'); });
    sub.classList.add('is-busy'); sub.setAttribute('aria-busy', 'true');
    later(function () {
      sub.classList.remove('is-busy'); sub.removeAttribute('aria-busy');
      to.textContent = em.value.trim();
      root.classList.add('is-done');
      form.inert = true; done.removeAttribute('aria-hidden'); back.tabIndex = 0;
      live.textContent = T.sent;
      back.focus({ preventScroll: true });
    }, rm ? 300 : 1100);
  });
  back.addEventListener('click', function () {
    form.reset(); pw.type = 'password'; eye.setAttribute('aria-pressed', 'false');
    bxs.forEach(function (b) { b.classList.remove('is-ok', 'is-err'); });
    say(msgs[0], ''); say(msgs[1], '');
    sync();
    root.classList.remove('is-done');
    form.inert = false; done.setAttribute('aria-hidden', 'true'); back.tabIndex = -1;
    nm.focus({ preventScroll: true });
  });
  sync();
})();
</script>
