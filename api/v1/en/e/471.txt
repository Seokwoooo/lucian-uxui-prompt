Element: Command K button
ID: 471 · Category: Buttons · Tags: button, keyboard-shortcut, kbd, command-k, 3d-keycap, ripple
What it does: A “Search docs” button with ⌘ and K keycaps on its right. Press ⌘K or click it and the two keycaps sink one after the other and spring back, a ripple spreads from the keys and the button squishes and bounces. How to use it: Press ⌘K (Ctrl+K off a Mac) or click the button
Footprint: fixed size, square

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var b = root.querySelector('.u471-b'), caps = root.querySelectorAll('.u471-k');
  var mod = caps[0], key = caps[1];
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var plat = (navigator.userAgentData && navigator.userAgentData.platform) || navigator.platform || navigator.userAgent || '';
  var mac = /mac|iphone|ipad|ipod/i.test(plat);
  if (!mac) { mod.classList.add('is-wide'); mod.firstChild.textContent = 'Ctrl'; }
  b.setAttribute('aria-keyshortcuts', mac ? 'Meta+K' : 'Control+K');
  var hovered = false, modHeld = false, timers = [];
  function later(fn, ms) { var t = setTimeout(function () { if (root.isConnected) fn(); }, ms); timers.push(t); return t; }
  function down(cap) { cap.classList.add('is-down'); cap._at = performance.now(); }
  function up(cap) {
    var left = 150 - (performance.now() - (cap._at || 0));
    clearTimeout(cap._t);
    if (left > 0) cap._t = later(function () { cap.classList.remove('is-down'); }, left);
    else cap.classList.remove('is-down');
  }
  function answer() {
    var br = b.getBoundingClientRect(), kr = root.querySelector('.u471-ks').getBoundingClientRect();
    var sc = br.width / b.offsetWidth || 1;
    var rp = document.createElement('span');
    rp.className = 'u471-rp';
    rp.style.left = ((kr.left + kr.width / 2 - br.left) / sc) + 'px';
    rp.style.top = ((kr.top + kr.height / 2 - br.top) / sc) + 'px';
    b.appendChild(rp);
    later(function () { rp.remove(); }, 800);
    b.classList.remove('is-pop'); void b.offsetWidth;
    b.classList.add('is-hit', 'is-pop');
    clearTimeout(b._t);
    b._t = later(function () { b.classList.remove('is-hit'); }, 420);
    later(function () { b.classList.remove('is-pop'); }, 600);
  }
  function chord() {
    key.classList.remove('is-ask');
    var wasHeld = mod.classList.contains('is-down');
    if (!wasHeld) down(mod);
    later(function () { down(key); answer(); later(function () { up(key); if (!modHeld) up(mod); }, rm ? 120 : 170); }, wasHeld || rm ? 0 : 70);
  }
  function isMod(e) { return e.key === 'Meta' || e.key === 'Control' || e.key === 'OS'; }
  function onKey(e) {
    if (e.type === 'keydown') {
      if (isMod(e)) { if (!e.repeat) { modHeld = true; down(mod); key.classList.add('is-ask'); } return; }
      if ((e.metaKey || e.ctrlKey) && (e.key === 'k' || e.key === 'K') && !e.altKey && !e.shiftKey) {
        e.preventDefault();
        if (!e.repeat) chord();
      }
    } else if (isMod(e)) {
      modHeld = false;
      key.classList.remove('is-ask');
      up(mod);
    }
  }
  root.addEventListener('keydown', onKey);
  root.addEventListener('keyup', onKey);
  function onDoc(e) {
    if (!root.isConnected) { window.removeEventListener('keydown', onDoc, true); window.removeEventListener('keyup', onDoc, true); window.removeEventListener('blur', reset); return; }
    if (root.contains(e.target)) return;
    var a = document.activeElement;
    if (a && !root.contains(a) && (a.isContentEditable || /^(INPUT|TEXTAREA|SELECT)$/.test(a.tagName))) return;
    if (!hovered && !(e.type === 'keyup' && modHeld)) return;
    if (e.type === 'keydown' && (e.metaKey || e.ctrlKey) && (e.key === 'k' || e.key === 'K') && !e.altKey && !e.shiftKey) e.stopPropagation();
    onKey(e);
  }
  function reset() { modHeld = false; hovered = false; key.classList.remove('is-ask'); mod.classList.remove('is-down'); key.classList.remove('is-down'); b.classList.remove('is-down'); }
  window.addEventListener('keydown', onDoc, true);
  window.addEventListener('keyup', onDoc, true);
  window.addEventListener('blur', reset);
  root.addEventListener('pointerenter', function () { hovered = true; });
  root.addEventListener('pointerleave', function () {
    hovered = false;
    if (modHeld) { modHeld = false; key.classList.remove('is-ask'); up(mod); }
  });
  var pressed = false;
  b.addEventListener('pointerdown', function (e) {
    if (e.button !== 0) return;
    pressed = true;
    b.classList.add('is-down');
    down(mod);
    later(function () { if (pressed) down(key); }, rm ? 0 : 50);
  });
  function release() {
    if (!pressed) return;
    pressed = false;
    b.classList.remove('is-down');
    up(key); later(function () { if (!modHeld) up(mod); }, rm ? 0 : 40);
  }
  ['pointerup', 'pointerleave', 'pointercancel'].forEach(function (t) { b.addEventListener(t, release); });
  b.addEventListener('click', function (e) {
    if (e.detail === 0) chord(); else answer();
  });
})();
</script>
