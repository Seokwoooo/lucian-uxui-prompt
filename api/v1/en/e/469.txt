<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var bk = root.querySelector('.u469-bk'), prev = root.querySelector('.u469-prev'), next = root.querySelector('.u469-next'), live = root.querySelector('.u469-live');
  var pgL = root.querySelector('.u469-pg.l'), pgR = root.querySelector('.u469-pg.r'), csR = root.querySelector('.u469-cs.r'), csL = root.querySelector('.u469-cs.l');
  var src = [].slice.call(root.querySelectorAll('.u469-src .u469-pc'));
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { pages: function (a, b, t) { return 'Pages ' + a + '–' + b + ' · ' + t; } };
  var PW = 132, N = 6, SW = PW / N, CURL = 11, PERSP = 820, LAST = src.length / 2 - 1, LIFT = 0.07;
  function clone(n) { var c = src[n - 1].cloneNode(true); c.setAttribute('aria-hidden', 'true'); return c; }
  function show(box, n) { box.textContent = ''; box.appendChild(clone(n)); }
  var lf = document.createElement('div');
  lf.className = 'u469-lf';
  var strips = [];
  for (var i = 0; i < N; i++) {
    var s = document.createElement('div'), f = document.createElement('div'), b = document.createElement('div');
    s.className = 'u469-s'; f.className = 'u469-f'; b.className = 'u469-f b';
    var df = document.createElement('i'), db = document.createElement('i');
    df.className = db.className = 'u469-sd';
    f.appendChild(df); b.appendChild(db);
    s.appendChild(f); s.appendChild(b);
    lf.appendChild(s);
    strips.push({ el: s, f: f, b: b, df: df, db: db });
  }
  bk.insertBefore(lf, csR);
  function mount(front, back) {
    strips.forEach(function (st, i) {
      var cf = clone(front), cb = clone(back);
      cf.style.setProperty('--u469-o', -(i * SW) + 'px');
      cb.style.setProperty('--u469-o', -((N - 1 - i) * SW) + 'px');
      if (st.f.firstChild !== st.df) st.f.removeChild(st.f.firstChild);
      if (st.b.firstChild !== st.db) st.b.removeChild(st.b.firstChild);
      st.f.insertBefore(cf, st.df); st.b.insertBefore(cb, st.db);
    });
  }
  var cur = 0, mode = '', p = 0, v = 0, t = 0, dir = 1, raf = 0, drag = null, flying = false;
  function paint() {
    var q = Math.max(0, Math.min(1, p)), d = -CURL * Math.sin(Math.PI * q) * dir;
    var base = Math.max(-180, Math.min(0, -180 * q - d * (N - 1) * 0.5)), chain = 'perspective(' + PERSP + 'px) rotateY(' + base.toFixed(2) + 'deg)', a = base;
    strips.forEach(function (st, i) {
      if (i) { chain += ' translateX(' + SW + 'px) rotateY(' + d.toFixed(2) + 'deg)'; a += d; }
      st.el.style.transform = chain;
      var c = Math.cos(a * Math.PI / 180);
      st.el.classList.toggle('is-b', c < 0);
      st.df.style.setProperty('--u469-d', ((1 - c) * 0.2).toFixed(3));
      st.db.style.setProperty('--u469-d', ((1 + c) * 0.2).toFixed(3));
    });
    var k = mode ? Math.sin(Math.PI * q) : 0;
    csR.style.opacity = (k * (q < 0.5 ? 1 : 0.35)).toFixed(3);
    csL.style.opacity = (k * (q > 0.5 ? 1 : 0.35)).toFixed(3);
  }
  function spread() { show(pgL, cur * 2 + 1); show(pgR, cur * 2 + 2); }
  function setup(m) {
    if (mode) return mode === m;
    if (m === 'fwd' && cur >= LAST || m === 'back' && cur <= 0) return false;
    mode = m;
    if (m === 'fwd') { mount(cur * 2 + 2, cur * 2 + 3); p = t = 0; dir = 1; }
    else { mount(cur * 2, cur * 2 + 1); p = t = 1; dir = -1; }
    v = 0;
    paint();
    lf.classList.add('is-on');
    if (m === 'fwd') show(pgR, cur * 2 + 4); else show(pgL, cur * 2 - 1);
    return true;
  }
  function teardown() {
    var done = mode === 'fwd' ? p >= 1 : p <= 0;
    if (done) cur += mode === 'fwd' ? 1 : -1;
    spread();
    lf.classList.remove('is-on');
    mode = ''; flying = false;
    csR.style.opacity = csL.style.opacity = 0;
    prev.disabled = cur === 0;
    next.disabled = cur === LAST;
    if (done) say();
  }
  function say() { var a = cur * 2 + 1; live.textContent = T.pages(a, a + 1, src[a - 1].querySelector('b').textContent + ', ' + src[a].querySelector('b').textContent); }
  function frame() {
    raf = 0;
    if (!root.isConnected || !mode) return;
    if (!drag) {
      for (var i = 0; i < 4; i++) { var acc = 48 * (t - p) - 12.6 * v; v += acc / 240; p += v / 240; }
      if (p > 1) { p = 1; v = -v * 0.2; }
      if (p < 0) { p = 0; v = -v * 0.2; }
      if (Math.abs(t - p) < 0.0015 && Math.abs(v) < 0.01) { p = t; v = 0; }
    }
    paint();
    var rest = mode === 'fwd' ? 0 : 1;
    if (!drag && p === t && (t === 0 || t === 1) && (flying || t === rest)) { teardown(); return; }
    if (drag || p !== t) raf = requestAnimationFrame(frame);
  }
  function run() { if (!raf) raf = requestAnimationFrame(frame); }
  function aim(to) { t = to; if (rm) { p = to; v = 0; paint(); frame(); return; } run(); }
  function go(d) {
    if (drag || flying) return;
    if (!setup(d > 0 ? 'fwd' : 'back')) return;
    flying = true;
    if (mode === 'fwd') dir = 1; else dir = -1;
    aim(d > 0 ? 1 : 0);
  }
  function hover(d, on) {
    if (drag || flying || rm) return;
    var m = d > 0 ? 'fwd' : 'back';
    if (on) { if (setup(m)) aim(d > 0 ? LIFT : 1 - LIFT); }
    else if (mode === m) aim(d > 0 ? 0 : 1);
  }
  function down(e, d) {
    if (e.button !== 0 || flying) return;
    if (!setup(d > 0 ? 'fwd' : 'back')) return;
    var r = bk.getBoundingClientRect(), k = bk.offsetWidth / (r.width || 1);
    drag = { d: d, x0: (e.clientX - r.left) * k - PW, k: k, left: r.left, p0: p, moved: false, lx: e.clientX, lt: performance.now(), v: 0 };
    v = 0;
    try { e.currentTarget.setPointerCapture(e.pointerId); } catch (er) {  }
    run();
  }
  function move(e) {
    if (!drag) return;
    var x = (e.clientX - drag.left) * drag.k - PW, now = performance.now();
    if (Math.abs(x - drag.x0) > 4) drag.moved = true;
    if (!drag.moved) return;
    drag.v = drag.v * 0.5 + ((e.clientX - drag.lx) * drag.k / Math.max(1, now - drag.lt)) * 0.5;
    drag.lx = e.clientX; drag.lt = now;
    var edge = (drag.d > 0 ? PW : -PW) + (x - drag.x0);
    p = Math.acos(Math.max(-1, Math.min(1, edge / PW))) / Math.PI;
    paint();
  }
  function up() {
    if (!drag) return;
    var g = drag;
    drag = null;
    flying = true;
    if (!g.moved) { aim(g.d > 0 ? 1 : 0); return; }
    aim(p - g.v * 0.25 > 0.5 ? 1 : 0);
  }
  [prev, next].forEach(function (btn, k) {
    var d = k ? 1 : -1;
    btn.addEventListener('pointerenter', function () { hover(d, true); });
    btn.addEventListener('pointerleave', function () { hover(d, false); });
    btn.addEventListener('pointerdown', function (e) { down(e, d); });
    btn.addEventListener('pointermove', move);
    btn.addEventListener('pointerup', up);
    btn.addEventListener('pointercancel', up);
    btn.addEventListener('click', function (e) { if (e.detail === 0) go(d); });
  });
  root.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowRight') { e.preventDefault(); go(1); }
    else if (e.key === 'ArrowLeft') { e.preventDefault(); go(-1); }
  });
  spread();
  say();
})();
