<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var cart = q('.u442-cart'), badge = q('.u442-bd'), pn = q('.u442-pn'), cnt = q('.u442-ph small'), fs = q('.u442-fs'), fsT = q('.u442-fs p span'), bar = q('.u442-bar'), ls = q('.u442-ls'), sumB = q('.u442-sum b'), ship = q('.u442-sum small'), go = q('.u442-go'), live = q('.u442-live');
  var T = { cart: 'Cart, {n} items', count: '{n} items', more: '{p} more for free shipping', free: 'Free shipping unlocked', fee: 'Shipping {p}', feeFree: 'Free shipping', pay: 'Checkout · {p}', removed: 'Removed {n}', undo: 'Undo', rm: 'Remove {n}', minus: 'One less {n}', plus: 'One more {n}', empty: 'Nothing here yet', qty: 'Quantity {n}' };
  var FREE = 50000, FEE = 3000;
  var ART = {
    tumbler: '<svg viewBox="0 0 28 28" aria-hidden="true"><rect x="10" y="3" width="8" height="3.5" rx="1.4" fill="#2A2D33"/><rect x="9.2" y="6" width="9.6" height="2" rx=".8" fill="#C4C8CF"/><path d="M9.4 8h9.2l-.8 15.6a1.4 1.4 0 0 1-1.4 1.3h-4.8a1.4 1.4 0 0 1-1.4-1.3z" fill="#9DB4A0"/><path d="M10.3 8h2l-.5 16.9h-.6a1.4 1.4 0 0 1-1.4-1.3z" fill="#fff" opacity=".3"/></svg>',
    mug: '<svg viewBox="0 0 28 28" aria-hidden="true"><path d="M19 11h1.6a3.4 3.4 0 0 1 0 6.8H19" fill="none" stroke="#D8D2C4" stroke-width="2"/><path d="M6.5 8.5h13v11a4 4 0 0 1-4 4h-5a4 4 0 0 1-4-4z" fill="#F4F0E6"/><path d="M6.5 8.5h13v2.2h-13z" fill="#E4DCCB"/><path d="M8 11h2.2v10.5A3 3 0 0 1 8 19.6z" fill="#fff" opacity=".7"/></svg>',
    towel: '<svg viewBox="0 0 28 28" aria-hidden="true"><path d="M5 7.5h18v14H5z" fill="#D9CDB8"/><path d="M5 10h18M5 12.2h18" stroke="#B9A98D" stroke-width="1"/><path d="M5 17.5h18v4H5z" fill="#CDBFA6"/><path d="M5 7.5h18v1.4H5z" fill="#fff" opacity=".35"/></svg>'
  };
  var items = [
    { name: 'Stainless tumbler', opt: 'Sage · 500 ml', price: 26900, qty: 1, art: 'tumbler' },
    { name: 'Ceramic mug', opt: 'Ivory · 350 ml', price: 12000, qty: 1, art: 'mug' },
    { name: 'Linen tea towel', opt: 'Oatmeal · set of 2', price: 3100, qty: 1, art: 'towel' }
  ];
  var undo = null, undoT = 0, lastCount = -1;
  var JOSA = ['', ''];
  function josa(w) { var c = w.charCodeAt(w.length - 1) - 0xAC00; return c >= 0 && c < 11172 ? JOSA[c % 28 ? 0 : 1] : JOSA[1]; }
  function removedText(n) { return T.removed.replace('{n}', n).replace('{j}', josa(n)); }
  function won(n) { return '₩' + n.toLocaleString('ko-KR'); }
  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function row(it, i) {
    return '<li class="u442-it" data-i="' + i + '"><span class="u442-th is-' + it.art + '">' + ART[it.art] + '</span>' +
      '<span class="u442-tx"><b>' + esc(it.name) + '</b><small>' + esc(it.opt) + '</small></span>' +
      '<span class="u442-rt"><span class="u442-pr">' + won(it.price * it.qty) + '</span><span class="u442-ctl"><span class="u442-st">' +
      '<button type="button" class="u442-minus" aria-label="' + esc(T.minus.replace('{n}', it.name)) + '"' + (it.qty <= 1 ? ' disabled' : '') + '><svg viewBox="0 0 24 24"><path d="M6 12h12"/></svg></button>' +
      '<span aria-label="' + esc(T.qty.replace('{n}', it.qty)) + '">' + it.qty + '</span>' +
      '<button type="button" class="u442-plus" aria-label="' + esc(T.plus.replace('{n}', it.name)) + '"' + (it.qty >= 9 ? ' disabled' : '') + '><svg viewBox="0 0 24 24"><path d="M12 6v12M6 12h12"/></svg></button></span>' +
      '<button type="button" class="u442-rm" aria-label="' + esc(T.rm.replace('{n}', it.name)) + '"><svg viewBox="0 0 24 24"><path d="M5 7h14M10 7V5h4v2M7 7l.8 12a2 2 0 0 0 2 1.8h4.4a2 2 0 0 0 2-1.8L17 7"/></svg></button></span></span></li>';
  }
  function patch(i) {
    var li = ls.querySelector('[data-i="' + i + '"]');
    if (!li) return;
    var it = items[i];
    li.querySelector('.u442-pr').textContent = won(it.price * it.qty);
    var sp = li.querySelector('.u442-st span');
    sp.textContent = it.qty; sp.setAttribute('aria-label', T.qty.replace('{n}', it.qty));
    li.querySelector('.u442-minus').disabled = it.qty <= 1;
    li.querySelector('.u442-plus').disabled = it.qty >= 9;
  }
  function draw() {
    var html = '';
    items.forEach(function (it, i) {
      if (it.gone && undo && undo.i === i) html += '<li class="u442-ud"><span>' + esc(removedText(it.name)) + '</span><button type="button" class="u442-undo">' + T.undo + '</button></li>';
      else if (!it.gone) html += row(it, i);
    });
    ls.innerHTML = html || '<li class="u442-em">' + T.empty + '</li>';
  }
  function totals() {
    var n = 0, sum = 0;
    items.forEach(function (it) { if (!it.gone) { n += it.qty; sum += it.qty * it.price; } });
    var free = sum >= FREE, fee = free || !sum ? 0 : FEE;
    badge.textContent = n;
    badge.hidden = !n;
    if (lastCount >= 0 && n !== lastCount) { badge.classList.remove('is-bump'); void badge.offsetWidth; badge.classList.add('is-bump'); }
    lastCount = n;
    cart.setAttribute('aria-label', T.cart.replace('{n}', n));
    cnt.textContent = T.count.replace('{n}', n);
    fs.classList.toggle('is-free', free);
    fsT.innerHTML = free ? esc(T.free) : esc(T.more).replace('{p}', '<b>' + won(FREE - sum) + '</b>');
    var pct = Math.min(1, sum / FREE);
    bar.style.setProperty('--p', pct.toFixed(3));
    bar.setAttribute('aria-valuenow', Math.round(pct * 100));
    sumB.textContent = won(sum);
    ship.textContent = '· ' + (free ? T.feeFree : T.fee.replace('{p}', won(FEE)));
    go.textContent = T.pay.replace('{p}', won(sum + fee));
    go.disabled = !sum;
  }
  function finishUndo() {
    if (undo) { var it = items[undo.i]; it.gone = true; undo = null; draw(); }
  }
  ls.addEventListener('click', function (e) {
    var b = e.target.closest('button');
    if (!b) return;
    if (b.classList.contains('u442-undo')) {
      clearTimeout(undoT);
      var i0 = undo.i; items[i0].gone = false; undo = null;
      draw(); totals();
      var back = ls.querySelector('[data-i="' + i0 + '"]');
      back.classList.add('is-back');
      back.querySelector('.u442-rm').focus({ preventScroll: true });
      return;
    }
    var li = b.closest('.u442-it'), i = +li.dataset.i, it = items[i];
    if (b.classList.contains('u442-plus')) { it.qty++; patch(i); totals(); }
    else if (b.classList.contains('u442-minus')) { it.qty = Math.max(1, it.qty - 1); patch(i); totals(); if (b.disabled) li.querySelector('.u442-plus').focus({ preventScroll: true }); }
    else if (b.classList.contains('u442-rm')) {
      finishUndo();
      it.gone = true; undo = { i: i };
      draw(); totals();
      live.textContent = removedText(it.name);
      var ub = ls.querySelector('.u442-undo');
      if (ub && e.detail === 0) ub.focus({ preventScroll: true });
      clearTimeout(undoT);
      undoT = setTimeout(function () { if (root.isConnected) finishUndo(); }, 5000);
    }
  });
  function setOpen(v, focus) {
    root.classList.toggle('is-open', v);
    cart.setAttribute('aria-expanded', v ? 'true' : 'false');
    if (!v && focus) cart.focus({ preventScroll: true });
  }
  cart.addEventListener('click', function () { setOpen(!root.classList.contains('is-open')); });
  root.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && root.classList.contains('is-open')) { e.preventDefault(); e.stopPropagation(); setOpen(false, true); }
  });
  root.addEventListener('pointerdown', function (e) {
    if (root.classList.contains('is-open') && !e.target.closest('.u442-pn, .u442-cart')) setOpen(false);
  });
  draw();
  totals();
})();
</script>
