<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var ring = root.querySelector('.u443-ring'), cards = root.querySelectorAll('.u443-c'), radios = root.querySelectorAll('.u443-c input');
  var uid = 'u443' + Math.random().toString(36).slice(2, 7);
  radios.forEach(function (r) { r.name = uid; });
  function sync() {
    radios.forEach(function (r, i) {
      cards[i].classList.toggle('is-sel', r.checked);
      if (r.checked) ring.style.setProperty('--y', cards[i].offsetTop + 'px');
    });
  }
  radios.forEach(function (r) { r.addEventListener('change', sync); });
  sync();
})();
</script>
