<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var cards = root.querySelectorAll('.u443-c'), radios = root.querySelectorAll('.u443-hd input'), pt = q('.u443-pt'), dcs = q('.u443-dcs'), tot = q('.u443-r.is-tot b'), go = q('.u443-go'), live = q('.u443-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { toss: 'Toss Pay first-payment discount', naver: 'Naver Pay points', pay: 'Pay {p}', busy: 'Requesting payment…', ok: 'Payment request sent' };
  var BASE = 68900;
  var uid = 'u443' + Math.random().toString(36).slice(2, 7);
  radios.forEach(function (r) { r.name = uid + 'pm'; });
  root.querySelectorAll('.u443-fl').forEach(function (f, i) { var c = f.querySelector('select, input'); c.id = uid + 'f' + i; f.querySelector('label').htmlFor = c.id; });
  var shown = BASE, animT = 0, busy = false;
  function won(n) { return '₩' + Math.round(n).toLocaleString('ko-KR'); }
  function discounts() {
    var v = radios[[].findIndex.call(radios, function (r) { return r.checked; })];
    var m = v ? v.value : '', out = [];
    if (m === 'toss') out.push([T.toss, 3000]);
    if (m === 'naver' && pt.getAttribute('aria-checked') === 'true') out.push([T.naver, 2340]);
    return out;
  }
  function setTotal(target) {
    cancelAnimationFrame(animT);
    var from = shown, t0 = performance.now(), dur = rm ? 0 : 380;
    (function step(now) {
      if (!root.isConnected) return;
      var k = dur ? Math.min(1, (now - t0) / dur) : 1, e = 1 - Math.pow(1 - k, 3);
      shown = from + (target - from) * e;
      tot.textContent = won(shown);
      if (!busy && !go.classList.contains('is-ok')) go.textContent = T.pay.replace('{p}', won(shown));
      if (k < 1) animT = requestAnimationFrame(step);
    })(t0);
  }
  function sync() {
    var sel = -1;
    radios.forEach(function (r, i) { if (r.checked) sel = i; });
    cards.forEach(function (c, i) { c.classList.toggle('is-sel', i === sel); c.querySelectorAll('.u443-ex select, .u443-ex input, .u443-ex button').forEach(function (el) { el.tabIndex = i === sel ? 0 : -1; }); });
    var d = discounts(), keep = dcs.children.length;
    dcs.innerHTML = d.map(function (x) { return '<div class="u443-r is-dc"><span>' + x[0] + '</span><span>−' + won(x[1]) + '</span></div>'; }).join('');
    if (keep) Array.prototype.forEach.call(dcs.children, function (r) { r.style.animation = 'none'; });
    setTotal(BASE - d.reduce(function (a, x) { return a + x[1]; }, 0));
  }
  radios.forEach(function (r) { r.addEventListener('change', sync); });
  pt.addEventListener('click', function () { pt.setAttribute('aria-checked', pt.getAttribute('aria-checked') === 'true' ? 'false' : 'true'); sync(); });
  go.addEventListener('click', function () {
    if (busy || go.classList.contains('is-ok')) return;
    busy = true;
    go.setAttribute('aria-disabled', 'true');
    go.innerHTML = '<i aria-hidden="true"></i>' + T.busy;
    live.textContent = T.busy;
    setTimeout(function () {
      if (!root.isConnected) return;
      busy = false;
      go.classList.add('is-ok');
      go.innerHTML = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="m5 12.5 4.5 4.5L19 7.5"/></svg>' + T.ok;
      live.textContent = T.ok;
      setTimeout(function () {
        if (!root.isConnected) return;
        go.classList.remove('is-ok');
        go.removeAttribute('aria-disabled');
        go.textContent = T.pay.replace('{p}', won(shown));
      }, rm ? 800 : 1500);
    }, rm ? 300 : 1100);
  });
  radios[0].checked = true;
  sync();
})();
</script>
