<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var pl = root.querySelector('.u417-pl'), tg = root.querySelector('.u417-tg'), sb = root.querySelector('.u417-sb');
  var its = Array.prototype.slice.call(root.querySelectorAll('.u417-it'));
  var uid = 'u417-' + Math.random().toString(36).slice(2, 7);
  var rail = false, act = its[0];
  sb.id = uid + '-sb'; tg.setAttribute('aria-controls', sb.id);
  function slide() { pl.style.setProperty('--py', act.parentNode.offsetTop + 'px'); }
  its.forEach(function (a, i) {
    a.setAttribute('title', a.querySelector('.u417-lb').textContent);
    a.addEventListener('click', function (e) {
      e.preventDefault();
      act.removeAttribute('aria-current');
      act = a; a.setAttribute('aria-current', 'page');
      slide();
    });
    a.addEventListener('keydown', function (e) {
      var to = e.key === 'ArrowDown' ? i + 1 : e.key === 'ArrowUp' ? i - 1 : null;
      if (to === null || !its[to]) return;
      e.preventDefault(); its[to].focus();
    });
  });
  tg.addEventListener('click', function () {
    rail = !rail;
    root.classList.toggle('is-rail', rail);
    tg.setAttribute('aria-expanded', String(!rail));
    tg.setAttribute('aria-label', rail ? 'Expand sidebar' : 'Collapse sidebar');
  });
  pl.style.transition = 'none'; slide(); void pl.offsetWidth; pl.style.transition = '';
})();
</script>
