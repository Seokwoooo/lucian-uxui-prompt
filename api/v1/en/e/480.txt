Element: PIN keypad
ID: 480 · Category: Sign-in & Sign-up · Tags: pin, passcode, keypad, shuffle, banking, shake
What it does: A six-digit PIN keypad like the ones in banking apps. The digits come shuffled each time, and every press pops a dot full. A wrong PIN turns the dots red, shakes them side to side, clears them and reshuffles the keys. The right one draws the six dots together into a single mint circle that becomes a check.
How to use it: Enter six digits. The demo PIN is 123456
Footprint: fixed size, square

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var title = root.querySelector('.u480-t'), dotsEl = root.querySelector('.u480-dots'), dots = [].slice.call(root.querySelectorAll('.u480-d'));
  var msg = root.querySelector('.u480-m'), pad = root.querySelector('.u480-pad'), live = root.querySelector('.u480-live');
  var keys = [].slice.call(root.querySelectorAll('.u480-k:not(.u480-fn)'));
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { title: 'Enter your 6-digit PIN', ok: 'Verified', bad: 'Wrong PIN ', n: ' digits entered' };
  var PIN = '123456', val = '', tries = 0, busy = false, timers = [];
  function later(fn, ms) { var t = setTimeout(function () { if (root.isConnected) fn(); }, ms); timers.push(t); return t; }
  function shuffle(anim) {
    var d = '0123456789'.split('');
    for (var i = d.length - 1; i > 0; i--) { var j = Math.floor(Math.random() * (i + 1)); var x = d[i]; d[i] = d[j]; d[j] = x; }
    keys.forEach(function (k, i) {
      var put = function () { k.dataset.d = d[i]; k.firstChild.textContent = d[i]; k.classList.remove('is-mix'); };
      if (!anim || rm) { put(); return; }
      later(function () { k.classList.add('is-mix'); }, i * 22);
      later(put, i * 22 + 170);
    });
  }
  function paint() {
    dots.forEach(function (d, i) {
      var on = i < val.length;
      if (on && !d.classList.contains('is-on')) { d.classList.remove('is-off'); d.classList.add('is-on'); }
      if (!on && d.classList.contains('is-on')) { d.classList.remove('is-on'); d.classList.add('is-off'); }
    });
    live.textContent = val.length + T.n;
  }
  function press(k) { k.classList.add('is-down'); later(function () { k.classList.remove('is-down'); }, 140); }
  function input(dg) {
    if (busy || val.length >= 6) return;
    val += dg;
    if (val.length === 1) msg.classList.remove('is-on');
    paint();
    if (val.length === 6) later(check, rm ? 0 : 260);
  }
  function del() { if (busy || !val) return; val = val.slice(0, -1); paint(); }
  function swapTitle(text) { title.textContent = text; title.classList.remove('is-swap'); void title.offsetWidth; title.classList.add('is-swap'); }
  function check() {
    busy = true;
    if (val === PIN) {
      var mid = dotsEl.getBoundingClientRect(), sc = mid.width / dotsEl.offsetWidth || 1, cx = mid.left + mid.width / 2;
      dots.forEach(function (d) { var r = d.getBoundingClientRect(); d.style.setProperty('--u480-tx', (cx - (r.left + r.width / 2)) / sc + 'px'); });
      dotsEl.classList.add('is-ok');
      pad.classList.add('is-lock');
      msg.classList.remove('is-on');
      title.style.color = 'var(--ux-mint)';
      swapTitle(T.ok);
      later(reset, 2400);
      return;
    }
    tries += 1;
    dotsEl.classList.add('is-bad', 'is-shake');
    msg.textContent = T.bad + '(' + tries + '/5)';
    msg.classList.add('is-on');
    later(function () {
      dotsEl.classList.remove('is-shake', 'is-bad');
      val = ''; paint();
      shuffle(true);
      later(function () { busy = false; }, rm ? 0 : 400);
    }, rm ? 300 : 620);
  }
  function reset() {
    val = ''; tries = 0;
    dots.forEach(function (d) { d.classList.remove('is-on', 'is-off'); });
    dotsEl.classList.remove('is-ok');
    pad.classList.remove('is-lock');
    title.style.color = '';
    swapTitle(T.title);
    msg.classList.remove('is-on');
    shuffle(true);
    busy = false;
  }
  pad.addEventListener('pointerdown', function (e) { var k = e.target.closest('.u480-k'); if (k) press(k); });
  pad.addEventListener('click', function (e) {
    var k = e.target.closest('.u480-k');
    if (!k) return;
    if (k.dataset.act === 'del') del();
    else if (k.dataset.act === 'mix') { if (!busy) shuffle(true); }
    else input(k.dataset.d);
  });
  root.addEventListener('keydown', function (e) {
    if (/^[0-9]$/.test(e.key)) {
      e.preventDefault();
      var k = keys.filter(function (x) { return x.dataset.d === e.key; })[0];
      if (k && !busy) press(k);
      input(e.key);
    } else if (e.key === 'Backspace') { e.preventDefault(); press(root.querySelector('[data-act="del"]')); del(); }
  });
  shuffle(false);
  paint();
  live.textContent = '';
})();
</script>
