<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var top = root.querySelector('.u450-top'), fab = root.querySelector('.u450-fab'), ob = root.querySelector('.u450-open'), p = root.querySelector('.u450-p');
  var x = root.querySelector('.u450-x'), form = root.querySelector('.u450-f'), inp = root.querySelector('.u450-in'), msgs = root.querySelector('.u450-msgs');
  var uid = 'u450' + Math.random().toString(36).slice(2, 7), gen = 0, READ = .68;
  p.id = uid; ob.setAttribute('aria-controls', uid); p.inert = true;
  function setP(v, ms) { top.style.setProperty('--pd', (ms || 800) + 'ms'); top.style.setProperty('--p', v.toFixed(3)); }
  setP(READ);
  function later(fn, ms) { var g = gen; setTimeout(function () { if (root.isConnected && g === gen) fn(); }, ms); }
  top.addEventListener('click', function () {
    gen += 1;
    top.classList.remove('is-fly'); void top.offsetWidth; top.classList.add('is-fly');
    setP(0, rm ? 1 : 700);
    later(function () { top.classList.add('is-hide'); }, rm ? 300 : 850);
    later(function () { top.classList.remove('is-hide', 'is-fly'); setP(READ, rm ? 1 : 1000); }, rm ? 1400 : 1950);
  });
  function open() {
    root.classList.add('is-open'); ob.setAttribute('aria-expanded', 'true'); p.inert = false;
    setTimeout(function () { if (root.classList.contains('is-open')) inp.focus({ preventScroll: true }); }, rm ? 0 : 320);
  }
  function close() {
    root.classList.remove('is-open'); ob.setAttribute('aria-expanded', 'false'); p.inert = true;
    ob.focus({ preventScroll: true });
    later(function () { msgs.querySelectorAll('.is-me').forEach(function (m) { m.remove(); }); inp.value = ''; }, 500);
  }
  ob.addEventListener('click', open);
  x.addEventListener('click', close);
  p.addEventListener('keydown', function (e) { if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); close(); } });
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    var v = inp.value.trim(); if (!v) return;
    msgs.querySelectorAll('.is-me').forEach(function (m) { m.remove(); });
    var m = document.createElement('p'); m.className = 'u450-bb is-me'; m.textContent = v; msgs.appendChild(m);
    inp.value = '';
  });
})();
</script>
