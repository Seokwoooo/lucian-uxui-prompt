What it does: A six-box verification code input. Digits pop into their boxes and the cursor moves on; pasting fills them all at once. A wrong code shakes the boxes and clears them; the right one draws the six boxes together into a single check.
How to use it: Try any wrong digits, then type or paste 482915
Footprint: fixed size, square

@keyframes u406-wave{40%{transform:translateY(-4px)}}

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { digit: function (k) { return 'Digit ' + k; }, wrong: 'That code didn’t match', ok: 'Verified' };
  var CODE = '482915';
  var row = root.querySelector('.u406-row'), live = root.querySelector('.u406-live');
  var vals = ['', '', '', '', '', ''], busy = false, gen = 0;
  var bs = [], ins = [], gs = [];
  for (var k = 0; k < 6; k++) {
    var b = document.createElement('div');
    b.className = 'u406-b'; b.style.setProperty('--i', k);
    b.style.setProperty('--mx', ((2.5 - k) * 50) + 'px');
    b.style.setProperty('--d', Math.abs(2.5 - k) * 2);
    b.innerHTML = '<input class="u406-in" inputmode="numeric" pattern="[0-9]*" autocomplete="' + (k ? 'off' : 'one-time-code') + '"><span class="u406-g" aria-hidden="true"></span>';
    b.querySelector('input').setAttribute('aria-label', T.digit(k + 1));
    row.appendChild(b); bs.push(b); ins.push(b.querySelector('input')); gs.push(b.querySelector('.u406-g'));
  }
  function later(fn, ms) { var g = gen; setTimeout(function () { if (root.isConnected && g === gen) fn(); }, ms); }
  function render(popAt) {
    ins.forEach(function (d, j) {
      d.value = vals[j]; bs[j].classList.toggle('is-fill', !!vals[j]);
      if (gs[j].textContent !== vals[j]) {
        gs[j].textContent = vals[j];
        if (vals[j] && popAt && popAt.indexOf(j) >= 0) { gs[j].classList.remove('is-pop'); void gs[j].offsetWidth; gs[j].classList.add('is-pop'); }
      }
    });
  }
  function cur(j) {
    bs.forEach(function (b, n) { b.classList.toggle('is-cur', n === j); });
    if (j >= 0 && ins[j]) ins[j].focus({ preventScroll: true });
  }
  function fill(i, text) {
    var ds = String(text).replace(/\D/g, '').split('');
    if (!ds.length) return;
    if (ds.length >= 6) i = 0;
    var changed = [];
    ds.forEach(function (c, j) { if (i + j < 6) { vals[i + j] = c; changed.push(i + j); } });
    row.classList.remove('is-err');
    render(changed);
    var next = vals.indexOf('');
    if (next < 0) { cur(-1); ins[5].blur(); check(); } else cur(Math.max(next, Math.min(5, i + ds.length)));
  }
  function check() {
    busy = true; row.classList.add('is-busy');
    later(function () {
      row.classList.remove('is-busy');
      if (vals.join('') === CODE) {
        row.classList.add('is-merge'); live.textContent = T.ok;
        later(function () {
          vals = ['', '', '', '', '', '']; render();
          row.classList.remove('is-merge'); busy = false;
        }, rm ? 1200 : 2200);
        return;
      }
      row.classList.add('is-err'); live.textContent = T.wrong;
      row.classList.remove('is-shake'); void row.offsetWidth; row.classList.add('is-shake');
      later(function () { vals = ['', '', '', '', '', '']; render(); row.classList.remove('is-err'); busy = false; cur(0); }, rm ? 300 : 700);
    }, rm ? 200 : 700);
  }
  ins.forEach(function (d, i) {
    d.addEventListener('focus', function () { bs.forEach(function (b, n) { b.classList.toggle('is-cur', n === i); }); });
    d.addEventListener('blur', function () { bs[i].classList.remove('is-cur'); });
    d.addEventListener('pointerdown', function () { if (!busy) bs.forEach(function (b, n) { b.classList.toggle('is-cur', n === i); }); });
    d.addEventListener('input', function () {
      if (busy) { render(); return; }
      var v = d.value.replace(/\D/g, ''), old = vals[i];
      if (!v) { vals[i] = ''; render(); return; }
      var add = v;
      if (old && v.length > 1) add = v.indexOf(old) === 0 ? v.slice(old.length) : v.slice(0, v.length - old.length) || v;
      fill(i, add);
    });
    d.addEventListener('paste', function (e) {
      var t = e.clipboardData && e.clipboardData.getData('text');
      if (!t) return;
      e.preventDefault();
      if (!busy) fill(i, t);
    });
    d.addEventListener('keydown', function (e) {
      if (busy) return;
      if (e.key === 'Backspace') {
        e.preventDefault();
        if (vals[i]) { vals[i] = ''; render(); cur(i); } else if (i > 0) { vals[i - 1] = ''; render(); cur(i - 1); }
      } else if (e.key === 'ArrowLeft' && i > 0) { e.preventDefault(); cur(i - 1); }
      else if (e.key === 'ArrowRight' && i < 5) { e.preventDefault(); cur(i + 1); }
    });
  });
})();
</script>
