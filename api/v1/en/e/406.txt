What it does: The production default for a “Check your email” card: a masked address and six code boxes. How to use it: Try any wrong digits, then type or paste 482915

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = {
    digit: function (k) { return 'Digit ' + k; },
    again: 'Resend code',
    wait: function (t) { return 'Resend in ' + t; },
    checking: 'Checking…',
    wrong: 'That code didn’t match. Please check it again.',
    sent: 'New code sent',
    ok: 'Verified'
  };
  var CODE = '482915';
  var row = root.querySelector('.u406-code'), ds = [].slice.call(row.querySelectorAll('.u406-d'));
  var msg = root.querySelector('.u406-msg'), live = root.querySelector('.u406-live'), main = root.querySelector('.u406-main');
  var done = root.querySelector('.u406-done'), go = root.querySelector('.u406-go');
  var vals = ['', '', '', '', '', ''], busy = false, gen = 0, tick = 0, left = 0, note0 = msg.textContent;
  var rs = document.createElement('button');
  rs.type = 'button'; rs.className = 'u406-rs';
  root.querySelector('.u406-ft span').appendChild(rs);
  ds.forEach(function (d, k) { d.setAttribute('aria-label', T.digit(k + 1)); });
  function later(fn, ms) { var g = gen; setTimeout(function () { if (root.isConnected && g === gen) fn(); }, ms); }
  function say(text, kind, spin) {
    msg.className = 'u406-msg' + (kind ? ' is-' + kind : '');
    msg.innerHTML = spin ? '<i aria-hidden="true"></i>' : '';
    msg.appendChild(document.createTextNode(text || (kind ? '' : note0)));
  }
  function render() { ds.forEach(function (d, k) { d.value = vals[k]; d.classList.toggle('is-fill', !!vals[k]); }); }
  function cur(k) {
    ds.forEach(function (d, j) { d.classList.toggle('is-cur', j === k); });
    if (k >= 0 && ds[k]) ds[k].focus({ preventScroll: true });
  }
  function fill(i, text) {
    var digits = String(text).replace(/\D/g, '').split('');
    if (!digits.length) return;
    if (digits.length >= 6) i = 0;
    digits.forEach(function (c, j) { if (i + j < 6) vals[i + j] = c; });
    if (row.classList.contains('is-err')) { row.classList.remove('is-err'); say(''); }
    render();
    var next = vals.indexOf('');
    if (next < 0) { cur(-1); ds[5].blur(); check(); } else cur(Math.max(next, Math.min(5, i + digits.length)));
  }
  function check() {
    busy = true; row.classList.add('is-busy');
    ds.forEach(function (d) { d.readOnly = true; });
    say(T.checking, '', true);
    later(function () {
      row.classList.remove('is-busy'); busy = false;
      if (vals.join('') === CODE) {
        row.classList.add('is-good'); say('');
        live.textContent = T.ok;
        later(function () {
          root.classList.add('is-done'); main.inert = true;
          done.removeAttribute('aria-hidden'); go.tabIndex = 0; clearInterval(tick);
          go.focus({ preventScroll: true });
        }, rm ? 200 : 600);
        return;
      }
      row.classList.add('is-err'); say(T.wrong, 'err');
      row.classList.remove('is-shake'); void row.offsetWidth; row.classList.add('is-shake');
      later(function () {
        vals = ['', '', '', '', '', '']; render();
        ds.forEach(function (d) { d.readOnly = false; });
        cur(0);
      }, rm ? 300 : 650);
    }, rm ? 300 : 900);
  }
  ds.forEach(function (d, i) {
    d.addEventListener('focus', function () { cur(-1); d.classList.add('is-cur'); if (d.select) d.select(); });
    d.addEventListener('blur', function () { d.classList.remove('is-cur'); });
    d.addEventListener('pointerdown', function () { if (!busy) { ds.forEach(function (o) { o.classList.toggle('is-cur', o === d); }); } });
    d.addEventListener('input', function () {
      if (busy) { render(); return; }
      var v = d.value.replace(/\D/g, ''), old = vals[i];
      if (!v) { vals[i] = ''; render(); return; }
      var add = v;
      if (old && v.length > 1) add = v.indexOf(old) === 0 ? v.slice(old.length) : v.slice(0, v.length - old.length) || v;
      fill(i, add);
    });
    d.addEventListener('paste', function (e) {
      var t = e.clipboardData && e.clipboardData.getData('text');
      if (!t) return;
      e.preventDefault();
      if (!busy) fill(i, t);
    });
    d.addEventListener('keydown', function (e) {
      if (busy) return;
      if (e.key === 'Backspace') {
        e.preventDefault();
        if (vals[i]) { vals[i] = ''; render(); cur(i); }
        else if (i > 0) { vals[i - 1] = ''; render(); cur(i - 1); }
      } else if (e.key === 'ArrowLeft' && i > 0) { e.preventDefault(); cur(i - 1); }
      else if (e.key === 'ArrowRight' && i < 5) { e.preventDefault(); cur(i + 1); }
    });
  });
  function fmt(s) { return Math.floor(s / 60) + ':' + ('0' + (s % 60)).slice(-2); }
  function cool() {
    clearInterval(tick); left = 60;
    rs.disabled = true; rs.textContent = T.wait(fmt(left));
    tick = setInterval(function () {
      if (!root.isConnected) { clearInterval(tick); return; }
      left -= 1;
      if (left <= 0) { clearInterval(tick); rs.disabled = false; rs.textContent = T.again; }
      else rs.textContent = T.wait(fmt(left));
    }, 1000);
  }
  rs.addEventListener('click', function () {
    vals = ['', '', '', '', '', '']; render(); row.classList.remove('is-err');
    say(T.sent, 'good'); cool(); cur(0);
  });
  go.addEventListener('click', function () {
    gen += 1;
    vals = ['', '', '', '', '', '']; render();
    row.classList.remove('is-good', 'is-err'); say('');
    ds.forEach(function (d) { d.readOnly = false; });
    root.classList.remove('is-done'); main.inert = false;
    done.setAttribute('aria-hidden', 'true'); go.tabIndex = -1;
    cool(); cur(0);
  });
  cool();
})();
</script>
