<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var bd = root.querySelector('.u465-bd'), pc = root.querySelector('.u465-pc'), lamp = root.querySelector('.u465-lamp'), live = root.querySelector('.u465-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { on: 'Light on', off: 'Light off' };
  var uid = 'u465' + Math.random().toString(36).slice(2, 7);
  root.querySelectorAll('svg [id]').forEach(function (n) {
    var o = n.id;
    n.id = uid + o;
    root.querySelectorAll('[fill="url(#' + o + ')"]').forEach(function (m) { m.setAttribute('fill', 'url(#' + uid + o + ')'); });
  });
  var PX = 150, PY = -40, AX = 176, AY = 126, RX = 176, RY = 206, PULL = 26;
  function S() { return { v: 0, t: 0, s: 0 }; }
  var by = S(), bx = S(), lm = S();
  var on = false, drag = null, raf = 0, ticked = false;
  function paint() {
    var th = lm.v * Math.PI / 180, c = Math.cos(th), s = Math.sin(th);
    var ax = PX + (AX - PX) * c - (AY - PY) * s, ay = PY + (AX - PX) * s + (AY - PY) * c;
    var ex = RX + bx.v, ey = RY + by.v;
    var mx = (ax + ex) / 2 - bx.s * 0.012, my = (ay + ey) / 2;
    pc.setAttribute('d', 'M' + ax.toFixed(2) + ' ' + ay.toFixed(2) + 'Q' + mx.toFixed(2) + ' ' + my.toFixed(2) + ' ' + ex.toFixed(2) + ' ' + ey.toFixed(2));
    lamp.setAttribute('transform', 'rotate(' + lm.v.toFixed(3) + ' ' + PX + ' ' + PY + ')');
    bd.style.setProperty('--u465-bx', bx.v.toFixed(2) + 'px');
    bd.style.setProperty('--u465-by', by.v.toFixed(2) + 'px');
    bd.style.setProperty('--u465-br', (-Math.atan2(ex - mx, ey - my) * 180 / Math.PI).toFixed(2) + 'deg');
  }
  function step(p, k, c, dt) { var a = k * (p.t - p.v) - c * p.s; p.s += a * dt; p.v += p.s * dt; }
  function frame() {
    raf = 0;
    if (!root.isConnected) return;
    for (var i = 0; i < 4; i++) {
      if (!drag) { step(by, 210, 6.5, 1 / 240); step(bx, 34, 1.6, 1 / 240); }
      step(lm, 26, 2.2, 1 / 240);
    }
    paint();
    var calm = [by, bx, lm].every(function (p) { return Math.abs(p.v - p.t) < 0.02 && Math.abs(p.s) < 0.05; });
    if (drag || !calm) raf = requestAnimationFrame(frame);
    else { by.v = bx.v = lm.v = 0; paint(); }
  }
  function run() { if (!raf) raf = requestAnimationFrame(frame); }
  function toggle() {
    on = !on;
    root.classList.toggle('is-on', on);
    bd.setAttribute('aria-checked', String(on));
    live.textContent = on ? T.on : T.off;
  }
  function release(fromY, sway) {
    if (rm) { by.v = bx.v = lm.v = 0; by.s = bx.s = lm.s = 0; paint(); return; }
    by.v = fromY; by.s = -fromY * 6;
    bx.s += sway; lm.s += sway * 0.05 + (fromY > 0 ? 3 : 0);
    run();
  }
  bd.addEventListener('pointerdown', function (e) {
    if (e.button !== 0) return;
    var r = root.getBoundingClientRect();
    drag = { x: e.clientX, y: e.clientY, k: root.offsetWidth / (r.width || 1), bx: bx.v, moved: false };
    ticked = false;
    by.s = bx.s = 0;
    try { bd.setPointerCapture(e.pointerId); } catch (er) {  }
    run();
  });
  bd.addEventListener('pointermove', function (e) {
    if (!drag) return;
    var dx = (e.clientX - drag.x) * drag.k, dy = (e.clientY - drag.y) * drag.k;
    if (Math.abs(dx) + Math.abs(dy) > 3) drag.moved = true;
    by.v = dy > 0 ? 64 * (1 - Math.exp(-dy / 64)) : dy * 0.12;
    bx.v = Math.max(-26, Math.min(26, drag.bx + dx * 0.45));
    if (!ticked && by.v > PULL) { ticked = true; bd.classList.remove('is-tick'); void bd.offsetWidth; bd.classList.add('is-tick'); }
    paint();
  });
  var skip = false;
  function up() {
    if (!drag) return;
    var d = drag;
    drag = null;
    if (!d.moved) return;
    skip = true;
    setTimeout(function () { skip = false; }, 0);
    if (by.v > PULL) toggle();
    release(by.v, bx.v * -2 + (by.v > PULL ? 40 : 0));
  }
  bd.addEventListener('pointerup', up);
  bd.addEventListener('pointercancel', up);
  bd.addEventListener('click', function () {
    if (skip) { skip = false; return; }
    toggle();
    release(40, 46);
  });
  paint();
})();
</script>
