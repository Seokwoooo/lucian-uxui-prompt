ID: 435 · Category: Alerts & Feedback · Tags: toast, notification, stack, swipe, undo, spring, auto-dismiss
What it does: Press Save and a toast springs up from below, a check draws itself and a thin time bar runs down. Press again and toasts stack like a deck (up to three); hover and they fan out while the timers pause. Flick one sideways or press × to dismiss, and Undo turns the toast into “Undone”.
How to use it: Press Save a few times, then hover the toasts. Swipe one sideways to dismiss
Footprint: fixed size, square

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var tr = root.querySelector('.u435-tr'), rg = root.querySelector('.u435-rg');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { saved: 'Changes saved', undone: 'Undone', undo: 'Undo', close: 'Dismiss' };
  var H = 56, GAP = 8, LIFE = 4500, MAX = 3;
  var toasts = [], open = false, paused = false;

  function layout() {
    var live = toasts.filter(function (t) { return !t.gone; }), n = live.length;
    live.forEach(function (t, k) {
      var i = n - 1 - k;
      var y = open ? -i * (H + GAP) : -i * 11, s = open ? 1 : 1 - i * 0.055;
      t.el.style.setProperty('--y', y + 'px');
      t.el.style.setProperty('--s', s.toFixed(3));
      t.el.style.setProperty('--o', i < MAX ? 1 : 0);
      t.el.style.zIndex = String(20 - i);
      t.el.classList.toggle('is-back', !open && i > 0);
    });
    rg.style.height = (open && n > 1 ? n * H + (n - 1) * GAP : H + 20) + 'px';
  }
  function timer(t) {
    clearTimeout(t.tm);
    t.t0 = Date.now();
    if (!paused) t.tm = setTimeout(function () { if (root.isConnected) dismiss(t); }, t.left);
  }
  function bar(t, ms) {
    var pb = t.el.querySelector('.u435-pb');
    pb.innerHTML = '<i style="animation-duration:' + ms + 'ms"></i>';
    t.left = ms;
    timer(t);
  }
  function make(animate) {
    var el = document.createElement('div');
    el.className = 'u435-t';
    el.setAttribute('role', 'status');
    el.innerHTML = '<span class="u435-ic" aria-hidden="true"><span class="ok"><svg viewBox="0 0 24 24"><path d="m6 12.5 4 4 8-9"/></svg></span><span class="bk"><svg viewBox="0 0 24 24"><path d="M9 14 4.5 9.5 9 5"/><path d="M4.5 9.5h9a5.5 5.5 0 0 1 0 11H11"/></svg></span></span>' +
      '<span class="u435-tx"><b></b><b aria-hidden="true"></b></span><button type="button" class="u435-un"></button><button type="button" class="u435-x"><svg viewBox="0 0 24 24" aria-hidden="true"><path d="M6 6l12 12M18 6 6 18"/></svg></button><span class="u435-pb" aria-hidden="true"></span>';
    var bs = el.querySelectorAll('.u435-tx b');
    bs[0].textContent = T.saved; bs[1].textContent = T.undone;
    el.querySelector('.u435-un').textContent = T.undo;
    el.querySelector('.u435-x').setAttribute('aria-label', T.close);
    var t = { el: el };
    el.querySelector('.u435-x').addEventListener('click', function () { dismiss(t); });
    el.querySelector('.u435-un').addEventListener('click', function () {
      if (t.undone) return;
      t.undone = true;
      el.classList.add('is-undo');
      bs[0].setAttribute('aria-hidden', 'true'); bs[1].removeAttribute('aria-hidden');
      el.querySelector('.u435-x').focus({ preventScroll: true });
      bar(t, 2000);
    });
    swipe(t);
    toasts.push(t);
    if (animate && !rm) {
      el.classList.add('is-in');
      el.style.setProperty('--y', '84px'); el.style.setProperty('--s', '.9'); el.style.setProperty('--o', '0');
      el.style.transition = 'none';
      rg.appendChild(el);
      void el.offsetWidth;
      el.style.transition = '';
    } else rg.appendChild(el);
    bar(t, LIFE);
    var live = toasts.filter(function (x) { return !x.gone; });
    if (live.length > MAX) dismiss(live[0], 0, true);
    layout();
    return t;
  }
  function dismiss(t, dir, quiet) {
    if (t.gone) return;
    t.gone = true;
    clearTimeout(t.tm);
    var el = t.el;
    if (el.contains(document.activeElement)) { var other = toasts.filter(function (x) { return !x.gone; }).pop(); (other ? other.el.querySelector('.u435-x') : tr).focus({ preventScroll: true }); }
    el.style.pointerEvents = 'none';
    var done = function () { el.remove(); var i = toasts.indexOf(t); if (i >= 0) toasts.splice(i, 1); };
    layout();
    if (rm || !el.animate) { done(); return; }
    var y = el.style.getPropertyValue('--y') || '0px', s = el.style.getPropertyValue('--s') || '1';
    if (dir) {
      var v = Math.max(1.2, Math.abs(t.v || 0)), dist = 380, dur = Math.max(160, Math.min(320, dist / v));
      el.classList.add('is-drag');
      el.animate([{ transform: 'translate3d(' + (t.dx || 0) + 'px,' + y + ',0) scale(' + s + ')', opacity: 1 }, { transform: 'translate3d(' + dir * dist + 'px,' + y + ',0) scale(' + s + ') rotate(' + dir * 6 + 'deg)', opacity: 0 }], { duration: dur, easing: 'cubic-bezier(.2,.7,.3,1)', fill: 'forwards' }).onfinish = done;
    } else {
      el.animate([{ opacity: quiet ? 0 : 1 }, { transform: 'translate3d(0,calc(' + y + ' + 14px),0) scale(' + (s * 0.94) + ')', opacity: 0 }], { duration: 260, easing: 'cubic-bezier(.4,0,.6,1)', fill: 'forwards' }).onfinish = done;
    }
  }
  function swipe(t) {
    var el = t.el, on = false, x0 = 0, last = [];
    el.addEventListener('pointerdown', function (e) {
      if (e.button !== 0 || e.target.closest('button') || t.gone) return;
      on = true; x0 = e.clientX; t.dx = 0; t.v = 0; last = [[e.clientX, e.timeStamp]];
      el.classList.add('is-drag');
      try { el.setPointerCapture(e.pointerId); } catch (er) {  }
    });
    el.addEventListener('pointermove', function (e) {
      if (!on) return;
      var k = root.getBoundingClientRect().width / root.offsetWidth || 1;
      t.dx = (e.clientX - x0) / k;
      last.push([e.clientX, e.timeStamp]);
      if (last.length > 5) last.shift();
      el.style.setProperty('--x', t.dx.toFixed(1) + 'px');
      el.style.opacity = String(Math.max(0.35, 1 - Math.abs(t.dx) / 320));
    });
    var end = function () {
      if (!on) return;
      on = false;
      var a = last[0], b = last[last.length - 1], k = root.getBoundingClientRect().width / root.offsetWidth || 1;
      t.v = b && a && b[1] > a[1] ? (b[0] - a[0]) / k / (b[1] - a[1]) : 0;
      el.style.opacity = '';
      if (Math.abs(t.dx) > 80 || Math.abs(t.v) > 0.6) { dismiss(t, (Math.abs(t.v) > 0.6 ? t.v : t.dx) > 0 ? 1 : -1); return; }
      el.classList.remove('is-drag');
      el.style.setProperty('--x', '0px');
      t.dx = 0;
    };
    el.addEventListener('pointerup', end);
    el.addEventListener('pointercancel', end);
  }
  function setOpen(v) {
    if (open === v) return;
    open = v;
    paused = v;
    rg.classList.toggle('is-paused', v);
    toasts.forEach(function (t) {
      if (t.gone) return;
      if (v) { clearTimeout(t.tm); t.left = Math.max(500, t.left - (Date.now() - t.t0)); } else timer(t);
    });
    layout();
  }
  var leaveT = 0;
  rg.addEventListener('pointerenter', function () { clearTimeout(leaveT); setOpen(true); });
  rg.addEventListener('pointerleave', function () { clearTimeout(leaveT); leaveT = setTimeout(function () { if (!rg.contains(document.activeElement)) setOpen(false); }, 120); });
  rg.addEventListener('focusin', function () { setOpen(true); });
  rg.addEventListener('focusout', function (e) { if (!rg.contains(e.relatedTarget)) setOpen(false); });
  rg.addEventListener('keydown', function (e) {
    if (e.key !== 'Escape') return;
    var el = e.target.closest('.u435-t'), t = toasts.filter(function (x) { return x.el === el; })[0];
    if (t && !t.gone) { e.preventDefault(); e.stopPropagation(); dismiss(t); }
  });
  tr.addEventListener('click', function () { make(true); });
  make(false);
})();
</script>
