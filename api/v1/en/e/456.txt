Element: Ghost suggestion
ID: 456 · Category: Inputs & Search · Tags: autocomplete, ghost-text, inline-suggestion, ai, textarea, keyboard
What it does: Pause while writing and the rest of the sentence fades in after the caret in grey, with a tiny Tab keycap at the end. Press Tab and ink spreads from the left as the grey text fills in solid and becomes yours; press Esc and it blurs away. Keep typing the same letters and the grey text shrinks to match.
How to use it: Pause while writing, then press Tab to accept or Esc to dismiss
Footprint: fixed size, square

  <p class="u456-sr u456-hint">Press Tab to accept the grey suggestion, or Esc to dismiss it</p>

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var box = q('.u456-box'), ta = q('.u456-ta'), ov = q('.u456-ov'), hint = q('.u456-hint'), live = q('.u456-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var uid = 'u456' + Math.random().toString(36).slice(2, 7);
  hint.id = uid;
  ta.setAttribute('aria-describedby', uid);
  var T = { sug: 'Suggestion: {s}', took: 'Suggestion accepted' };
  var SUG = ['Can we move tomorrow’s meeting to 3 PM?', 'I’ll share the files by end of day.', 'Please take a look. Thank you!'];
  var PAUSE = 420;
  var ghost = '', shown = false, taking = false, dismissed = false, tm = 0, hideT = 0;

  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function find() {
    var v = ta.value;
    if (v.length < 2 || ta.selectionStart !== v.length || ta.selectionEnd !== v.length) return '';
    for (var i = 0; i < SUG.length; i++) if (SUG[i].length > v.length && SUG[i].indexOf(v) === 0) return SUG[i].slice(v.length);
    return '';
  }
  function paint() {
    ov.innerHTML = '<span>' + esc(ta.value) + '</span>' + (ghost ? '<span class="u456-gh"><span class="u456-gt">' + esc(ghost) + '</span><kbd class="u456-kb">Tab</kbd></span>' : '');
    ov.scrollTop = ta.scrollTop;
  }
  function show(g) {
    clearTimeout(hideT);
    ghost = g;
    paint();
    void ov.offsetWidth;
    shown = true;
    root.classList.add('is-ghost');
    live.textContent = T.sug.replace('{s}', g.trim());
  }
  function hide() {
    clearTimeout(tm);
    if (!shown) { ghost = ''; paint(); return; }
    shown = false;
    root.classList.remove('is-ghost');
    hideT = setTimeout(function () { if (root.isConnected && !shown) { ghost = ''; paint(); } }, rm ? 0 : 320);
  }
  function schedule() {
    clearTimeout(tm);
    tm = setTimeout(function () {
      if (!root.isConnected || dismissed || taking) return;
      var g = find();
      if (g) show(g);
    }, rm ? 0 : PAUSE);
  }
  function accept() {
    if (!shown || taking) return false;
    taking = true;
    root.classList.add('is-take');
    var full = ta.value + ghost;
    setTimeout(function () {
      if (!root.isConnected) return;
      ta.value = full;
      ta.setSelectionRange(full.length, full.length);
      shown = false;
      ghost = '';
      root.classList.remove('is-ghost', 'is-take');
      paint();
      taking = false;
      live.textContent = T.took;
    }, rm ? 0 : 460);
    return true;
  }
  ta.addEventListener('input', function () {
    if (taking) return;
    dismissed = false;
    var g = find();
    if (shown && g && g.length < ghost.length) { ghost = g; paint(); return; }
    hide();
    schedule();
  });
  ta.addEventListener('keydown', function (e) {
    if (e.key === 'Tab' && !e.shiftKey && shown) { e.preventDefault(); accept(); }
    else if (e.key === 'ArrowRight' && shown && ta.selectionStart === ta.value.length) { e.preventDefault(); accept(); }
    else if (e.key === 'Escape' && shown) { e.preventDefault(); e.stopPropagation(); dismissed = true; hide(); }
  });
  ta.addEventListener('click', function () { if (shown && ta.selectionStart !== ta.value.length) hide(); });
  ta.addEventListener('blur', function () { if (shown) hide(); });
  ta.addEventListener('scroll', function () { ov.scrollTop = ta.scrollTop; });
  box.addEventListener('mousedown', function (e) { if (e.target === box) { e.preventDefault(); ta.focus({ preventScroll: true }); } });

  ta.setSelectionRange(ta.value.length, ta.value.length);
  var g0 = find();
  if (g0) show(g0); else paint();
})();
</script>
