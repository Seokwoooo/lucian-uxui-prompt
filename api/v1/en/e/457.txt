How to use it: Press the mic to record, then press stop

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var p = q('.u457-p'), b = q('.u457-b'), tmEl = q('.u457-tm'), wv = q('.u457-wv'), tx = q('.u457-tx'), live = q('.u457-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { idle: 'Voice input', stop: 'Stop recording', again: 'Speak again', rec: 'Listening', got: 'Transcribed: ', cancel: 'Recording canceled' };
  var SAY = ['Wake me up at 7 tomorrow', 'Tell Mom I’m almost there', 'Suggest something for dinner'];
  var N = 18, REC_W = 252;
  wv.innerHTML = new Array(N + 1).join('<i></i>');
  var bars = [].slice.call(wv.children), lv = bars.map(function () { return 0.1; });
  var state = 'idle', said = 0, raf = 0, clock = 0, t0 = 0, gen = 0;

  function size(w, h) { p.style.setProperty('--u457-w', w + 'px'); p.style.setProperty('--u457-h', h + 'px'); }
  function loop(now) {
    if (!root.isConnected || state !== 'rec') { raf = 0; return; }
    var t = (now - t0) / 1000;
    var env = 0.18 + 0.82 * Math.pow(Math.abs(Math.sin(t * 2.4) * Math.sin(t * 3.9 + 1.2)), 0.55);
    for (var i = 0; i < N; i++) {
      var mid = 1 - Math.abs(i - (N - 1) / 2) / (N / 2) * 0.5;
      var n = 0.5 + 0.5 * Math.sin(t * (6.5 + (i % 5) * 1.7) + i * 2.3);
      var target = 0.1 + 0.9 * env * mid * (0.3 + 0.7 * n);
      lv[i] += (target - lv[i]) * 0.3;
      bars[i].style.transform = 'scaleY(' + lv[i].toFixed(3) + ')';
    }
    raf = requestAnimationFrame(loop);
  }
  function fmt(s) { return Math.floor(s / 60) + ':' + ('0' + (s % 60)).slice(-2); }
  function start() {
    gen += 1;
    state = 'rec';
    p.dataset.s = 'rec';
    size(REC_W, 64);
    wv.classList.remove('is-flat');
    lv = bars.map(function () { return 0.1; });
    bars.forEach(function (x) { x.style.transform = 'scaleY(.1)'; });
    t0 = performance.now();
    tmEl.textContent = '0:00';
    clearInterval(clock);
    var g = gen;
    clock = setInterval(function () {
      if (!root.isConnected || g !== gen || state !== 'rec') { clearInterval(clock); return; }
      var s = Math.floor((performance.now() - t0) / 1000);
      tmEl.textContent = fmt(s);
      if (s >= 30) stop();
    }, 250);
    if (rm) bars.forEach(function (x, i) { x.style.transform = 'scaleY(' + (0.3 + 0.4 * Math.abs(Math.sin(i * 1.3))).toFixed(2) + ')'; });
    else if (!raf) raf = requestAnimationFrame(loop);
    b.setAttribute('aria-label', T.stop);
    live.textContent = T.rec;
  }
  function stop() {
    if (state !== 'rec') return;
    gen += 1;
    clearInterval(clock);
    state = 'flat';
    wv.classList.add('is-flat');
    bars.forEach(function (x) { x.style.transform = 'scaleY(.07)'; });
    var text = SAY[said % SAY.length];
    said += 1;
    tx.innerHTML = '<span class="u457-tw"></span>';
    var tw = tx.firstChild;
    text.split(' ').forEach(function (w, i) {
      if (i) tw.appendChild(document.createTextNode(' '));
      var s = document.createElement('span');
      s.textContent = w;
      s.style.setProperty('--u457-i', i);
      tw.appendChild(s);
    });
    var g = gen;
    setTimeout(function () {
      if (!root.isConnected || g !== gen) return;
      state = 'done';
      p.dataset.s = 'done';
      size(Math.ceil(20 + tw.offsetWidth + 14 + 40 + 7), 54);
      b.setAttribute('aria-label', T.again);
      live.textContent = T.got + text;
    }, rm ? 0 : 230);
  }
  function cancel() {
    gen += 1;
    clearInterval(clock);
    state = 'idle';
    p.dataset.s = 'idle';
    size(64, 64);
    b.setAttribute('aria-label', T.idle);
    live.textContent = T.cancel;
  }
  b.addEventListener('click', function () {
    if (state === 'rec') stop();
    else if (state === 'idle' || state === 'done') start();
  });
  root.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && state === 'rec') { e.preventDefault(); e.stopPropagation(); cancel(); }
  });
})();
</script>
