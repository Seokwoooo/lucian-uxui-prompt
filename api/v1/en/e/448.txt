<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var T = { changed: function (n) { return 'Language: ' + n; } };
  var b = root.querySelector('.u448-b'), p = root.querySelector('.u448-p'), lb = root.querySelector('.u448-lb'), g = root.querySelector('.u448-g'), w = root.querySelector('.u448-w'), live = root.querySelector('.u448-live');
  var os = [].slice.call(root.querySelectorAll('.u448-o')), cur = 0, act = 0;
  var uid = 'u448' + Math.random().toString(36).slice(2, 7);
  var cp = g.querySelector('.cp'); cp.id = uid + 'c'; g.querySelector('.clip').setAttribute('clip-path', 'url(#' + cp.id + ')');
  p.id = uid + 'p'; b.setAttribute('aria-controls', p.id);
  os.forEach(function (o, k) { o.id = uid + 'o' + k; o.setAttribute('aria-selected', String(k === cur)); });
  function fit() { var s = lb.querySelector('span:not(.is-out)'); if (s) lb.style.width = s.offsetWidth + 'px'; }
  fit();
  if (document.fonts && document.fonts.ready) document.fonts.ready.then(fit);
  function setAct(k) {
    act = (k + os.length) % os.length;
    os.forEach(function (o, n) { o.classList.toggle('is-act', n === act); });
    p.setAttribute('aria-activedescendant', os[act].id);
  }
  function open() { root.classList.add('is-open'); b.setAttribute('aria-expanded', 'true'); setAct(cur); p.focus({ preventScroll: true }); }
  function close(back) {
    root.classList.remove('is-open'); b.setAttribute('aria-expanded', 'false');
    os.forEach(function (o) { o.classList.remove('is-act'); });
    if (back) b.focus({ preventScroll: true });
  }
  function pick(k) {
    close(true);
    if (k === cur) return;
    cur = k;
    os.forEach(function (o, n) { o.setAttribute('aria-selected', String(n === k)); });
    var name = os[k].querySelector('span').textContent;
    lb.querySelectorAll('.is-out').forEach(function (s) { s.remove(); });
    var old = lb.querySelector('span'), s = document.createElement('span');
    s.textContent = name; s.lang = os[k].lang; s.className = 'is-in';
    old.classList.add('is-out'); lb.appendChild(s);
    setTimeout(function () { if (old.isConnected) old.remove(); }, 420);
    fit();
    g.classList.remove('is-spin'); void g.offsetWidth; g.classList.add('is-spin');
    live.textContent = T.changed(name);
  }
  b.addEventListener('click', function () { if (root.classList.contains('is-open')) close(); else open(); });
  b.addEventListener('keydown', function (e) { if (e.key === 'ArrowDown' || e.key === 'ArrowUp') { e.preventDefault(); open(); } });
  os.forEach(function (o, k) {
    o.addEventListener('click', function () { pick(k); });
    o.addEventListener('pointerenter', function () { if (root.classList.contains('is-open')) setAct(k); });
  });
  p.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowDown') { e.preventDefault(); setAct(act + 1); }
    else if (e.key === 'ArrowUp') { e.preventDefault(); setAct(act - 1); }
    else if (e.key === 'Home') { e.preventDefault(); setAct(0); }
    else if (e.key === 'End') { e.preventDefault(); setAct(os.length - 1); }
    else if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); pick(act); }
    else if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); close(true); }
    else if (e.key === 'Tab') close();
  });
  function outside(e) {
    if (!root.isConnected) { document.removeEventListener('pointerdown', outside, true); return; }
    if (root.classList.contains('is-open') && !w.contains(e.target)) close();
  }
  document.addEventListener('pointerdown', outside, true);
})();
</script>
