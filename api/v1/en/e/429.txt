(function () {

  var P = {
    doc: '<path d="M14 3H7.5A2.5 2.5 0 0 0 5 5.5v13A2.5 2.5 0 0 0 7.5 21h9a2.5 2.5 0 0 0 2.5-2.5V8z"/><path d="M14 3v5h5M9 13h6M9 17h4"/>',

    x: '<path d="M6 6l12 12M18 6 6 18"/>'
  };
  function svg(p) { return '<svg viewBox="0 0 24 24" aria-hidden="true">' + p + '</svg>'; }
  var DATA = [
    ['doc', 'Design system guide', 'Design'],
    ['doc', 'Design review checklist', 'Design'],
    ['project', 'Design tokens v2 migration', 'Project'],
    ['person', 'Seoyeon Han', 'Product designer'],
    ['doc', 'Payment error playbook', 'Engineering'],
    ['doc', 'Onboarding guide', 'People'],
    ['doc', 'Weekly sync, Nov week 3', 'Operations'],
    ['person', 'Jihoon Park', 'Frontend engineer'],
    ['project', 'H2 roadmap', 'Product'],
    ['cmd', 'Create new doc', 'Command'],
    ['doc', 'Marketing campaign brief', 'Marketing'],
    ['person', 'Sumin Lee', 'Growth marketer']
  ];

  var recent = ['payment error', 'onboarding', 'weekly sync'];
  var mode = 'recent', list = [], act = -1;

  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function hl(s, t) {
    var i = t ? s.toLowerCase().indexOf(t) : -1;
    if (i < 0) return esc(s);
    return esc(s.slice(0, i)) + '<mark>' + esc(s.slice(i, i + t.length)) + '</mark>' + esc(s.slice(i + t.length));
  }
  function icon(d) {
    if (d[0] === 'person') return '<span class="u429-ic is-av" aria-hidden="true">' + esc(d[1].charAt(0)) + '</span>';
    return '<span class="u429-ic">' + svg(P[d[0]]) + '</span>';
  }
  function search(t) {
    var a = [];
    DATA.forEach(function (d, i) {
      var ti = d[1].toLowerCase().indexOf(t), mi = d[2].toLowerCase().indexOf(t);
      if (ti < 0 && mi < 0) return;
      a.push({ d: d, s: ti === 0 ? 0 : ti > 0 ? 1 : 2, i: i });
    });
    a.sort(function (x, y) { return x.s - y.s || x.i - y.i; });
    return a.slice(0, 5).map(function (x) { return x.d; });
  }
  function itemRow(d, i, t, side) {
    return '<div class="u429-opt" role="option" id="' + uid + i + '" aria-selected="false">' + icon(d) + '<span class="u429-t">' + hl(d[1], t) + '</span>' + (side ? '<span class="u429-m">' + side + '</span>' : '') + '<span class="u429-en" aria-hidden="true">↵</span></div>';
  }
  function render() {
    var t = input.value.trim().toLowerCase();
    mode = t ? 'search' : 'recent';
    var html = '';
    if (mode === 'recent') {
      list = recent.map(function (r) { return { r: r }; }).concat(SHORT.map(function (d) { return { d: d }; }));
      hdT.textContent = T.recent;
      all.hidden = !recent.length;
      html = recent.map(function (r, i) {

      }).join('') + (recent.length ? '' : '<p class="u429-nr" role="presentation">' + T.noRecent + '</p>');
      html += '<div role="group" aria-labelledby="' + uid + 'g"><div class="u429-gh" id="' + uid + 'g" role="presentation">' + T.short + '</div>' +
        SHORT.map(function (d, k) { return itemRow(d, recent.length + k, '', d[2] ? '<kbd>' + esc(d[2]) + '</kbd>' : ''); }).join('') + '</div>';
      live.textContent = T.count.replace('{n}', recent.length);
    } else {
      list = search(t).map(function (d) { return { d: d }; });
      hdT.textContent = list.length ? T.results.replace('{n}', list.length) : T.none;
      all.hidden = true;
      html = list.map(function (x, i) { return itemRow(x.d, i, t, hl(x.d[2], t)); }).join('');
      live.textContent = hdT.textContent;
    }
    lb.innerHTML = html;
    em.hidden = !(mode === 'search' && !list.length);
    if (!em.hidden) {
      var parts = T.empty.split('{q}');
      emT.textContent = parts[0] + input.value.trim() + (parts[1] || '');
    }
    act = mode === 'search' && list.length ? 0 : -1;
    clr.hidden = !input.value;
    paint();
  }
  function opts() { return lb.querySelectorAll('.u429-opt'); }
  function paint() {
    Array.prototype.forEach.call(opts(), function (o, i) { o.setAttribute('aria-selected', i === act ? 'true' : 'false'); });
    if (act >= 0 && root.classList.contains('is-open')) input.setAttribute('aria-activedescendant', uid + act);
    else input.removeAttribute('aria-activedescendant');
  }
  function setOpen(v) {
    root.classList.toggle('is-open', v);
    input.setAttribute('aria-expanded', v ? 'true' : 'false');
    kb.textContent = v ? 'esc' : mac ? '⌘K' : 'Ctrl K';
    if (v) render(); else { act = -1; paint(); }
  }
  function choose(i) {
    var x = list[i];
    if (!x) return;
    if (x.r) { input.value = x.r; render(); return; }
    var d = x.d, term = input.value.trim();
    if (mode === 'search') recent = [term].concat(recent.filter(function (r) { return r !== term; })).slice(0, 3);
    input.value = mode === 'search' ? d[1] : '';
    clr.hidden = !input.value;
    home.hidden = true;
    doc.hidden = false;
    doc.style.animation = 'none'; void doc.offsetWidth; doc.style.animation = '';
    q('.u429-cr .u429-ic').outerHTML = icon(d);
    q('.u429-crt').textContent = d[2] && d[0] !== 'cmd' ? d[2] : T.ws;
    q('.u429-h').textContent = d[0] === 'cmd' ? T.untitled : d[1];
    setOpen(false);
  }
  function removeRecent(i, kb) {
    if (!list[i] || !list[i].r) return;
    recent.splice(i, 1);
    render();
    act = kb && recent.length ? Math.min(i, recent.length - 1) : -1;
    paint();
  }
  function move(dir) {
    if (!list.length) return;
    act = act < 0 ? (dir > 0 ? 0 : list.length - 1) : (act + dir + list.length) % list.length;
    paint();
  }
  function clearQuery() { input.value = ''; setOpen(true); input.focus({ preventScroll: true }); }

  input.addEventListener('input', function () { if (!root.classList.contains('is-open')) setOpen(true); else render(); });
  field.addEventListener('click', function (e) {

  });
  clr.addEventListener('click', clearQuery);
  emX.addEventListener('click', clearQuery);
  all.addEventListener('click', function () { recent = []; render(); input.focus({ preventScroll: true }); });
  input.addEventListener('keydown', function (e) {
    var openNow = root.classList.contains('is-open');
    if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
      e.preventDefault();
      if (!openNow) { setOpen(true); return; }
      move(e.key === 'ArrowDown' ? 1 : -1);
    } else if (e.key === 'Enter') {
      if (openNow && act >= 0) { e.preventDefault(); choose(act); }
    } else if (e.key === 'Escape') {
      if (openNow) { e.preventDefault(); e.stopPropagation(); setOpen(false); }
      else if (input.value) { e.preventDefault(); e.stopPropagation(); input.value = ''; clr.hidden = true; }
    } else if (e.key === 'Delete' && openNow && list[act] && list[act].r) {
      e.preventDefault(); removeRecent(act, true);
    } else if (e.key === 'Tab' && openNow) setOpen(false);
  });
  root.addEventListener('keydown', function (e) {
    if ((e.metaKey || e.ctrlKey) && !e.altKey && (e.key === 'k' || e.key === 'K')) {
      e.preventDefault();
      input.value = '';
      setOpen(true);
      input.focus({ preventScroll: true });
    }
  });
  pop.addEventListener('mousedown', function (e) { if (!e.target.closest('.u429-em button')) e.preventDefault(); });
  lb.addEventListener('pointermove', function (e) {
    var o = e.target.closest('.u429-opt');
    if (!o) return;
    var i = Array.prototype.indexOf.call(opts(), o);
    if (i !== act) { act = i; paint(); }
  });
  lb.addEventListener('click', function (e) {
    var o = e.target.closest('.u429-opt');
    if (!o) return;
    var i = Array.prototype.indexOf.call(opts(), o);
    if (e.target.closest('.u429-rm')) removeRecent(i); else choose(i);
  });
  root.addEventListener('pointerdown', function (e) {
    if (root.classList.contains('is-open') && !e.target.closest('.u429-f, .u429-pop')) setOpen(false);
  });
  root.addEventListener('focusout', function (e) {
    if (root.classList.contains('is-open') && e.relatedTarget && !root.contains(e.relatedTarget)) setOpen(false);
  });
  render();
})();
