<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var T = {
    total: function (n) { return n + ' reviews'; },
    star: function (s) { return s + '★'; },
    barLabel: function (s, n) { return 'Show only ' + s + '-star reviews (' + n + ')'; },
    head: function (s, p, n) { return n + ' ' + (s ? s + '-star ' : '') + (p ? 'photo ' : '') + (n === 1 ? 'review' : 'reviews'); },
    help: 'Helpful',
    rated: function (r) { return r + ' out of 5 stars'; },
    photo: 'Photo'
  };
  var DIST = [912, 248, 78, 26, 20];
  var REV = [
    ['Seoyeon K.', 'Nov 18, 2026', 5, 1, 24, 'Noise canceling is far better than I expected. I can focus on music even on the subway.'],
    ['Junho L.', 'Nov 15, 2026', 4, 0, 9, 'Happy with the sound, but the case is a little slippery.'],
    ['Jieun P.', 'Nov 12, 2026', 5, 1, 17, 'My ears don’t hurt even after a full day, and the battery lasts two days.'],
    ['Minho C.', 'Nov 9, 2026', 3, 0, 6, 'Call quality is average. On windy days people can’t hear me well.'],
    ['Yuseo Y.', 'Nov 5, 2026', 4, 1, 11, 'The color is nicer than in the photos, and it’s light enough to wear for hours.'],
    ['Hana J.', 'Nov 3, 2026', 1, 1, 3, 'One side kept cutting out, so I asked for an exchange.'],
    ['Jiwoo H.', 'Oct 29, 2026', 2, 0, 4, 'The app keeps disconnecting. Waiting for an update.']
  ];
  var PICS = [['#E8E2D6', '#C9BBA4'], ['#2B2D36', '#6B6F80'], ['#DCE7F2', '#9FB7D2'], ['#F1DCD6', '#D7A99B']];
  var total = DIST.reduce(function (a, b) { return a + b; }, 0);
  var avg = DIST.reduce(function (a, n, k) { return a + n * (5 - k); }, 0) / total;
  var fmt = function (n) { return n.toLocaleString('en-US'); };
  var STAR = '<path d="M8 1.2l2.1 4.3 4.7.7-3.4 3.3.8 4.7L8 12l-4.2 2.2.8-4.7L1.2 6.2l4.7-.7z"/>';
  function stars(r) {
    var half = Math.round(r * 2) / 2, h = '';
    for (var i = 1; i <= 5; i++) {
      var f = Math.max(0, Math.min(1, half - (i - 1)));
      h += '<span class="u446-st" style="--cut:' + (100 - f * 100) + '%"><svg class="bg" viewBox="0 0 16 16" aria-hidden="true">' + STAR + '</svg><svg class="fg" viewBox="0 0 16 16" aria-hidden="true">' + STAR + '</svg></span>';
    }
    return h;
  }
  var bigStars = root.querySelector('.u446-avg .u446-stars');
  root.querySelector('.u446-avg b').textContent = avg.toFixed(1);
  bigStars.innerHTML = stars(avg); bigStars.setAttribute('aria-label', T.rated(avg.toFixed(1)));
  root.querySelector('.u446-avg small').textContent = T.total(fmt(total));
  var barsBox = root.querySelector('.u446-bars'), list = root.querySelector('.u446-ls'), head = root.querySelector('.u446-fl b');
  var ph = root.querySelector('.u446-ph'), empty = root.querySelector('.u446-empty');
  var star = 0, photo = false, helped = {};
  var bars = DIST.map(function (n, k) {
    var s = 5 - k, b = document.createElement('button');
    b.type = 'button'; b.className = 'u446-bar'; b.setAttribute('aria-pressed', 'false'); b.setAttribute('aria-label', T.barLabel(s, fmt(n)));
    b.innerHTML = '<span aria-hidden="true"></span><i aria-hidden="true"></i><span aria-hidden="true"></span>';
    b.firstChild.textContent = T.star(s); b.lastChild.textContent = fmt(n);
    b.querySelector('i').style.setProperty('--p', 0);
    b.addEventListener('click', function () { star = star === s ? 0 : s; draw(); });
    barsBox.appendChild(b);
    return b;
  });
  requestAnimationFrame(function () { requestAnimationFrame(function () { bars.forEach(function (b, k) { b.querySelector('i').style.setProperty('--p', (DIST[k] / DIST[0]).toFixed(3)); }); }); });
  function draw() {
    bars.forEach(function (b, k) { b.setAttribute('aria-pressed', String(5 - k === star)); });
    barsBox.classList.toggle('is-f', !!star);
    ph.setAttribute('aria-pressed', String(photo));
    var rows = REV.map(function (r, i) { return { r: r, i: i }; }).filter(function (o) { return (!star || o.r[2] === star) && (!photo || o.r[3]); });
    head.textContent = T.head(star, photo, rows.length);
    list.innerHTML = '';
    rows.forEach(function (o, n) {
      var r = o.r, li = document.createElement('li');
      li.className = 'u446-rv'; li.style.setProperty('--i', n);
      li.innerHTML = '<div class="u446-rt"><span class="u446-stars" role="img"></span><b></b><span class="u446-dt"></span></div><p></p>' +
        (r[3] ? '<div class="u446-pics">' + [0, 1].map(function (j) { var c = PICS[(o.i + j) % PICS.length]; return '<i role="img" style="--g:linear-gradient(140deg,' + c[0] + ',' + c[1] + ')"></i>'; }).join('') + '</div>' : '') +
        '<button type="button" class="u446-help"><svg viewBox="0 0 14 14" aria-hidden="true"><path d="M4.5 6.2 6.6 2c.9 0 1.5.7 1.4 1.6l-.3 2h3.1c.8 0 1.4.8 1.2 1.6l-.9 3.6a1.3 1.3 0 0 1-1.3 1H4.5zM4.5 6.2H2.3v5.6h2.2"/></svg><span></span></button>';
      var st = li.querySelector('.u446-stars'); st.innerHTML = stars(r[2]); st.setAttribute('aria-label', T.rated(r[2]));
      li.querySelector('.u446-rt b').textContent = r[0];
      li.querySelector('.u446-dt').textContent = r[1];
      li.querySelector('p').textContent = r[5];
      [].slice.call(li.querySelectorAll('.u446-pics i')).forEach(function (pic) { pic.setAttribute('aria-label', T.photo); });
      var hb = li.querySelector('.u446-help'), hs = hb.querySelector('span');
      function syncHelp() { var on = !!helped[o.i]; hb.setAttribute('aria-pressed', String(on)); hs.textContent = T.help + ' ' + (r[4] + (on ? 1 : 0)); }
      hb.addEventListener('click', function () { helped[o.i] = !helped[o.i]; syncHelp(); });
      syncHelp();
      list.appendChild(li);
    });
    empty.hidden = rows.length > 0; list.hidden = !rows.length;
    list.scrollTop = 0;
  }
  ph.addEventListener('click', function () { photo = !photo; draw(); });
  empty.querySelector('button').addEventListener('click', function () { star = 0; photo = false; draw(); });
  draw();
})();
</script>
