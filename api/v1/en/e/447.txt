<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var T = {
    L: ['Light', 'Dark', 'System'],
    now: function (m) { return 'System · ' + m + ' now'; },
    set: function (m) { return 'Theme: ' + m; }
  };
  var seg = root.querySelector('.u447-seg'), opts = [].slice.call(seg.querySelectorAll('.u447-opt'));
  var pv = root.querySelector('.u447-pv'), em = root.querySelector('.u447-pl em'), live = root.querySelector('.u447-live');
  var mq = window.matchMedia('(prefers-color-scheme: dark)'), cur = 2;
  function apply() {
    var dark = cur === 1 || (cur === 2 && mq.matches);
    pv.setAttribute('data-mode', dark ? 'dark' : 'light');
    em.textContent = cur === 2 ? T.now(T.L[dark ? 1 : 0]) : T.L[cur];
  }
  function pick(i, focus) {
    cur = i;
    opts.forEach(function (o, k) { o.setAttribute('aria-checked', String(k === i)); o.tabIndex = k === i ? 0 : -1; });
    seg.style.setProperty('--k', i);
    apply();
    if (focus) opts[i].focus();
  }
  opts.forEach(function (o, i) {
    o.addEventListener('click', function () { if (cur !== i) { pick(i); live.textContent = T.set(T.L[i]); } });
    o.addEventListener('keydown', function (e) {
      var d = e.key === 'ArrowRight' || e.key === 'ArrowDown' ? 1 : e.key === 'ArrowLeft' || e.key === 'ArrowUp' ? -1 : 0;
      if (!d) return;
      e.preventDefault();
      var n = (i + d + 3) % 3; pick(n, true); live.textContent = T.set(T.L[n]);
    });
  });
  function onChange() { if (!root.isConnected) { if (mq.removeEventListener) mq.removeEventListener('change', onChange); return; } if (cur === 2) apply(); }
  if (mq.addEventListener) mq.addEventListener('change', onChange);
  pick(2);
})();
</script>
