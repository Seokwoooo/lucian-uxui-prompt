What it does: Six months of results for four regions, shown as a block city seen from an angle. Blocks rise row by row from the front when it first appears, and switching between orders and visits with the chips above re-levels the heights in a wave. How to use it: Hover over the blocks, and switch the metric with the chips above

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var sv = root.querySelector('.u202-sv'), tip = root.querySelector('.u202-tip'), tl = root.querySelector('.u202-tl'), tv = root.querySelector('.u202-tv');
  var chips = root.querySelectorAll('.u202-chip'), lg = root.querySelector('.u202-lg');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var REG = ['Seoul', 'Busan', 'Daegu', 'Gwangju'];
  var MON = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun'];
  var UNIT = [' orders', ' visits'];
  var COL = [['#7C8CFF', '#5A69E6', '#4352C4'], ['#4FD1C5', '#2FB3A8', '#22948A'], ['#FFB86B', '#F2994A', '#D97F2E'], ['#F78FB3', '#E26D98', '#C4537D']];
  var DATA = [
    [[42, 55, 61, 58, 74, 88], [30, 34, 41, 47, 45, 52], [22, 25, 24, 31, 36, 40], [15, 19, 23, 21, 27, 33]],
    [[66, 70, 64, 80, 85, 92], [48, 44, 57, 60, 66, 63], [35, 41, 38, 46, 44, 55], [28, 26, 34, 38, 41, 47]]
  ];
  var NS = 'http://www.w3.org/2000/svg', OX = 149, OY = 160, DX = 21, DY = 12.2, FW = 0.72, metric = 0;
  function iso(c, r) { return [OX + (c - r) * DX, OY + (c + r) * DY]; }
  var B = [];
  for (var r = 0; r < 4; r++) for (var c = 0; c < 6; c++) B.push({ r: r, c: c, h: 0, from: 0, to: 0, t0: 0 });
  B.sort(function (a, b) { return (a.r + a.c) - (b.r + b.c) || a.c - b.c; });
  var floor = document.createElementNS(NS, 'path');
  var f0 = iso(-0.6, -0.6), f1 = iso(5.6, -0.6), f2 = iso(5.6, 3.6), f3 = iso(-0.6, 3.6);
  floor.setAttribute('d', 'M' + f0 + 'L' + f1 + 'L' + f2 + 'L' + f3 + 'Z'); floor.setAttribute('fill', 'rgba(30,36,64,.05)'); floor.setAttribute('stroke', 'rgba(30,36,64,.1)');
  sv.appendChild(floor);
  B.forEach(function (b, i) {
    var g = document.createElementNS(NS, 'g'); g.setAttribute('class', 'u202-b');
    ['top', 'left', 'right'].forEach(function (k, j) { var p = document.createElementNS(NS, 'path'); p.setAttribute('fill', COL[b.r][j]); g.appendChild(p); b[k] = p; });
    g.addEventListener('pointerenter', function () { focus(b); });
    g.addEventListener('pointerleave', function () { focus(null); });
    b.g = g; sv.appendChild(g);
  });
  REG.forEach(function (n, i) { var s = document.createElement('span'); var dot = document.createElement('i'); dot.style.background = COL[i][0]; s.appendChild(dot); s.appendChild(document.createTextNode(n)); lg.appendChild(s); });
  function shape(b) {
    var hw = FW / 2, h = b.h * 1.25, p = function (dc, dr, up) { var q = iso(b.c + dc, b.r + dr); return (q[0]).toFixed(1) + ',' + (q[1] - up).toFixed(1); };
    b.top.setAttribute('d', 'M' + p(-hw, -hw, h) + 'L' + p(hw, -hw, h) + 'L' + p(hw, hw, h) + 'L' + p(-hw, hw, h) + 'Z');
    b.left.setAttribute('d', 'M' + p(-hw, hw, h) + 'L' + p(hw, hw, h) + 'L' + p(hw, hw, 0) + 'L' + p(-hw, hw, 0) + 'Z');
    b.right.setAttribute('d', 'M' + p(hw, -hw, h) + 'L' + p(hw, hw, h) + 'L' + p(hw, hw, 0) + 'L' + p(hw, -hw, 0) + 'Z');
  }
  var cur = null;
  function focus(b) {
    cur = b; root.classList.toggle('u202-dim', !!b);
    B.forEach(function (o) { o.g.classList.toggle('is-on', o === b); o.g.classList.toggle('is-line', !!b && o !== b && (o.r === b.r || o.c === b.c)); });
    place();
  }
  function place() {
    if (!cur) { tip.classList.remove('is-on'); return; }
    var q = iso(cur.c, cur.r), k = sv.getBoundingClientRect().width / 340 || 1;
    tip.style.left = q[0] + 'px'; tip.style.top = (q[1] - cur.h * 1.25 - 6) + 'px';
    tl.textContent = REG[cur.r] + ' · ' + MON[cur.c];
    tv.textContent = Math.round(DATA[metric][cur.r][cur.c] * 37).toLocaleString() + UNIT[metric];
    tip.classList.add('is-on');
  }
  function retarget(first) {
    var now = performance.now();
    B.forEach(function (b) { b.from = b.h; b.to = DATA[metric][b.r][b.c]; b.t0 = now + (first ? (8 - b.r - b.c) * 70 : (b.c + b.r) * 45); });
    if (rm) { B.forEach(function (b) { b.h = b.to; shape(b); }); place(); return; }
    run();
  }
  function ease(t) { return 1 - Math.pow(1 - t, 3) * Math.cos(t * 4.2) * (1 - t * 0.2); }
  var raf = 0;
  function frame(now) {
    raf = 0;
    if (!root.isConnected) return;
    var busy = false;
    B.forEach(function (b) {
      var t = Math.max(0, Math.min(1, (now - b.t0) / 650));
      b.h = b.from + (b.to - b.from) * (t <= 0 ? 0 : ease(t)); if (t < 1) busy = true; else b.h = b.to;
      shape(b);
    });
    place();
    if (busy) raf = requestAnimationFrame(frame);
  }
  function run() { if (!raf) raf = requestAnimationFrame(frame); }
  [].forEach.call(chips, function (ch, i) {
    ch.addEventListener('click', function () { if (metric === i) return; metric = i; [].forEach.call(chips, function (o, j) { o.setAttribute('aria-pressed', j === i ? 'true' : 'false'); }); retarget(false); });
  });
  B.forEach(shape);
  var shown = false;
  if ('IntersectionObserver' in window && !rm) new IntersectionObserver(function (es) { if (es[0].isIntersecting && !shown) { shown = true; retarget(true); } }).observe(root);
  else retarget(true);
})();
</script>
