How to use it: Press … to open the hidden steps, then press a step in the middle
Footprint: fixed size, square

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var ol = root.querySelector('.u415-ol');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var SEP = '<span class="u415-sep" aria-hidden="true"><svg viewBox="0 0 10 10"><path d="M3.8 2 6.8 5l-3 3"/></svg></span>';
  var HOME = '<svg viewBox="0 0 16 16" aria-hidden="true"><path d="M2.5 7 8 2.5 13.5 7v6.5H9.8V10H6.2v3.5H2.5z"/></svg>';
  var PATH = [null, 'Teams', 'Marketing', '2026', 'Launch event'];
  var cur = 4, open = false;
  var lis = PATH.map(function (name, i) {
    var li = document.createElement('li');
    li.className = 'u415-li';
    li.innerHTML = '<span><span class="u415-in">' + (i ? SEP : '') + '</span></span>';
    var inn = li.querySelector('.u415-in');
    if (i === 0) inn.insertAdjacentHTML('beforeend', '<a class="u415-a hm" href="#" aria-label="Home">' + HOME + '</a>');
    else { var a = document.createElement('a'); a.className = 'u415-a'; a.href = '#'; a.textContent = name; inn.appendChild(a); var c = document.createElement('span'); c.className = 'u415-cur'; c.setAttribute('aria-current', 'page'); c.textContent = name; c.hidden = true; inn.appendChild(c); }
    var link = inn.querySelector('a');
    link.addEventListener('click', function (e) { e.preventDefault(); go(i); });
    ol.appendChild(li);
    return li;
  });
  var moreLi = document.createElement('li');
  moreLi.className = 'u415-li';
  moreLi.innerHTML = '<span><span class="u415-in">' + SEP + '<button class="u415-more" type="button" aria-label="Show 2 hidden steps" aria-expanded="false">…</button></span></span>';
  ol.insertBefore(moreLi, lis[1]);
  var more = moreLi.querySelector('button');
  function shut(li, v) { li.classList.toggle('is-shut', v); li.inert = v; }
  function draw() {
    lis.forEach(function (li, i) {
      var hide = i > cur || (!open && cur >= 4 && (i === 1 || i === 2));
      shut(li, hide);
      if (i) { li.querySelector('a').hidden = i === cur; li.querySelector('.u415-cur').hidden = i !== cur; }
    });
    shut(moreLi, open || cur < 4);
    more.setAttribute('aria-expanded', String(open));
  }
  more.addEventListener('click', function () {
    open = true; draw();
    var a = lis[1].querySelector('a'); a.focus({ preventScroll: true });
  });
  function go(i) {
    if (i === cur) return;
    cur = i; draw();
    var c = i ? lis[i].querySelector('.u415-cur') : null;
    if (!c) lis[0].querySelector('a').focus({ preventScroll: true });
  }
  lis.forEach(function (li) { li.style.transition = 'none'; });
  moreLi.style.transition = 'none';
  draw();
  void ol.offsetWidth;
  lis.forEach(function (li) { li.style.transition = ''; });
  moreLi.style.transition = '';
})();
</script>
