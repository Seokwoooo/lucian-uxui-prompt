Element: Breadcrumb
ID: 415 · Category: Navigation · Tags: breadcrumb, navigation, overflow, truncate, path, file-browser
What it does: A path from the home icon to the current page. When it gets long, the middle steps fold into a single … button that opens the hidden steps in order. The current page ends the trail in bold text, not a link, and steps grow and shrink smoothly as you open folders or jump back.
How to use it: Open a folder or document in the list, then press … to jump back to a hidden step
Footprint: fixed size, square
Needs: tokens.css (the --ux-* theme tokens and the .ux base). Map the tokens to the project's theme variables.
Live demo: https://lucianlabs.dev/uxui-prompt/#415

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var ol = root.querySelector('.u415-ol'), mn = root.querySelector('.u415-mn'), pg = root.querySelector('.u415-pg');
  var uid = 'u415-' + Math.random().toString(36).slice(2, 7);
  function F(n, c) { return { n: n, c: c || [] }; }
  function D(n, d) { return { n: n, d: d }; }
  var TREE = F('Home', [
    F('Marketing', [
      F('2026 Campaigns', [
        F('Spring', [
          F('Launch event', [D('Proposal v3', 'Nov 21'), D('Budget', 'Nov 18'), D('Guest list', 'Nov 20'), F('Event photos')]),
          F('Social posts'), D('Results recap', 'Nov 12'), D('Press release', 'Nov 8')]),
        F('Summer'), F('Fall'), D('Campaign calendar', 'Oct 30')]),
      F('Brand guide'), F('Meeting notes'), D('Monthly report', 'Nov 3')]),
    F('Product'), F('People'), D('Company deck', 'Sep 24')]);
  var ICON = {
    home: '<svg viewBox="0 0 14 14" aria-hidden="true"><path d="M2 6.2 7 2l5 4.2V12H8.6V8.8H5.4V12H2z"/></svg>',
    sep: '<span class="u415-sep" aria-hidden="true"><svg viewBox="0 0 10 10"><path d="M3.8 2 6.8 5l-3 3"/></svg></span>',
    fd: '<svg class="ic" viewBox="0 0 16 16" aria-hidden="true"><path class="fd" d="M1.5 4a1.5 1.5 0 0 1 1.5-1.5h3l1.5 1.5H13a1.5 1.5 0 0 1 1.5 1.5V12A1.5 1.5 0 0 1 13 13.5H3A1.5 1.5 0 0 1 1.5 12z"/><path class="fd2" d="M1.5 6.5h13V12a1.5 1.5 0 0 1-1.5 1.5H3A1.5 1.5 0 0 1 1.5 12z"/></svg>',
    dc: '<svg class="ic" viewBox="0 0 16 16" aria-hidden="true"><path class="dc" d="M4 1.8h5.4L12.5 5v9.2H4z"/><path class="dc" d="M9.2 2v3.2h3.2M6 8.2h4.2M6 10.8h3"/></svg>',
    cv: '<svg class="cv" viewBox="0 0 10 10" aria-hidden="true"><path d="M3.8 2 6.8 5l-3 3"/></svg>'
  };
  var path = [TREE, TREE.c[0], TREE.c[0].c[0], TREE.c[0].c[0].c[0], TREE.c[0].c[0].c[0].c[0]];
  var shown = [], mOpen = false, mCur = -1, hid = [];
  var MAX = 4;
  function cnt(n) { return n ? n + (n === 1 ? ' item' : ' items') : 'Empty'; }
  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function go(depth, node, dir) {
    path = path.slice(0, depth + 1);
    if (node) path.push(node);
    menu(false);
    render(dir);
  }
  function render(dir) {
    var n = path.length;
    var vis = n > MAX ? [0, -1, n - 2, n - 1] : path.map(function (_, i) { return i; });
    hid = n > MAX ? path.slice(1, n - 2) : [];
    var keys = vis.map(function (i) { return i < 0 ? 'more' : i + ':' + path[i].n; });
    ol.innerHTML = '';
    vis.forEach(function (i, k) {
      var li = document.createElement('li');
      if (shown.indexOf(keys[k]) < 0 && shown.length) li.className = 'is-new';
      var sep = k ? ICON.sep : '';
      if (i < 0) {
        li.innerHTML = sep + '<button class="u415-more" type="button" aria-haspopup="menu" aria-expanded="false">…</button>';
        var b = li.querySelector('button');
        b.setAttribute('aria-label', 'Show ' + hid.length + ' hidden steps');
        b.id = uid + '-mb'; b.setAttribute('aria-controls', uid + '-m');
        b.addEventListener('click', function () { menu(!mOpen); });
        b.addEventListener('keydown', function (e) { if (e.key === 'ArrowDown') { e.preventDefault(); menu(true); mark(0); } });
      } else if (i === n - 1) {
        li.innerHTML = sep + '<span class="u415-cur" aria-current="page"></span>';
        li.lastChild.textContent = path[i].n;
      } else if (i === 0) {
        li.innerHTML = '<a class="hm" href="#">' + ICON.home + '</a>';
        li.lastChild.setAttribute('aria-label', path[0].n);
        li.lastChild.addEventListener('click', function (e) { e.preventDefault(); go(0, null, -1); });
      } else {
        li.innerHTML = sep + '<a href="#"><span></span></a>';
        li.querySelector('a span').textContent = path[i].n;
        li.lastChild.addEventListener('click', function (e) { e.preventDefault(); go(i, null, -1); });
      }
      ol.appendChild(li);
    });
    shown = keys;
    mn.id = uid + '-m';
    mn.innerHTML = hid.map(function (h, k) { return '<button class="u415-mi" type="button" role="menuitem" tabindex="-1" style="--lv:' + k + ';--tick:' + (k ? 1 : 0) + '">' + ICON.fd + '<span>' + esc(h.n) + '</span></button>'; }).join('');
    Array.prototype.forEach.call(mn.children, function (b, k) { b.addEventListener('click', function () { go(k + 1, null, -1); var r = pg.querySelector('.u415-row'); if (r) r.focus({ preventScroll: true }); }); });
    page(dir);
  }
  function page(dir) {
    var node = path[path.length - 1];
    var h = '<div class="u415-ph"><span class="tl">' + (node.c ? ICON.fd : ICON.dc) + '</span><div><h3>' + esc(node.n) + '</h3><p>';
    if (node.c) {
      var n = node.c.length;
      h += cnt(n) + '</p></div></div>';
      h += n ? '<ul class="u415-ls">' + node.c.map(function (c, k) {
        var m = c.c ? cnt(c.c.length) : c.d;
        return '<li><button class="u415-row" type="button" data-k="' + k + '">' + (c.c ? ICON.fd : ICON.dc) + '<b>' + esc(c.n) + '</b><small>' + esc(m) + '</small>' + ICON.cv + '</button></li>';
      }).join('') + '</ul>' : '<div class="u415-em">' + 'Nothing in this folder yet' + '</div>';
    } else {
      var d = node.d;
      h += 'Jieun Kim · edited ' + d + '</p></div></div><div class="u415-doc" aria-hidden="true"><i></i><i style="width:92%"></i><i style="width:84%"></i><i style="width:88%"></i><i style="width:56%"></i></div>';
    }
    pg.innerHTML = h;
    pg.querySelectorAll('.u415-row').forEach(function (b) {
      var c = node.c[+b.getAttribute('data-k')];
      b.setAttribute('aria-label', (c.c ? 'Folder' : 'Document') + ' ' + c.n);
      b.addEventListener('click', function () { go(path.length - 1, c, 1); });
    });
    pg.classList.remove('is-fw', 'is-bw');
    if (dir) { void pg.offsetWidth; pg.classList.add(dir > 0 ? 'is-fw' : 'is-bw'); }
  }
  function mark(i) {
    mCur = i;
    Array.prototype.forEach.call(mn.children, function (b, k) { b.classList.toggle('is-cur', k === i); });
    if (i >= 0 && mn.children[i]) mn.children[i].focus({ preventScroll: true });
  }
  function menu(v) {
    var mb = root.querySelector('.u415-more');
    if (!mb) v = false;
    mOpen = v;
    root.classList.toggle('is-mn', v);
    mn.inert = !v;
    if (mb) {
      mb.setAttribute('aria-expanded', String(v));
      if (v) mn.style.left = Math.max(6, mb.offsetLeft + mb.parentNode.offsetLeft - 4) + 'px';
    }
    if (!v) mark(-1);
  }
  mn.addEventListener('keydown', function (e) {
    var n = mn.children.length, to = e.key === 'ArrowDown' ? mCur + 1 : e.key === 'ArrowUp' ? mCur - 1 : e.key === 'Home' ? 0 : e.key === 'End' ? n - 1 : null;
    if (e.key === 'Escape' || e.key === 'Tab') { if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); } menu(false); var mb = root.querySelector('.u415-more'); if (mb && e.key === 'Escape') mb.focus({ preventScroll: true }); return; }
    if (to === null) return;
    e.preventDefault(); mark((to + n) % n);
  });
  mn.addEventListener('pointermove', function () { if (mCur >= 0) mark(-1); });
  root.addEventListener('keydown', function (e) { if (e.key === 'Escape' && mOpen) { e.stopPropagation(); menu(false); } });
  function outside(e) {
    if (!root.isConnected) { document.removeEventListener('pointerdown', outside, true); return; }
    if (mOpen && !mn.contains(e.target) && !e.target.closest('.u415-more')) menu(false);
  }
  document.addEventListener('pointerdown', outside, true);
  render(0);
})();
</script>
