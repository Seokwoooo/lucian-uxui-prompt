<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { switched: 'Switched to ' };
  var av = root.querySelector('.u408-av'), menu = root.querySelector('.u408-menu'), v1 = root.querySelector('.u408-v1'), v2 = root.querySelector('.u408-v2');
  var seg = root.querySelector('.u408-seg'), tms = [].slice.call(root.querySelectorAll('.u408-tm')), toast = root.querySelector('.u408-toast');
  var ws = root.querySelector('.u408-ws'), wis = [].slice.call(root.querySelectorAll('.u408-wi'));
  var act = -1, tt = 0, gen = 0;
  var tl = root.querySelector('.u408-tl'); tl.id = 'u408' + Math.random().toString(36).slice(2, 8); root.querySelector('.u408-th').setAttribute('aria-labelledby', tl.id);
  function items() { return [].slice.call((menu.classList.contains('is-ws') ? v2 : v1).querySelectorAll('[role^="menuitem"]')); }
  function setAct(k, focus) {
    var list = items();
    if (!list.length) return;
    act = (k + list.length) % list.length;
    [].slice.call(menu.querySelectorAll('.is-act')).forEach(function (x) { x.classList.remove('is-act'); });
    list[act].classList.add('is-act');
    if (focus !== false) list[act].focus({ preventScroll: true });
  }
  function open() {
    if (root.classList.contains('is-open')) return;
    gen += 1;
    menu.classList.remove('is-ws'); v1.inert = false; v2.inert = true;
    root.classList.add('is-open'); av.setAttribute('aria-expanded', 'true');
    items().forEach(function (x) { x.tabIndex = -1; });
    setAct(0);
  }
  function close(refocus) {
    root.classList.remove('is-open'); av.setAttribute('aria-expanded', 'false');
    [].slice.call(menu.querySelectorAll('.is-act')).forEach(function (x) { x.classList.remove('is-act'); });
    act = -1;
    if (refocus !== false) av.focus({ preventScroll: true });
  }
  function drill(on) {
    menu.classList.toggle('is-ws', on);
    v1.inert = on; v2.inert = !on;
    items().forEach(function (x) { x.tabIndex = -1; });
    setAct(on ? Math.max(1, wis.findIndex(function (w) { return w.getAttribute('aria-checked') === 'true'; }) + 1) : 0);
  }
  function say(text) {
    toast.textContent = text; toast.classList.add('is-on');
    clearTimeout(tt); tt = setTimeout(function () { if (root.isConnected) toast.classList.remove('is-on'); }, 2000);
  }
  av.addEventListener('click', function () { if (root.classList.contains('is-open')) close(); else open(); });
  av.addEventListener('keydown', function (e) { if (e.key === 'ArrowDown' || e.key === 'ArrowUp') { e.preventDefault(); open(); if (e.key === 'ArrowUp') setAct(-1); } });
  menu.addEventListener('keydown', function (e) {
    var k = e.key;
    if (k === 'ArrowDown') { e.preventDefault(); setAct(act + 1); }
    else if (k === 'ArrowUp') { e.preventDefault(); setAct(act - 1); }
    else if (k === 'Home') { e.preventDefault(); setAct(0); }
    else if (k === 'End') { e.preventDefault(); setAct(-1); }
    else if (k === 'ArrowRight' && items()[act] === ws) { e.preventDefault(); drill(true); }
    else if (k === 'ArrowLeft' && menu.classList.contains('is-ws')) { e.preventDefault(); drill(false); setAct(items().indexOf(ws)); }
    else if (k === 'Escape') { e.preventDefault(); e.stopPropagation(); if (menu.classList.contains('is-ws')) { drill(false); setAct(items().indexOf(ws)); } else close(); }
    else if (k === 'Tab') close(false);
    else if ((k === 'Enter' || k === ' ') && items()[act]) { e.preventDefault(); items()[act].click(); }
  });
  menu.addEventListener('pointermove', function (e) { var it = e.target.closest && e.target.closest('[role^="menuitem"]'); if (it) { var i = items().indexOf(it); if (i >= 0 && i !== act) setAct(i, false); } });
  [].slice.call(menu.querySelectorAll('[data-t]')).forEach(function (b) { b.addEventListener('click', function () { close(); say(b.getAttribute('data-t')); }); });
  ws.addEventListener('click', function () { drill(true); });
  root.querySelector('.u408-back').addEventListener('click', function () { drill(false); setAct(items().indexOf(ws)); });
  wis.forEach(function (w) {
    w.addEventListener('click', function () {
      wis.forEach(function (o) { o.setAttribute('aria-checked', String(o === w)); });
      var c = w.getAttribute('data-c'), n = w.getAttribute('data-n');
      root.style.setProperty('--wc', c);
      root.querySelector('.u408-org span').textContent = n;
      root.querySelector('.u408-org i').textContent = w.querySelector('i').textContent;
      ws.querySelector('i').textContent = w.querySelector('i').textContent;
      root.querySelector('.u408-wn').textContent = n;
      ws.querySelector('small').textContent = ws.querySelector('small').textContent.split(' · ')[0] + ' · ' + w.querySelector('small').textContent;
      close(); say(T.switched + n);
    });
  });
  tms.forEach(function (t, k) {
    t.addEventListener('click', function () {
      tms.forEach(function (o) { o.setAttribute('aria-checked', String(o === t)); });
      seg.style.setProperty('--k', k);
      setAct(items().indexOf(t), false);
    });
  });
  root.addEventListener('pointerdown', function (e) { if (root.classList.contains('is-open') && !menu.contains(e.target) && !av.contains(e.target)) close(false); });
  v2.inert = true;
})();
</script>
