<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var btns = root.querySelectorAll('.u437-b'), tip = root.querySelector('.u437-tip'), tl = root.querySelector('.u437-tl'), kb = root.querySelector('.u437-tip kbd');
  var mac = /Mac|iPhone|iPad/.test(navigator.platform || navigator.userAgent || '');
  var TIPS = [['Bold', 'B'], ['Italic', 'I'], ['Link', 'K']];
  var uid = 'u437' + Math.random().toString(36).slice(2, 7);
  tip.id = uid;
  btns.forEach(function (b, i) { b.setAttribute('aria-label', TIPS[i][0]); b.setAttribute('aria-keyshortcuts', (mac ? 'Meta+' : 'Control+') + TIPS[i][1]); });
  var cur = null, openT = 0, closeT = 0, lastClose = 0, muted = null;
  function place(b) {
    var rr = root.getBoundingClientRect(), br = b.getBoundingClientRect(), k = rr.width / root.offsetWidth || 1;
    var cx = (br.left - rr.left + br.width / 2) / k, top = (br.top - rr.top) / k;
    var w = tip.offsetWidth, h = tip.offsetHeight;
    var x = Math.max(8, Math.min(root.offsetWidth - 8 - w, cx - w / 2));
    tip.style.setProperty('--x', x.toFixed(1) + 'px');
    tip.style.setProperty('--y', (top - h - 10).toFixed(1) + 'px');
    tip.style.setProperty('--ax', (cx - x).toFixed(1) + 'px');
  }
  function show(b, glide) {
    var t = TIPS[Array.prototype.indexOf.call(btns, b)];
    tl.textContent = t[0];
    kb.innerHTML = '<span>' + (mac ? '⌘' : 'Ctrl') + '</span><span>' + t[1] + '</span>';
    tip.classList.toggle('is-glide', !!glide);
    place(b);
    if (cur) { cur.removeAttribute('aria-describedby'); cur.classList.remove('is-on'); }
    cur = b;
    b.setAttribute('aria-describedby', tip.id);
    b.classList.add('is-on');
    tip.classList.add('is-on');
  }
  function hide() {
    clearTimeout(openT);
    if (!cur) return;
    cur.removeAttribute('aria-describedby'); cur.classList.remove('is-on');
    cur = null;
    tip.classList.remove('is-on');
    lastClose = Date.now();
  }
  function want(b) {
    if (b === muted) return;
    clearTimeout(closeT); clearTimeout(openT);
    if (cur === b) return;
    if (cur || Date.now() - lastClose < 300) show(b, !!cur);
    else openT = setTimeout(function () { if (root.isConnected) show(b, false); }, 400);
  }
  function unwant(b) {
    if (muted === b) muted = null;
    clearTimeout(openT); clearTimeout(closeT);
    closeT = setTimeout(hide, 100);
  }
  btns.forEach(function (b) {
    b.addEventListener('pointerenter', function (e) { if (e.pointerType !== 'touch') want(b); });
    b.addEventListener('pointerleave', function () { unwant(b); });
    b.addEventListener('focus', function () { if (b.matches(':focus-visible')) want(b); });
    b.addEventListener('blur', function () { unwant(b); });
    b.addEventListener('pointerdown', function () { muted = b; hide(); });
    b.addEventListener('click', function () { if (b.hasAttribute('aria-pressed')) b.setAttribute('aria-pressed', b.getAttribute('aria-pressed') === 'true' ? 'false' : 'true'); });
  });
  setTimeout(function () { if (root.isConnected && !cur) show(btns[0], false); }, 300);
  root.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && cur) { e.preventDefault(); e.stopPropagation(); muted = cur; hide(); }
  });
})();
</script>
