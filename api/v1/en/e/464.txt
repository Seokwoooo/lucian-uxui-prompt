<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var vp = root.querySelector('.u464-vp'), st = root.querySelector('.u464-st'), bb = root.querySelector('.u464-bb'), sc = root.querySelector('.u464-sc');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var GAP = 20, N = 6, W = 272, MID = W / 2;
  var uid = 'u464' + Math.random().toString(36).slice(2, 7);
  root.querySelectorAll('svg [id]').forEach(function (n) {
    var o = n.id;
    n.id = uid + o;
    root.querySelectorAll('[fill="url(#' + o + ')"]').forEach(function (m) { m.setAttribute('fill', 'url(#' + uid + o + ')'); });
  });
  var pos = 0, vel = 0, mode = 'rest', target = 0, raf = 0, drag = null, shown = 0;
  for (var i = -N; i <= N; i++) {
    var t = document.createElement('i');
    t.className = 'u464-t' + (i % 3 === 0 ? ' is-maj' : '');
    t.style.left = ((i + N) * GAP) + 'px';
    if (i % 3 === 0) { var b = document.createElement('b'); b.textContent = i === 0 ? '0' : (i > 0 ? '+' : '−') + Math.abs(i / 3); t.appendChild(b); }
    st.appendChild(t);
  }
  function fmt(n) { var v = Math.round(n / 3 * 10) / 10; return (v > 0 ? '+' : v < 0 ? '−' : '') + Math.abs(v).toFixed(1); }
  function paint() {
    st.style.setProperty('--u464-off', (MID - (pos + N) * GAP).toFixed(2) + 'px');
    var ev = Math.max(-2.3, Math.min(2.3, pos / 3));
    var br = 1 + ev * (ev > 0 ? 0.2 : 0.21), ct = ev > 0 ? 1 - ev * 0.08 : 1 - ev * 0.04, sa = ev > 0 ? 1 - ev * 0.07 : 1 - ev * 0.05;
    sc.style.filter = 'brightness(' + br.toFixed(3) + ') contrast(' + ct.toFixed(3) + ') saturate(' + sa.toFixed(3) + ')';
    var n = Math.max(-N, Math.min(N, Math.round(pos)));
    if (n !== shown) {
      shown = n;
      bb.textContent = fmt(n);
      vp.setAttribute('aria-valuenow', String(Math.round(n / 3 * 10) / 10));
      vp.setAttribute('aria-valuetext', fmt(n) + ' EV');
      if (!rm) { bb.classList.remove('is-tick'); void bb.offsetWidth; bb.classList.add('is-tick'); }
    }
  }
  function frame() {
    raf = 0;
    if (!root.isConnected) return;
    for (var i = 0; i < 4; i++) {
      if (mode === 'coast') {
        vel *= Math.pow(0.0022, 1 / 240);
        if (pos < -N) vel += (-N - pos) * 120 / 240; else if (pos > N) vel += (N - pos) * 120 / 240;
        pos += vel / 240;
        if (Math.abs(vel) < 3.2) { mode = 'snap'; target = Math.max(-N, Math.min(N, Math.round(pos + vel * 0.08))); }
      } else if (mode === 'snap') {
        var a = 260 * (target - pos) - 17 * vel; vel += a / 240; pos += vel / 240;
        if (Math.abs(target - pos) < 0.002 && Math.abs(vel) < 0.01) { pos = target; vel = 0; mode = 'rest'; }
      }
    }
    paint();
    if (mode !== 'rest') raf = requestAnimationFrame(frame);
  }
  function run() { if (!raf) raf = requestAnimationFrame(frame); }
  function snapTo(n) {
    target = Math.max(-N, Math.min(N, n));
    if (rm) { pos = target; vel = 0; mode = 'rest'; paint(); return; }
    mode = 'snap'; run();
  }
  vp.addEventListener('pointerdown', function (e) {
    if (e.button !== 0) return;
    var r = root.getBoundingClientRect();
    drag = { x: e.clientX, p0: pos, k: root.offsetWidth / (r.width || 1), lx: e.clientX, lt: performance.now(), v: 0, moved: false };
    mode = 'drag'; vel = 0;
    try { vp.setPointerCapture(e.pointerId); } catch (er) {  }
  });
  vp.addEventListener('pointermove', function (e) {
    if (!drag) return;
    var now = performance.now(), dt = Math.max(1, now - drag.lt);
    var dx = (e.clientX - drag.x) * drag.k;
    if (Math.abs(dx) > 3) drag.moved = true;
    drag.v = drag.v * 0.5 + (-(e.clientX - drag.lx) * drag.k / GAP / dt * 1000) * 0.5;
    drag.lx = e.clientX; drag.lt = now;
    var raw = drag.p0 - dx / GAP;
    pos = raw < -N ? -N + (raw + N) * 0.35 : raw > N ? N + (raw - N) * 0.35 : raw;
    paint();
  });
  function up(e) {
    if (!drag) return;
    var d = drag;
    drag = null;
    if (!d.moved) {
      var r = vp.getBoundingClientRect(), x = (e.clientX - r.left) * d.k;
      snapTo(Math.round(pos + (x - MID) / GAP));
      return;
    }
    if (rm) { snapTo(Math.round(pos)); return; }
    vel = Math.max(-60, Math.min(60, performance.now() - d.lt > 90 ? 0 : d.v));
    mode = 'coast'; run();
  }
  vp.addEventListener('pointerup', up);
  vp.addEventListener('pointercancel', function () { if (drag) { drag = null; snapTo(Math.round(pos)); } });
  vp.addEventListener('dblclick', function () { snapTo(0); });
  vp.addEventListener('keydown', function (e) {
    var k = e.key, n = mode === 'snap' ? target : Math.round(pos);
    if (k === 'ArrowRight' || k === 'ArrowUp') n += 1;
    else if (k === 'ArrowLeft' || k === 'ArrowDown') n -= 1;
    else if (k === 'PageUp') n += 3;
    else if (k === 'PageDown') n -= 3;
    else if (k === 'Home') n = -N;
    else if (k === 'End') n = N;
    else if (k === '0') n = 0;
    else return;
    e.preventDefault();
    snapTo(n);
  });
  paint();
})();
</script>
