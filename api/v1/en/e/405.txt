<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { again: 'Resend', wait: function (s) { return 'Resend in ' + s + 's'; }, sent: 'Reset email sent', bad: 'Check your email address' };
  var COOL = 20;
  var form = root.querySelector('.u405-a'), b = root.querySelector('.u405-b'), f = root.querySelector('.u405-f'), inp = root.querySelector('.u405-in');
  var send = root.querySelector('.u405-send'), plane = root.querySelector('.u405-plane'), em = root.querySelector('.u405-em');
  var re = root.querySelector('.u405-re'), rl = root.querySelector('.u405-rl'), pr = root.querySelector('.u405-ring .pr'), num = root.querySelector('.u405-ring em'), live = root.querySelector('.u405-live');
  var uid = 'u405' + Math.random().toString(36).slice(2, 7), tick = 0, left = 0, gen = 0;
  inp.id = uid; f.querySelector('label').htmlFor = uid;
  function later(fn, ms) { var g = gen; setTimeout(function () { if (root.isConnected && g === gen) fn(); }, ms); }
  function up() { f.classList.toggle('is-up', f.classList.contains('is-f') || !!inp.value); }
  root.addEventListener('pointerdown', function (e) { f.classList.toggle('is-f', f.contains(e.target)); up(); });
  inp.addEventListener('focus', function () { f.classList.add('is-f'); up(); });
  inp.addEventListener('blur', function () { f.classList.remove('is-f'); up(); });
  inp.addEventListener('input', function () { f.classList.remove('is-err'); inp.removeAttribute('aria-invalid'); up(); });
  function fly(from) {
    if (rm || !plane.animate) return;
    var box = root.querySelector('.u405-box'), r = from.getBoundingClientRect(), bb = box.getBoundingClientRect(), k = box.offsetWidth / (bb.width || 1);
    var x = (r.left + r.width / 2 - bb.left - bb.width / 2) * k, y = (r.top + r.height / 2 - bb.top - bb.height / 2) * k;
    plane.animate([
      { opacity: 0, transform: 'translate(' + x + 'px,' + y + 'px) rotate(10deg) scale(.6)' },
      { opacity: 1, transform: 'translate(' + (x + 10) + 'px,' + (y - 18) + 'px) rotate(-6deg) scale(1.1)', offset: .2 },
      { opacity: 1, transform: 'translate(' + (x + 70) + 'px,' + (y - 70) + 'px) rotate(-18deg) scale(1)', offset: .6 },
      { opacity: 0, transform: 'translate(' + (x + 150) + 'px,' + (y - 150) + 'px) rotate(-30deg) scale(.55)' }
    ], { duration: 1000, easing: 'cubic-bezier(.3,.1,.4,1)' });
  }
  function cool() {
    clearInterval(tick); left = COOL;
    re.disabled = true; re.classList.remove('is-ready');
    pr.style.transition = 'none'; pr.style.strokeDashoffset = '0'; void pr.offsetWidth; pr.style.transition = '';
    num.textContent = left; rl.textContent = T.again; re.setAttribute('aria-label', T.wait(left));
    tick = setInterval(function () {
      if (!root.isConnected) { clearInterval(tick); return; }
      left -= 1;
      pr.style.strokeDashoffset = (62.8 * (1 - left / COOL)).toFixed(1);
      num.textContent = Math.max(0, left);
      if (left <= 0) { clearInterval(tick); re.disabled = false; re.classList.add('is-ready'); re.removeAttribute('aria-label'); pr.style.strokeDashoffset = '0'; }
      else re.setAttribute('aria-label', T.wait(left));
    }, 1000);
  }
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    var v = inp.value.trim();
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v)) {
      f.classList.remove('is-shake'); void f.offsetWidth; f.classList.add('is-err', 'is-shake');
      inp.setAttribute('aria-invalid', 'true'); live.textContent = T.bad; inp.focus(); return;
    }
    gen += 1; f.classList.remove('is-f'); up();
    fly(send);
    em.textContent = v;
    later(function () {
      root.classList.add('is-sent'); form.inert = true; b.inert = false;
      live.textContent = T.sent; cool();
    }, rm ? 0 : 420);
  });
  re.addEventListener('click', function () { if (re.disabled) return; fly(re); live.textContent = T.sent; cool(); });
  root.querySelector('.u405-chg').addEventListener('click', function () {
    gen += 1; clearInterval(tick);
    root.classList.remove('is-sent'); form.inert = false; b.inert = true;
    later(function () { inp.focus({ preventScroll: true }); inp.select(); }, rm ? 0 : 300);
  });
  b.inert = true;
})();
</script>
