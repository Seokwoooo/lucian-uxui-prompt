<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var c = q('.u455-c'), full = q('.u455-full'), line = q('.u455-line'), lt = q('.u455-lt'), tag = q('.u455-tag');
  var cb = q('.u455-cb'), ok = q('.u455-ok'), no = q('.u455-no');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { run: 'Running…', ok: '12 tests passed', no: 'Denied', always: 'Always' };
  var W = 296, H = 44, gen = 0;
  tag.textContent = T.always;
  function later(fn, ms) { var g = gen; setTimeout(function () { if (root.isConnected && g === gen) fn(); }, ms); }
  function fitLine() { c.style.width = Math.ceil(line.offsetWidth) + 'px'; }
  function swap(text) {
    if (rm || !lt.animate) { lt.textContent = text; fitLine(); return; }
    lt.animate([{ opacity: 1, transform: 'none' }, { opacity: 0, transform: 'translateY(-6px)' }], { duration: 140, easing: 'ease-in' }).onfinish = function () {
      if (!root.isConnected) return;
      lt.textContent = text;
      fitLine();
      lt.animate([{ opacity: 0, transform: 'translateY(6px)' }, { opacity: 1, transform: 'none' }], { duration: 320, easing: 'cubic-bezier(.3,1.4,.5,1)' });
    };
  }
  function collapse(state) {
    gen += 1;
    var had = c.contains(document.activeElement);
    c.style.width = c.offsetWidth + 'px';
    c.style.height = c.offsetHeight + 'px';
    void c.offsetWidth;
    c.dataset.s = state;
    full.inert = true;
    lt.textContent = state === 'no' ? T.no : T.run;
    tag.hidden = !(cb.checked && state !== 'no');
    c.style.height = H + 'px';
    fitLine();
    if (had) { c.tabIndex = -1; c.focus({ preventScroll: true }); }
  }
  function comeBack() {
    gen += 1;
    var had = document.activeElement === c;
    cb.checked = false;
    c.dataset.s = 'ask';
    full.inert = false;
    c.style.width = W + 'px';
    c.style.height = full.offsetHeight + 'px';
    if (had) ok.focus({ preventScroll: true });
    c.removeAttribute('tabindex');
    later(function () { c.style.width = ''; c.style.height = ''; }, 650);
  }
  ok.addEventListener('click', function () {
    if (c.dataset.s !== 'ask') return;
    collapse('run');
    later(function () { c.dataset.s = 'ok'; swap(T.ok); }, 1500);
    later(comeBack, 3600);
  });
  no.addEventListener('click', function () {
    if (c.dataset.s !== 'ask') return;
    collapse('no');
    later(comeBack, 2800);
  });
})();
</script>
