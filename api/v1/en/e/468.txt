(function () {
  var root = document.currentScript.previousElementSibling;
  var pd = root.querySelector('.u468-pd'), cv = root.querySelector('.u468-cv'), sc = root.querySelector('.u468-sc'), cl = root.querySelector('.u468-cl'), ok = root.querySelector('.u468-ok'), okL = ok.querySelector('span'), live = root.querySelector('.u468-live'), gp = root.querySelector('.u468-gp');

  var W = 280, H = 158, DPR = Math.min(3, Math.max(2, window.devicePixelRatio || 1));
  var cx = cv.getContext('2d'), sx = sc.getContext('2d');
  [cv, sc].forEach(function (c) { c.width = W * DPR; c.height = H * DPR; });
  cx.scale(DPR, DPR); sx.scale(DPR, DPR);
  var strokes = [], cur = null, done = false, ink = '', demoT = 0;
  var MAXW = 3.8, MINW = 0.9;
  function colorOf(c) { return getComputedStyle(c).color; }
  function dots(g, x0, y0, w0, x1, y1, w1, cxp, cyp) {
    var len = Math.hypot(cxp - x0, cyp - y0) + Math.hypot(x1 - cxp, y1 - cyp), n = Math.max(1, Math.ceil(len / 0.45));
    g.beginPath();
    for (var i = 0; i <= n; i++) {
      var t = i / n, a = (1 - t) * (1 - t), b = 2 * (1 - t) * t, c = t * t;
      var x = a * x0 + b * cxp + c * x1, y = a * y0 + b * cyp + c * y1, r = (w0 + (w1 - w0) * t) / 2;
      g.moveTo(x + r, y); g.arc(x, y, r, 0, Math.PI * 2);
    }
    g.fill();
  }
  function seg(g, s, k) {
    var p = s[k], q = s[k - 1];
    if (k === 0) { g.beginPath(); g.arc(p.x, p.y, p.w / 2, 0, Math.PI * 2); g.fill(); return; }

    var o = s[k - 2];
    dots(g, (o.x + q.x) / 2, (o.y + q.y) / 2, (o.w + q.w) / 2, (q.x + p.x) / 2, (q.y + p.y) / 2, (q.w + p.w) / 2, q.x, q.y);
  }
  function redraw() {
    cx.clearRect(0, 0, W, H);
    cx.fillStyle = ink = colorOf(cv);
    strokes.forEach(function (s) { for (var k = 0; k < s.length; k++) seg(cx, s, k); });
  }
  function add(x, y, t, pr) {
    var l = cur[cur.length - 1], w = MAXW * 0.7;
    if (l) {
      var d = Math.hypot(x - l.x, y - l.y);
      if (d < 0.8) return;
      var v = d / Math.max(1, t - l.t);
      var tw = Math.max(MINW, Math.min(MAXW, MAXW / (1 + v * 1.7)));
      if (pr) tw = MINW + (MAXW - MINW) * pr;
      w = l.w * 0.65 + tw * 0.35;
    }
    cur.push({ x: x, y: y, t: t, w: w });
    cx.fillStyle = ink;
    seg(cx, cur, cur.length - 1);
  }
  function begin(x, y, t) {
    if (done) return;
    ink = colorOf(cv);
    cur = [];
    strokes.push(cur);
    root.classList.add('is-ink');
    cl.disabled = ok.disabled = false;
    add(x, y, t, 0);
  }
  function end() {
    if (!cur) return;
    cur = null;
    cl.disabled = ok.disabled = !strokes.length;
  }
  function local(e) {
    var r = pd.getBoundingClientRect();
    return { x: (e.clientX - r.left) * W / (r.width || 1), y: (e.clientY - r.top) * H / (r.height || 1) };
  }
  cv.addEventListener('pointerdown', function (e) {
    if (e.button !== 0 || done) return;
    stopDemo();
    try { cv.setPointerCapture(e.pointerId); } catch (er) {  }
    var p = local(e);
    begin(p.x, p.y, e.timeStamp || performance.now());
  });
  cv.addEventListener('pointermove', function (e) {
    if (!cur) return;
    var list = e.getCoalescedEvents ? e.getCoalescedEvents() : [];
    if (!list.length) list = [e];
    list.forEach(function (ev) {
      var p = local(ev), pr = ev.pointerType === 'pen' && ev.pressure > 0 && ev.pressure < 1 ? ev.pressure : 0;
      add(p.x, p.y, ev.timeStamp || performance.now(), pr);
    });
  });
  cv.addEventListener('pointerup', end);
  cv.addEventListener('pointercancel', end);
  cv.addEventListener('transitionend', function (e) { if (e.propertyName === 'color') redraw(); });
  sc.addEventListener('transitionend', function (e) { if (e.propertyName === 'color' && done) stamp(); });
  function speck(g) {
    g.save();
    g.globalCompositeOperation = 'destination-out';
    for (var i = 0; i < 420; i++) {
      var x = Math.random() * W, y = Math.random() * H, r = Math.random() < 0.85 ? 0.35 + Math.random() * 0.5 : 0.9 + Math.random() * 0.9;
      g.globalAlpha = 0.5 + Math.random() * 0.5;
      g.beginPath(); g.arc(x, y, r, 0, Math.PI * 2); g.fill();
    }
    g.restore();
  }
  function rrect(g, x, y, w, h, r) {
    g.beginPath(); g.moveTo(x + r, y); g.arcTo(x + w, y, x + w, y + h, r); g.arcTo(x + w, y + h, x, y + h, r); g.arcTo(x, y + h, x, y, r); g.arcTo(x, y, x + w, y, r); g.closePath();
  }
  function bounds() {
    var a = { x0: 1e9, y0: 1e9, x1: -1e9, y1: -1e9 };
    strokes.forEach(function (s) { s.forEach(function (p) { a.x0 = Math.min(a.x0, p.x); a.y0 = Math.min(a.y0, p.y); a.x1 = Math.max(a.x1, p.x); a.y1 = Math.max(a.y1, p.y); }); });
    return a;
  }
  function stamp() {
    var col = colorOf(sc), b = bounds(), FW = 204, FH = 112;
    var k = Math.min((FW - 36) / Math.max(1, b.x1 - b.x0), (FH - 44) / Math.max(1, b.y1 - b.y0), 0.9);
    sx.clearRect(0, 0, W, H);
    sx.save();
    sx.translate(W / 2, H / 2);
    sx.rotate(-5 * Math.PI / 180);
    sx.fillStyle = col; sx.strokeStyle = col;
    sx.lineWidth = 2.6; rrect(sx, -FW / 2, -FH / 2, FW, FH, 14); sx.stroke();
    sx.lineWidth = 1; rrect(sx, -FW / 2 + 5, -FH / 2 + 5, FW - 10, FH - 10, 10); sx.stroke();
    sx.font = '800 10.5px -apple-system,BlinkMacSystemFont,"Apple SD Gothic Neo","Segoe UI",Roboto,sans-serif';
    sx.textAlign = 'center'; sx.textBaseline = 'middle';
    var tw = sx.measureText(T.seal).width + 14;
    sx.clearRect(-tw / 2, -FH / 2 - 4, tw, 14);
    sx.fillText(T.seal, 0, -FH / 2 + 2.5);
    var d = new Date(), ds = d.getFullYear() + '. ' + (d.getMonth() + 1) + '. ' + d.getDate();
    sx.font = '700 9.5px -apple-system,BlinkMacSystemFont,"Segoe UI",Roboto,sans-serif';
    var dw = sx.measureText(ds).width + 12;
    sx.clearRect(-dw / 2, FH / 2 - 9, dw, 14);
    sx.fillText(ds, 0, FH / 2 - 1);
    sx.translate(-(b.x0 + b.x1) / 2 * k, -(b.y0 + b.y1) / 2 * k + 3);
    sx.scale(k, k);
    strokes.forEach(function (s) { for (var i = 0; i < s.length; i++) seg(sx, s, i); });
    sx.restore();
    speck(sx);
  }
  ok.addEventListener('click', function () {
    if (done || !strokes.length) return;
    stopDemo();
    end();
    done = true;
    stamp();
    root.classList.add('is-done');
    if (!rm) { pd.classList.remove('is-thump'); void pd.offsetWidth; setTimeout(function () { if (root.isConnected) pd.classList.add('is-thump'); }, 300); }
    ok.disabled = true;
    okL.textContent = T.done;
    cl.textContent = T.again;
    live.textContent = T.said;
  });

    done = false;
    cur = null;

    pd.classList.remove('is-thump');

    cl.textContent = T.clear;
    cl.disabled = ok.disabled = true;
    live.textContent = '';

    cv.classList.add('is-fade');
    setTimeout(function () { strokes = []; cx.clearRect(0, 0, W, H); cv.classList.remove('is-fade'); }, rm ? 0 : 230);
  });

    { d: 'M98 64C99 63 100 62 101 61', ms: 90, gap: 160 },
    { d: 'M42 122C88 112 160 108 238 100', ms: 260, gap: 170 }
  ];
  function stopDemo() { if (demoT) { cancelAnimationFrame(demoT); demoT = 0; if (cur) end(); } }
  function plan(st) {
    gp.setAttribute('d', st.d);
    var L = gp.getTotalLength(), pts = [], n = Math.max(2, Math.ceil(L / 1.6)), prev = null, acc = 0;
    for (var i = 0; i <= n; i++) {
      var p = gp.getPointAtLength(L * i / n);
      pts.push({ x: (p.x - 140) * 1.12 + 150, y: (p.y - 90) * 1.08 + 84, c: 1 });
    }
    for (var j = 1; j < pts.length - 1; j++) {
      var a = Math.atan2(pts[j].y - pts[j - 1].y, pts[j].x - pts[j - 1].x), b = Math.atan2(pts[j + 1].y - pts[j].y, pts[j + 1].x - pts[j].x), t = Math.abs(b - a);
      if (t > Math.PI) t = 2 * Math.PI - t;
      pts[j].c = 1 + t * 14;
    }
    pts.forEach(function (p) { acc += p.c; p.at = acc; });
    pts.forEach(function (p) { p.at = p.at / acc * st.ms; });
    return pts;
  }
  root.__u468demo = function () {
    if (done || !root.isConnected) return;
    stopDemo();
    if (strokes.length) { strokes = []; cx.clearRect(0, 0, W, H); }
    var plans = SIGN.map(plan), si = 0, pi = 0, t0 = performance.now(), base = 0;
    function tick(now) {
      demoT = 0;
      if (!root.isConnected || done) return;
      var st = SIGN[si], pts = plans[si], el = now - t0 - base;
      if (pi === 0) { if (el < 0) { demoT = requestAnimationFrame(tick); return; } begin(pts[0].x, pts[0].y, now - el); pi = 1; }
      while (pi < pts.length && pts[pi].at <= el) { add(pts[pi].x, pts[pi].y, t0 + base + pts[pi].at, 0); pi++; }
      if (pi >= pts.length) {
        end();
        si++; pi = 0;
        if (si >= SIGN.length) return;
        base += st.ms + (SIGN[si].gap || 120);
      }
      demoT = requestAnimationFrame(tick);
    }
    if (rm) { plans.forEach(function (pts) { begin(pts[0].x, pts[0].y, 0); pts.forEach(function (p, i) { if (i) add(p.x, p.y, p.at, 0); }); end(); }); return; }
    demoT = requestAnimationFrame(tick);
  };
  ink = colorOf(cv);
})();
