Element: Date picker
ID: 431 · Category: Inputs & Search · Tags: date-picker, calendar, popover, booking, quick-picks, keyboard, disabled-dates
What it does: Press the booking date field and a calendar opens. Today is marked with a dot, past dates can’t be picked, and changing months slides the calendar sideways. Pick instantly with Today, Tomorrow or Next week, and once a date is set the times you can book that day appear.
How to use it: Press Tomorrow, or go to next month and pick a date. Inside the calendar, arrow keys, PageUp/PageDown and Enter work too
Footprint: fixed size, square

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var tr = q('.u431-tr'), trv = q('.u431-tt b'), pop = q('.u431-pop'), qp = q('.u431-qp'), hdT = q('.u431-hd b'), navs = root.querySelectorAll('.u431-nav button'), wk = q('.u431-wk'), grid = q('.u431-g'), ph = q('.u431-ph'), sl = q('.u431-sl'), note = q('.u431-note'), go = q('.u431-go'), live = q('.u431-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { ph: 'Select a date', book: 'Book', bookAt: 'Book for {t}', fmt: '{w}, {M} {d}', head: '{M} {y}', aria: '{W}, {M} {d}, {y}', today: ', today' };
  var MON = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  var WD = ['Su', 'Mo', 'Tu', 'We', 'Th', 'Fr', 'Sa'];
  var WDF = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
  var WDS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
  var SLOTS = ['09:30', '10:00', '11:00', '13:30', '14:30', '15:00', '16:30', '17:00', '18:00'];
  var now = new Date();
  var today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  var last = new Date(today.getFullYear(), today.getMonth() + 3, 0);
  var view = new Date(today.getFullYear(), today.getMonth(), 1);
  var sel = null, fd = new Date(today), slot = -1;
  wk.innerHTML = WD.map(function (w) { return '<span>' + w + '</span>'; }).join('');

  function key(d) { return d.getFullYear() * 10000 + d.getMonth() * 100 + d.getDate(); }
  function same(a, b) { return !!a && !!b && key(a) === key(b); }
  function addDays(d, n) { return new Date(d.getFullYear(), d.getMonth(), d.getDate() + n); }
  function ok(d) { return key(d) >= key(today) && key(d) <= key(last); }
  function fill(t, d) { return t.replace('{y}', d.getFullYear()).replace('{M}', MON[d.getMonth()]).replace('{d}', d.getDate()).replace('{w}', WDS[d.getDay()]).replace('{W}', WDF[d.getDay()]); }

  var shown = -1;
  function renderGrid(dir) {
    var y = view.getFullYear(), m = view.getMonth();
    if (shown !== y * 12 + m) {
      shown = y * 12 + m;
      var off = new Date(y, m, 1).getDay(), n = new Date(y, m + 1, 0).getDate();
      var html = '';
      for (var r = 0; r < 6; r++) {
        html += '<div class="u431-row" role="row">';
        for (var c = 0; c < 7; c++) {
          var day = r * 7 + c - off + 1;
          if (day < 1 || day > n) { html += '<span role="gridcell"></span>'; continue; }
          var d = new Date(y, m, day), isT = same(d, today);
          html += '<span role="gridcell"><button type="button" class="u431-d" data-day="' + day + '"' + (isT ? ' aria-current="date"' : '') +
            ' aria-label="' + fill(T.aria, d) + (isT ? T.today : '') + '"' + (ok(d) ? '' : ' disabled') + '>' + day + '</button></span>';
        }
        html += '</div>';
      }
      grid.innerHTML = html;
      hdT.textContent = fill(T.head, view);
      navs[0].disabled = y * 12 + m <= today.getFullYear() * 12 + today.getMonth();
      navs[1].disabled = y * 12 + m >= last.getFullYear() * 12 + last.getMonth();
      grid.classList.remove('is-l', 'is-r');
      if (dir && !rm) { void grid.offsetWidth; grid.classList.add(dir > 0 ? 'is-l' : 'is-r'); }
    }
    Array.prototype.forEach.call(grid.querySelectorAll('.u431-d'), function (b) {
      var d = new Date(y, m, +b.dataset.day), isS = same(d, sel);
      b.setAttribute('aria-pressed', isS ? 'true' : 'false');
      b.parentNode.setAttribute('aria-selected', isS ? 'true' : 'false');
      b.tabIndex = same(d, fd) ? 0 : -1;
    });
    Array.prototype.forEach.call(qp.children, function (b) { b.setAttribute('aria-pressed', same(sel, addDays(today, +b.dataset.add)) ? 'true' : 'false'); });
  }
  function shiftMonth(n) {
    var nv = new Date(view.getFullYear(), view.getMonth() + n, 1);
    if (key(nv) < key(new Date(today.getFullYear(), today.getMonth(), 1)) || key(nv) > key(last)) return false;
    view = nv;
    renderGrid(n);
    return true;
  }
  function renderSlots() {
    ph.hidden = !!sel;
    sl.hidden = note.hidden = !sel;
    if (!sel) { go.disabled = true; go.textContent = T.book; return; }
    var seed = sel.getDate() * 7 + sel.getMonth() * 3;
    sl.innerHTML = SLOTS.map(function (t, i) {
      var full = (seed * 13 + i * 29 + i * i * 7) % 9 < 3;
      return '<button type="button" class="u431-slot" role="radio" aria-checked="' + (i === slot) + '"' + (full ? ' disabled' : '') + '>' + t + '</button>';
    }).join('');
    go.disabled = slot < 0;
    go.textContent = slot < 0 ? T.book : T.bookAt.replace('{t}', fill(T.fmt, sel) + ' ' + SLOTS[slot]);
  }
  function syncTrigger() {
    tr.classList.toggle('is-empty', !sel);
    trv.textContent = sel ? fill(T.fmt, sel) : T.ph;
  }
  function focusDay() {
    var b = grid.querySelector('[data-day="' + fd.getDate() + '"]');
    if (b) b.focus({ preventScroll: true });
  }
  function setOpen(v, focus) {
    root.classList.toggle('is-open', v);
    tr.setAttribute('aria-expanded', v ? 'true' : 'false');
    if (v) {
      fd = sel ? new Date(sel) : new Date(today);
      view = new Date(fd.getFullYear(), fd.getMonth(), 1);
      renderGrid(0);
      if (focus) focusDay();
    } else if (focus) tr.focus({ preventScroll: true });
  }
  var closeT = 0;
  function pick(d, focus) {
    if (!ok(d)) return;
    sel = new Date(d); fd = new Date(d); slot = -1;
    if (sel.getMonth() !== view.getMonth() || sel.getFullYear() !== view.getFullYear()) { view = new Date(sel.getFullYear(), sel.getMonth(), 1); }
    renderGrid(0);
    syncTrigger();
    renderSlots();
    live.textContent = fill(T.aria, sel);
    clearTimeout(closeT);
    closeT = setTimeout(function () { if (root.isConnected) setOpen(false, focus); }, rm ? 0 : 220);
  }

  tr.addEventListener('click', function () { setOpen(!root.classList.contains('is-open'), true); });
  tr.addEventListener('keydown', function (e) { if (e.key === 'ArrowDown') { e.preventDefault(); setOpen(true, true); } });
  navs[0].addEventListener('click', function () { shiftMonth(-1); });
  navs[1].addEventListener('click', function () { shiftMonth(1); });
  qp.addEventListener('click', function (e) { var b = e.target.closest('button'); if (b) pick(addDays(today, +b.dataset.add), e.detail === 0); });
  grid.addEventListener('click', function (e) {
    var b = e.target.closest('.u431-d');
    if (b && !b.disabled) pick(new Date(view.getFullYear(), view.getMonth(), +b.dataset.day), e.detail === 0);
  });
  grid.addEventListener('keydown', function (e) {
    var b = e.target.closest('.u431-d');
    if (!b) return;
    var cur = new Date(view.getFullYear(), view.getMonth(), +b.dataset.day), nd = null;
    if (e.key === 'ArrowRight') nd = addDays(cur, 1);
    else if (e.key === 'ArrowLeft') nd = addDays(cur, -1);
    else if (e.key === 'ArrowDown') nd = addDays(cur, 7);
    else if (e.key === 'ArrowUp') nd = addDays(cur, -7);
    else if (e.key === 'Home') nd = addDays(cur, -cur.getDay());
    else if (e.key === 'End') nd = addDays(cur, 6 - cur.getDay());
    else if (e.key === 'PageDown' || e.key === 'PageUp') {
      var dm = e.key === 'PageDown' ? 1 : -1;
      nd = new Date(cur.getFullYear(), cur.getMonth() + dm, Math.min(cur.getDate(), new Date(cur.getFullYear(), cur.getMonth() + dm + 1, 0).getDate()));
    } else return;
    e.preventDefault();
    e.stopPropagation();
    if (key(nd) < key(today)) nd = new Date(today);
    if (key(nd) > key(last)) nd = new Date(last);
    var dir = (nd.getFullYear() * 12 + nd.getMonth()) - (view.getFullYear() * 12 + view.getMonth());
    fd = nd;
    if (dir) view = new Date(nd.getFullYear(), nd.getMonth(), 1);
    renderGrid(dir);
    focusDay();
  });
  sl.addEventListener('click', function (e) {
    var b = e.target.closest('.u431-slot');
    if (!b || b.disabled) return;
    slot = Array.prototype.indexOf.call(sl.children, b);
    renderSlots();
  });
  root.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && root.classList.contains('is-open')) { e.preventDefault(); e.stopPropagation(); setOpen(false, true); }
  });
  root.addEventListener('pointerdown', function (e) {
    if (root.classList.contains('is-open') && !e.target.closest('.u431-pop, .u431-tr')) setOpen(false);
  });
  root.addEventListener('focusout', function (e) {
    if (root.classList.contains('is-open') && e.relatedTarget && !pop.contains(e.relatedTarget) && e.relatedTarget !== tr) setOpen(false);
  });
  syncTrigger();
  renderGrid(0);
  renderSlots();
})();
</script>
