<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var cd = root.querySelector('.u425-cd');
  cd.addEventListener('pointermove', function (e) {
    var r = cd.getBoundingClientRect();
    cd.style.setProperty('--mx', ((e.clientX - r.left) / r.width * 100).toFixed(1) + '%');
    cd.style.setProperty('--my', ((e.clientY - r.top) / r.height * 100).toFixed(1) + '%');
  });
})();
</script>
