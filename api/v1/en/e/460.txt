<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var fab = q('.u460-fab'), group = q('.u460-ic'), filt = q('.u460-f'), goo = q('.u460-goo'), fabBlob = q('.u460-bb.is-fab');
  var blobs = [].slice.call(root.querySelectorAll('.u460-bb:not(.is-fab)')), acts = [].slice.call(root.querySelectorAll('.u460-a'));
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var uid = 'u460' + Math.random().toString(36).slice(2, 7);
  filt.id = uid + 'g';
  var gooT = 0;
  function gooOn(v) {
    clearTimeout(gooT);
    if (v) goo.style.filter = 'url(#' + uid + 'g)';
    else gooT = setTimeout(function () { if (root.isConnected && !open) goo.style.filter = ''; }, rm ? 0 : 900);
  }
  group.id = uid + 'm';
  fab.setAttribute('aria-controls', group.id);
  var R = 94, open = false;
  [-157.5, -112.5, -67.5, -22.5].forEach(function (deg, i) {
    var a = deg * Math.PI / 180, x = (Math.cos(a) * R).toFixed(1) + 'px', y = (Math.sin(a) * R).toFixed(1) + 'px';
    acts[i].dataset.x = x; acts[i].dataset.y = y;
  });
  function squish(k) {
    if (rm || !fabBlob.animate) return;
    fabBlob.animate([{ transform: 'scale(1)' }, { transform: 'scale(' + k + ')', offset: .3 }, { transform: 'scale(1.06)', offset: .65 }, { transform: 'scale(1)' }], { duration: 560, easing: 'ease-out' });
  }
  function setOpen(v, focusFab) {
    if (open === v) return;
    open = v;
    gooOn(v);
    root.classList.toggle('is-open', v);
    fab.setAttribute('aria-expanded', v ? 'true' : 'false');
    [blobs, acts].forEach(function (list) {
      list.forEach(function (el, i) {
        el.style.setProperty('--u460-tx', v ? acts[i].dataset.x : '0px');
        el.style.setProperty('--u460-ty', v ? acts[i].dataset.y : '0px');
      });
    });
    acts.forEach(function (b) { b.tabIndex = v ? 0 : -1; });
    blobs.forEach(function (b) { b.classList.remove('is-hov'); });
    squish(v ? 0.86 : 0.9);
    if (!v && focusFab) fab.focus({ preventScroll: true });
  }
  fab.addEventListener('click', function () { setOpen(!open); });
  acts.forEach(function (b, i) {
    var on = function () { if (open) blobs[i].classList.add('is-hov'); }, off = function () { blobs[i].classList.remove('is-hov'); };
    b.addEventListener('pointerenter', on);
    b.addEventListener('pointerleave', off);
    b.addEventListener('focus', on);
    b.addEventListener('blur', off);
    b.addEventListener('click', function () {
      if (!open) return;
      var had = document.activeElement === b;
      if (!rm && blobs[i].animate) blobs[i].animate([{ transform: 'translate(' + b.dataset.x + ',' + b.dataset.y + ') scale(1.1)' }, { transform: 'translate(' + b.dataset.x + ',' + b.dataset.y + ') scale(1.28)' }, { transform: 'translate(' + b.dataset.x + ',' + b.dataset.y + ') scale(1)' }], { duration: 260, easing: 'ease-out' });
      setTimeout(function () { if (root.isConnected) setOpen(false, had); }, rm ? 0 : 200);
    });
  });
  root.addEventListener('keydown', function (e) {
    if (!open) return;
    if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); setOpen(false, true); return; }
    var i = acts.indexOf(document.activeElement);
    if (/^Arrow(Left|Up|Right|Down)$/.test(e.key)) {
      e.preventDefault();
      var d = e.key === 'ArrowRight' || e.key === 'ArrowDown' ? 1 : -1;
      acts[i < 0 ? (d > 0 ? 0 : 3) : (i + d + 4) % 4].focus({ preventScroll: true });
    }
  });
  root.addEventListener('focusout', function (e) { if (open && e.relatedTarget && !root.contains(e.relatedTarget)) setOpen(false); });
  root.addEventListener('pointerdown', function (e) { if (open && !e.target.closest('button')) setOpen(false); });
})();
</script>
