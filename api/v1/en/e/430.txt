<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var tr = q('.u430-tr'), who = q('.u430-who'), input = q('.u430-q'), lb = q('.u430-lb'), live = q('.u430-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { none: 'No one matches', picked: 'Assignee: {n}' };
  var PEOPLE = [['Seoyeon Han', 'Product designer', '#7C6CF2'], ['Jihoon Park', 'Frontend engineer', '#2F8FE0'], ['Minjae Kim', 'Backend engineer', '#E08A2F'], ['Sumin Lee', 'Growth marketer', '#16A37B']];
  var uid = 'u430' + Math.random().toString(36).slice(2, 7);
  lb.id = uid + 'lb'; input.setAttribute('aria-controls', lb.id);
  var sel = 0, shown = [], act = -1;
  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function av(p) { return '<span class="u430-av" style="--c:' + p[2] + '" aria-hidden="true">' + esc(p[0].charAt(0)) + '</span>'; }
  function trigger(anim) {
    var p = PEOPLE[sel];
    who.innerHTML = av(p) + '<span class="u430-tx"><b>' + esc(p[0]) + '</b><small>' + esc(p[1]) + '</small></span>';
    tr.setAttribute('aria-label', T.picked.replace('{n}', p[0]));
    if (anim) { tr.classList.remove('is-new'); void tr.offsetWidth; tr.classList.add('is-new'); }
  }
  function render(stagger) {
    var t = input.value.trim().toLowerCase();
    shown = [];
    lb.innerHTML = PEOPLE.map(function (p, k) {
      var i = p[0].toLowerCase().indexOf(t), j = p[1].toLowerCase().indexOf(t);
      if (t && i < 0 && j < 0) return '';
      var n = shown.push(k) - 1;
      var name = i >= 0 && t ? esc(p[0].slice(0, i)) + '<mark>' + esc(p[0].slice(i, i + t.length)) + '</mark>' + esc(p[0].slice(i + t.length)) : esc(p[0]);
      return '<div class="u430-o" role="option" id="' + uid + k + '" data-k="' + k + '" aria-selected="' + (k === sel) + '" style="--i:' + n + (stagger ? '' : ';animation:none') + '">' + av(p) + '<b>' + name + '</b><svg class="u430-ck" viewBox="0 0 24 24" aria-hidden="true"><path d="m5 12.5 4.5 4.5L19 7.5"/></svg></div>';
    }).join('') || '<div class="u430-no">' + T.none + '</div>';
    act = t ? (shown.length ? 0 : -1) : shown.indexOf(sel);
    paint();
  }
  function paint() {
    Array.prototype.forEach.call(lb.querySelectorAll('.u430-o'), function (o, i) { o.classList.toggle('is-act', i === act); });
    var o = lb.querySelectorAll('.u430-o')[act];
    if (o) input.setAttribute('aria-activedescendant', o.id); else input.removeAttribute('aria-activedescendant');
  }
  function setOpen(v, focus) {
    root.classList.toggle('is-open', v);
    tr.setAttribute('aria-expanded', v ? 'true' : 'false');
    input.setAttribute('aria-expanded', v ? 'true' : 'false');
    if (v) { input.value = ''; render(true); if (focus) input.focus({ preventScroll: true }); }
    else if (focus) tr.focus({ preventScroll: true });
  }
  var closeT = 0;
  function choose(k, focus) {
    sel = k;
    Array.prototype.forEach.call(lb.querySelectorAll('.u430-o'), function (o) { o.setAttribute('aria-selected', +o.dataset.k === k ? 'true' : 'false'); });
    live.textContent = T.picked.replace('{n}', PEOPLE[k][0]);
    clearTimeout(closeT);
    closeT = setTimeout(function () { if (!root.isConnected) return; setOpen(false, focus); trigger(true); }, rm ? 0 : 380);
  }
  tr.addEventListener('click', function () { setOpen(!root.classList.contains('is-open'), true); });
  tr.addEventListener('keydown', function (e) { if (e.key === 'ArrowDown') { e.preventDefault(); setOpen(true, true); } });
  input.addEventListener('input', function () { render(false); });
  input.addEventListener('keydown', function (e) {
    var n = lb.querySelectorAll('.u430-o').length;
    if (e.key === 'ArrowDown' || e.key === 'ArrowUp') { e.preventDefault(); if (n) { act = (act + (e.key === 'ArrowDown' ? 1 : -1) + n) % n; paint(); } }
    else if (e.key === 'Enter') { e.preventDefault(); var o = lb.querySelectorAll('.u430-o')[act]; if (o) choose(+o.dataset.k, true); }
    else if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); setOpen(false, true); }
  });
  q('.u430-pop').addEventListener('mousedown', function (e) { if (e.target !== input) e.preventDefault(); });
  lb.addEventListener('pointermove', function (e) { var o = e.target.closest('.u430-o'); if (!o) return; var i = Array.prototype.indexOf.call(lb.querySelectorAll('.u430-o'), o); if (i !== act) { act = i; paint(); } });
  lb.addEventListener('click', function (e) { var o = e.target.closest('.u430-o'); if (o) choose(+o.dataset.k, false); });
  root.addEventListener('pointerdown', function (e) { if (root.classList.contains('is-open') && !e.target.closest('.u430-w')) setOpen(false); });
  root.addEventListener('focusout', function (e) { if (root.classList.contains('is-open') && e.relatedTarget && !root.contains(e.relatedTarget)) setOpen(false); });
  trigger(false);
  render(false);
  setTimeout(function () { if (root.isConnected && !root.classList.contains('is-open')) setOpen(true, false); }, rm ? 0 : 250);
})();
</script>
