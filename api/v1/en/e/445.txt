<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var fd = q('.u445-fd'), input = q('.u445-in'), ap = q('.u445-ap'), msg = q('.u445-msg'), nw = q('.u445-new'), live = q('.u445-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { apply: 'Apply', remove: 'Remove', bad: 'That code doesn’t exist', ok: 'Coupon applied', off: 'Coupon removed' };
  var PRICE = 48200, OFF = 4820, on = false, busy = false, raf = 0;
  function won(n) { return '₩' + Math.round(n).toLocaleString('ko-KR'); }
  function roll() {
    cancelAnimationFrame(raf);
    var t0 = performance.now() + 150, dur = rm ? 0 : 700;
    (function step(now) {
      if (!root.isConnected) return;
      var k = dur ? Math.max(0, Math.min(1, (now - t0) / dur)) : 1, e = 1 - Math.pow(1 - k, 3);
      nw.textContent = won(PRICE - OFF * e);
      if (k < 1) raf = requestAnimationFrame(step);
    })(performance.now());
  }
  function setOn(v) {
    on = v;
    root.classList.toggle('is-on', v);
    nw.setAttribute('aria-hidden', v ? 'false' : 'true');
    ap.textContent = v ? T.remove : T.apply;
    if (v) { roll(); root.classList.remove('is-thud'); void root.offsetWidth; root.classList.add('is-thud'); }
    live.textContent = v ? T.ok : T.off;
  }
  function apply() {
    if (busy) return;
    if (on) { setOn(false); input.value = ''; input.focus({ preventScroll: true }); return; }
    var code = input.value.trim().toUpperCase();
    if (!code) { input.focus({ preventScroll: true }); return; }
    busy = true;
    ap.innerHTML = '<i aria-hidden="true"></i>';
    setTimeout(function () {
      if (!root.isConnected) return;
      busy = false;
      if (code === 'WELCOME10') { msg.classList.remove('is-on'); fd.classList.remove('is-err'); setOn(true); input.classList.remove('is-act'); return; }
      ap.textContent = T.apply;
      msg.textContent = T.bad; msg.classList.add('is-on');
      fd.classList.add('is-err');
      fd.classList.remove('is-shake'); void fd.offsetWidth; fd.classList.add('is-shake');
    }, rm ? 150 : 600);
  }
  input.addEventListener('input', function () {
    var up = input.value.toUpperCase().replace(/\s/g, '');
    if (up !== input.value) input.value = up;
    input.classList.add('is-act');
    if (fd.classList.contains('is-err')) { fd.classList.remove('is-err'); msg.classList.remove('is-on'); }
  });
  input.addEventListener('blur', function () { input.classList.remove('is-act'); });
  input.addEventListener('keydown', function (e) { if (e.key === 'Enter') { e.preventDefault(); apply(); } });
  ap.addEventListener('click', apply);
  ap.textContent = T.apply;
  nw.textContent = won(PRICE - OFF);
})();
</script>
