<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = {
    sum: function (a, b, t) { return '<b>' + a + '–' + b + '</b> of ' + t; },
    prev: 'Previous page',
    next: 'Next page',
    page: function (n) { return 'Page ' + n; },
    jumpF: 'Jump forward 5 pages',
    jumpB: 'Jump back 5 pages',
    now: function (n, t) { return 'Page ' + n + ' of ' + t; }
  };
  var NAMES = ['S. Kim', 'J. Lee', 'J. Park', 'M. Choi', 'H. Jung', 'J. Han', 'Y. Yoon', 'D. Kang', 'S. Oh', 'J. Seo'];
  var ST = [['Paid', 'pay'], ['Shipping', 'ship'], ['Delivered', 'done'], ['Delivered', 'done'], ['Canceled', 'off']];
  var TOTAL = 1284, size = 20, page = 1, swapT = 0;
  var body = root.querySelector('.u449-body'), sum = root.querySelector('.u449-sum'), nav = root.querySelector('.u449-nav'), sel = root.querySelector('select'), live = root.querySelector('.u449-live');
  var fmt = function (n) { return n.toLocaleString('en-US'); };
  function pages() { return Math.ceil(TOTAL / size); }
  function slots(cur, last) {
    if (last <= 7) return Array.from({ length: last }, function (_, i) { return i + 1; });
    if (cur <= 4) return [1, 2, 3, 4, 5, 'R', last];
    if (cur >= last - 3) return [1, 'L', last - 4, last - 3, last - 2, last - 1, last];
    return [1, 'L', cur - 1, cur, cur + 1, 'R', last];
  }
  function rows() {
    var a = (page - 1) * size, b = Math.min(TOTAL, a + size), h = '';
    for (var i = a; i < b; i++) {
      var hsh = Math.imul(i + 17, 2654435761) >>> 0, s = ST[(hsh >>> 9) % ST.length], amt = ((hsh >>> 3) % 2400 + 90) * 100;
      h += '<div class="u449-row" role="row"><span class="no" role="cell">#' + (20261284 - i) + '</span><span class="nm" role="cell">' + NAMES[(hsh >>> 5) % NAMES.length] + '</span><span class="am" role="cell">₩' + fmt(amt) + '</span><span role="cell"><span class="u449-chip ' + s[1] + '">' + s[0] + '</span></span></div>';
    }
    body.innerHTML = h; body.scrollTop = 0;
    sum.innerHTML = T.sum(fmt(a + 1), fmt(b), fmt(TOTAL));
  }
  function btn(cls, label, html, go, dis) {
    var b = document.createElement('button');
    b.type = 'button'; b.className = cls; b.innerHTML = html;
    if (label) b.setAttribute('aria-label', label);
    b.disabled = !!dis;
    b.addEventListener('click', function () { to(go, true); });
    nav.appendChild(b); return b;
  }
  function draw() {
    var last = pages(), had = nav.contains(document.activeElement);
    nav.innerHTML = '';
    btn('u449-prev', T.prev, '<svg viewBox="0 0 14 14" aria-hidden="true"><path d="M8.5 3 4.5 7l4 4"/></svg>', page - 1, page === 1);
    slots(page, last).forEach(function (s) {
      if (s === 'L' || s === 'R') { var j = btn('u449-el', s === 'L' ? T.jumpB : T.jumpF, '…', s === 'L' ? Math.max(1, page - 5) : Math.min(last, page + 5)); j.title = j.getAttribute('aria-label'); return; }
      var b = btn('u449-n', T.page(s), String(s), s);
      if (s === page) b.setAttribute('aria-current', 'page');
    });
    btn('u449-next', T.next, '<svg viewBox="0 0 14 14" aria-hidden="true"><path d="M5.5 3l4 4-4 4"/></svg>', page + 1, page === last);
    if (had) { var c = nav.querySelector('[aria-current]'); if (c) c.focus({ preventScroll: true }); }
  }
  function to(n, announce) {
    var last = pages();
    n = Math.max(1, Math.min(last, n));
    if (n === page && body.childElementCount) { draw(); return; }
    page = n;
    clearTimeout(swapT);
    body.classList.add('is-swap');
    swapT = setTimeout(function () { if (!root.isConnected) return; rows(); body.classList.remove('is-swap'); }, rm ? 0 : 140);
    draw();
    if (announce) live.textContent = T.now(page, last);
  }
  nav.addEventListener('keydown', function (e) {
    var d = { ArrowLeft: page - 1, ArrowRight: page + 1, Home: 1, End: pages() }[e.key];
    if (d == null) return;
    e.preventDefault(); to(d, true);
  });
  sel.addEventListener('change', function () {
    var first = (page - 1) * size;
    size = +sel.value;
    page = 0; to(Math.floor(first / size) + 1, true);
  });
  rows(); draw();
})();
</script>
