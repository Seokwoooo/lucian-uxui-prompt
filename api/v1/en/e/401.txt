<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { busy: 'Connecting to Google', ok: 'Signed in with Google' };
  var b = root.querySelector('.u401-b'), lb = root.querySelector('.u401-lb'), live = root.querySelector('.u401-live');
  var sp = lb.querySelectorAll('span'), timer = 0;
  function fit(k) {
    var w = sp[k].offsetWidth;
    if (!w) { lb.style.width = ''; return; }
    lb.style.width = w + 'px';
  }
  function later(fn, ms) { timer = setTimeout(function () { if (root.isConnected) fn(); }, ms); }
  b.addEventListener('click', function () {
    if (b.classList.contains('is-busy') || b.classList.contains('is-ok')) return;
    fit(0); void lb.offsetWidth;
    b.classList.add('is-busy'); b.setAttribute('aria-busy', 'true'); fit(1);
    live.textContent = T.busy;
    later(function () {
      b.classList.remove('is-busy'); b.removeAttribute('aria-busy');
      b.classList.add('is-ok'); fit(2);
      live.textContent = T.ok;
      later(function () {
        b.classList.remove('is-ok'); fit(0);
        later(function () { if (!b.classList.contains('is-busy')) lb.style.width = ''; }, rm ? 0 : 600);
      }, rm ? 900 : 1300);
    }, rm ? 500 : 1400);
  });
})();
</script>
