<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var ms = Array.prototype.slice.call(root.querySelectorAll('.u414-m'));
  var box = root.querySelector('.u414-msgs'), x = root.querySelector('.u414-x'), sp = root.querySelector('.u414-sp'), bd = root.querySelector('.u414-bd');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var cur = 0, t = 0, hover = false, min = false;
  function show(i) {
    var prev = cur;
    ms.forEach(function (m, k) {
      m.classList.toggle('is-out', k === prev && k !== i);
      m.classList.toggle('is-on', k === i);
      m.tabIndex = k === i ? 0 : -1;
      m.setAttribute('aria-hidden', String(k !== i));
    });
    cur = i;
    setTimeout(function () { if (ms[prev] && prev !== cur) ms[prev].classList.remove('is-out'); }, 650);
  }
  function tick() {
    clearTimeout(t);
    if (rm) return;
    t = setTimeout(function () {
      if (!root.isConnected) return;
      if (!hover && !min && !root.contains(document.activeElement)) show((cur + 1) % ms.length);
      tick();
    }, 3000);
  }
  box.addEventListener('pointerenter', function () { hover = true; });
  box.addEventListener('pointerleave', function () { hover = false; });
  function fold(v) {
    min = v;
    root.classList.toggle('is-min', v);
    bd.inert = v;
    sp.tabIndex = v ? 0 : -1;
    sp.setAttribute('aria-label', v ? 'Show announcement' : 'Collapse announcement');
    sp.setAttribute('aria-expanded', String(!v));
    (v ? sp : x).focus({ preventScroll: true });
  }
  x.addEventListener('click', function () { fold(true); });
  sp.addEventListener('click', function () { if (min) fold(false); });
  ms.forEach(function (m) { m.addEventListener('click', function (e) { e.preventDefault(); }); });
  ms.forEach(function (m, k) { m.tabIndex = k === 0 ? 0 : -1; m.setAttribute('aria-hidden', String(k !== 0)); });
  sp.removeAttribute('aria-expanded');
  tick();
})();
</script>
