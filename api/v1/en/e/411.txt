<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { view: function (n) { return 'View ' + n; }, done: 'Terms accepted' };
  var all = root.querySelector('.u411-all input'), go = root.querySelector('.u411-go'), live = root.querySelector('.u411-live');
  var labs = [].slice.call(root.querySelectorAll('.u411-row .u411-r'));
  var items = labs.map(function (l) { return l.querySelector('input'); });
  var req = labs.map(function (l) { return l.hasAttribute('data-req'); });
  var gen = 0;
  root.querySelectorAll('.u411-v').forEach(function (v, k) { v.setAttribute('aria-label', T.view(labs[k].querySelector('.u411-nm').textContent)); });
  function sync() {
    var n = items.filter(function (i) { return i.checked; }).length;
    all.checked = n === items.length; all.indeterminate = n > 0 && n < items.length;
    var ok = items.every(function (i, k) { return !req[k] || i.checked; });
    go.classList.toggle('is-ready', ok); go.setAttribute('aria-disabled', String(!ok));
    if (!ok) go.classList.remove('is-ok');
  }
  all.addEventListener('change', function () {
    var want = all.checked, g = ++gen;
    all.indeterminate = false;
    var order = want ? [0, 1, 2] : [2, 1, 0];
    order.forEach(function (k, j) {
      setTimeout(function () {
        if (!root.isConnected || g !== gen) return;
        items[k].checked = want; sync();
        if (j < order.length - 1) { all.checked = want; all.indeterminate = false; }
      }, rm ? 0 : j * 110);
    });
  });
  items.forEach(function (i) { i.addEventListener('change', function () { gen += 1; sync(); }); });
  go.addEventListener('click', function () {
    if (!go.classList.contains('is-ready') || go.classList.contains('is-ok')) return;
    go.classList.add('is-ok'); live.textContent = T.done;
  });
  sync();
})();
</script>
