<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = {
    left: function (n) { return n + ' required ' + (n === 1 ? 'item' : 'items') + ' left'; },
    ready: 'All required items agreed'
  };
  var DOCS = [
    ['Terms of Service (required)', '<b>Article 1 (Purpose)</b><p>These terms set out the conditions and procedures for using the services Lucian (the “Company”) provides, and the rights and duties of the Company and its members.</p><b>Article 2 (Effect)</b><p>The terms are posted in the service. If they change, we’ll let you know 7 days before they take effect.</p><b>Article 3 (Membership)</b><p>Membership begins once you agree to these terms and the Company accepts your application.</p>'],
    ['Privacy notice (required)', '<dl><dt>What we collect</dt><dd>Name, email, phone number</dd><dt>Why</dt><dd>Identifying members, running the service, support</dd><dt>How long</dt><dd>Until 30 days after you leave</dd></dl><p>You may decline, but you won’t be able to sign up.</p>'],
    ['Marketing messages (optional)', '<p>We’ll send news about features and offers by email and SMS.</p><p>You can use the service without agreeing, and turn this off any time in settings.</p>']
  ];
  var main = root.querySelector('.u411-main'), all = root.querySelector('.u411-all input');
  var req = [].slice.call(root.querySelectorAll('input[data-req]')), mk = root.querySelector('.u411-mk'), ch = [].slice.call(root.querySelectorAll('input[data-ch]'));
  var leaves = req.concat(ch), next = root.querySelector('.u411-next'), live = root.querySelector('.u411-live');
  var sh = root.querySelector('.u411-sh'), shTitle = sh.querySelector('.u411-shd b'), doc = sh.querySelector('.u411-doc'), ok = sh.querySelector('.u411-ok');
  var uid = 'u411' + Math.random().toString(36).slice(2, 8), cur = -1, opener = null, lastLeft = -1, gen = 0;
  shTitle.id = uid + 't'; sh.setAttribute('aria-labelledby', shTitle.id);
  function later(fn, ms) { var g = gen; setTimeout(function () { if (root.isConnected && g === gen) fn(); }, ms); }
  function sync() {
    var nc = ch.filter(function (c) { return c.checked; }).length;
    mk.checked = nc === ch.length; mk.indeterminate = nc > 0 && nc < ch.length;
    var n = leaves.filter(function (c) { return c.checked; }).length;
    all.checked = n === leaves.length; all.indeterminate = n > 0 && n < leaves.length;
    var left = req.filter(function (c) { return !c.checked; }).length;
    next.disabled = left > 0;
    if (left !== lastLeft && lastLeft !== -1) live.textContent = left ? T.left(left) : T.ready;
    lastLeft = left;
  }
  all.addEventListener('change', function () { leaves.forEach(function (c) { c.checked = all.checked; }); sync(); });
  mk.addEventListener('change', function () { ch.forEach(function (c) { c.checked = mk.checked; }); sync(); });
  leaves.forEach(function (c) { c.addEventListener('change', sync); });
  function openDoc(k, from) {
    cur = k; opener = from;
    shTitle.textContent = DOCS[k][0]; doc.innerHTML = DOCS[k][1]; doc.scrollTop = 0;
    root.classList.add('is-sheet'); main.inert = true; sh.inert = false;
    later(function () { ok.focus({ preventScroll: true }); }, rm ? 0 : 300);
  }
  function closeDoc(agree) {
    if (agree) {
      if (cur === 2) { ch.forEach(function (c) { c.checked = true; }); } else { req[cur + 1].checked = true; }
      sync();
    }
    root.classList.remove('is-sheet'); main.inert = false; sh.inert = true;
    if (opener) opener.focus({ preventScroll: true });
  }
  [].slice.call(root.querySelectorAll('.u411-view')).forEach(function (b) { b.addEventListener('click', function () { openDoc(+b.getAttribute('data-d'), b); }); });
  ok.addEventListener('click', function () { closeDoc(true); });
  root.querySelector('.u411-x').addEventListener('click', function () { closeDoc(false); });
  root.querySelector('.u411-scrim').addEventListener('click', function () { closeDoc(false); });
  sh.addEventListener('keydown', function (e) {
    if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); closeDoc(false); return; }
    if (e.key !== 'Tab') return;
    var f = [root.querySelector('.u411-x'), doc, ok];
    var i = f.indexOf(document.activeElement);
    if (e.shiftKey && i === 0) { e.preventDefault(); ok.focus(); }
    else if (!e.shiftKey && i === f.length - 1) { e.preventDefault(); f[0].focus(); }
  });
  next.addEventListener('click', function () {
    if (next.disabled || next.classList.contains('is-busy') || next.classList.contains('is-ok')) return;
    next.classList.add('is-busy'); next.setAttribute('aria-busy', 'true');
    later(function () {
      next.classList.remove('is-busy'); next.removeAttribute('aria-busy'); next.classList.add('is-ok');
      later(function () { next.classList.remove('is-ok'); }, rm ? 900 : 1500);
    }, rm ? 300 : 800);
  });
  sh.inert = true;
  sync();
})();
</script>
