<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var fd = q('.u445-fd'), input = q('.u445-in'), ap = q('.u445-ap'), chipB = q('.u445-chip b'), chipS = q('.u445-chip small'), chipE = q('.u445-chip em'), x = q('.u445-x'), msg = q('.u445-msg'), msgT = q('.u445-msg span'), cnt = q('.u445-sh small'), ts = q('.u445-ts'), dl = q('.u445-dl'), dv = q('.u445-dv'), tot = q('.u445-r.is-tot b'), go = q('.u445-go'), live = q('.u445-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { hint: '코드는 영문과 숫자예요. 예: WELCOME10', bad: '없는 쿠폰 코드예요. 철자를 확인해 주세요.', exp: '8월 31일에 끝난 쿠폰이에요.', min: '{min} 이상 주문할 때 쓸 수 있어요. {gap} 더 담아 주세요.', ok: '쿠폰을 적용했어요.', off: '쿠폰을 뺐어요.', dc: '쿠폰 할인', pay: '{p} 결제하기', count: '{n}장', gap: '{gap} 더 담으면 쓸 수 있어요', checking: '확인하는 중' };
  var ITEMS = 48200, SHIP = 3000;
  // 쿠폰: 코드, 표 앞 큰 글자, 이름, 조건, 최소 주문 금액, 할인 계산
  var COUPONS = [
    { code: 'WELCOME10', big: '10%', name: '신규 가입 10% 할인', cond: '최대 5,000원 · 12월 31일까지', min: 0, calc: function () { return Math.min(5000, Math.floor(ITEMS * 0.1 / 10) * 10); } },
    { code: 'FREESHIP', big: '무료배송', sm: true, name: '무료배송 쿠폰', cond: '3만 원 이상 · 11월 30일까지', min: 30000, calc: function () { return SHIP; } },
    { code: 'VIP10000', big: '₩10,000', sm: true, name: '1만 원 할인', cond: '10만 원 이상 주문 시', min: 100000, calc: function () { return 10000; } }
  ];
  var EXPIRED = ['SUMMER2026', 'CHUSEOK15'];
  var applied = null, shown = ITEMS + SHIP, animT = 0, busy = false;
  var CHECK = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="m5.5 12.5 4 4 9-9.5"/></svg>';
  function won(n) { return '₩' + Math.round(n).toLocaleString('ko-KR'); }
  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  ts.innerHTML = COUPONS.map(function (c) {
    var ok = ITEMS >= c.min;
    return '<button type="button" class="u445-tk" role="radio" aria-checked="false"' + (ok ? '' : ' disabled') + '><span class="u445-tv' + (c.sm ? ' is-sm' : '') + '">' + esc(c.big) + '</span><span class="u445-tt"><b>' + esc(c.name) + '</b><small>' + esc(ok ? c.cond : T.gap.replace('{gap}', won(c.min - ITEMS))) + '</small></span><span class="u445-tc" aria-hidden="true">' + CHECK + '</span></button>';
  }).join('');
  var TK = ts.querySelectorAll('.u445-tk');
  cnt.textContent = T.count.replace('{n}', COUPONS.length);

  function setMsg(kind, text) {
    msg.classList.toggle('is-err', kind === 'err');
    msg.classList.toggle('is-ok', kind === 'ok');
    msgT.textContent = text;
    fd.classList.toggle('is-err', kind === 'err');
    input.setAttribute('aria-invalid', kind === 'err' ? 'true' : 'false');
  }
  // 총액은 숫자가 굴러가듯 새 금액으로 옮겨 갑니다
  function rollTo(target) {
    cancelAnimationFrame(animT);
    var from = shown, t0 = performance.now(), dur = rm ? 0 : 520;
    (function step(now) {
      if (!root.isConnected) return;
      var k = dur ? Math.min(1, (now - t0) / dur) : 1, e = 1 - Math.pow(1 - k, 3);
      shown = from + (target - from) * e;
      tot.textContent = won(shown);
      go.textContent = T.pay.replace('{p}', won(shown));
      if (k < 1) animT = requestAnimationFrame(step);
    })(t0);
  }
  function apply(c) {
    applied = c;
    var off = c ? c.calc() : 0;
    root.classList.toggle('is-on', !!c);
    if (c) {
      chipB.textContent = c.code;
      chipS.textContent = c.name;
      chipE.textContent = '−' + won(off);
      dl.textContent = T.dc + ' · ' + c.code;
      dv.textContent = '−' + won(off);
    }
    TK.forEach(function (b, i) { b.setAttribute('aria-checked', COUPONS[i] === c ? 'true' : 'false'); });
    rollTo(ITEMS + SHIP - off);
  }
  // 코드 확인: 없는 코드, 끝난 쿠폰, 최소 금액 미달을 각각 다르게 알려 줍니다
  function check() {
    var code = input.value.trim().toUpperCase();
    if (!code || busy) return;
    busy = true;
    ap.disabled = true;
    ap.innerHTML = '<i aria-hidden="true"></i>';
    ap.setAttribute('aria-label', T.checking);
    setTimeout(function () {
      if (!root.isConnected) return;
      busy = false;
      ap.textContent = ap.dataset.t;
      ap.removeAttribute('aria-label');
      ap.disabled = !input.value.trim();
      var c = COUPONS.filter(function (k) { return k.code === code; })[0];
      var err = !c ? (EXPIRED.indexOf(code) >= 0 ? T.exp : T.bad) : ITEMS < c.min ? T.min.replace('{min}', won(c.min)).replace('{gap}', won(c.min - ITEMS)) : '';
      if (err) {
        setMsg('err', err);
        fd.classList.remove('is-shake'); void fd.offsetWidth; fd.classList.add('is-shake');
        input.focus({ preventScroll: true });
        return;
      }
      setMsg('ok', T.ok);
      apply(c);
      x.focus({ preventScroll: true });
    }, rm ? 200 : 750);
  }
  ap.dataset.t = ap.textContent;
  input.addEventListener('input', function () {
    var up = input.value.toUpperCase().replace(/\s/g, '');
    if (up !== input.value) input.value = up;
    if (!busy) ap.disabled = !up;
    if (msg.classList.contains('is-err')) setMsg('', T.hint);
  });
  input.addEventListener('keydown', function (e) { if (e.key === 'Enter') { e.preventDefault(); check(); } });
  ap.addEventListener('click', check);
  x.addEventListener('click', function () {
    apply(null);
    input.value = '';
    ap.disabled = true;
    setMsg('', T.off);
    input.focus({ preventScroll: true });
  });
  TK.forEach(function (b, i) {
    b.addEventListener('click', function () {
      if (b.disabled) return;
      var c = COUPONS[i];
      if (applied === c) { apply(null); setMsg('', T.off); return; }
      input.value = '';
      ap.disabled = true;
      apply(c);
      setMsg('ok', T.ok);
    });
  });
  // 보유 쿠폰 사이를 방향키로 옮깁니다(쓸 수 없는 쿠폰은 건너뜁니다)
  ts.addEventListener('keydown', function (e) {
    var d = e.key === 'ArrowDown' || e.key === 'ArrowRight' ? 1 : e.key === 'ArrowUp' || e.key === 'ArrowLeft' ? -1 : 0;
    if (!d) return;
    e.preventDefault();
    var on = Array.prototype.filter.call(TK, function (b) { return !b.disabled; }), i = on.indexOf(document.activeElement);
    on[(i + d + on.length) % on.length].focus({ preventScroll: true });
  });
  setMsg('', T.hint);
  apply(null);
})();
</script>
