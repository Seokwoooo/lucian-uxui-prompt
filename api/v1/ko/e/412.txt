<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var bg = root.querySelector('.u412-bg'), sh = root.querySelector('.u412-sh');
  var uid = 'u412-' + Math.random().toString(36).slice(2, 7);
  var open = false;
  sh.id = uid + '-m'; bg.setAttribute('aria-controls', sh.id);
  sh.inert = true;
  function set(v, back) {
    open = v;
    root.classList.toggle('is-open', v);
    bg.setAttribute('aria-expanded', String(v));
    bg.setAttribute('aria-label', v ? '메뉴 닫기' : '메뉴 열기');
    sh.inert = !v;
    if (back) bg.focus({ preventScroll: true });
  }
  bg.addEventListener('click', function () { set(!open); });
  root.addEventListener('keydown', function (e) { if (e.key === 'Escape' && open) { e.preventDefault(); e.stopPropagation(); set(false, true); } });
  root.querySelectorAll('a').forEach(function (a) {
    a.addEventListener('click', function (e) { e.preventDefault(); if (sh.contains(a)) set(false, true); });
  });
})();
</script>
