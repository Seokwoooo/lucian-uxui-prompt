<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var sw = root.querySelector('.u466-sw'), fg = root.querySelector('.u466-rg .fg'), tm = root.querySelector('.u466-tm'), sub = root.querySelector('.u466-tx span'), ttl = root.querySelector('.u466-tx b'), live = root.querySelector('.u466-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { on: '집중 모드를 켰어요. 25분 타이머를 시작해요', off: '집중 모드를 껐어요' };
  var TOTAL = 25 * 60, C = 2 * Math.PI * 23, left = TOTAL, t0 = 0, tick = 0, swapT = 0;
  // 같은 요소가 여러 번 놓여도 스위치 이름이 제 카드 제목을 가리키게 합니다
  var uid = 'u466' + Math.random().toString(36).slice(2, 7);
  ttl.id = uid + 't'; sw.setAttribute('aria-labelledby', uid + 't');
  function fmt(s) { var m = Math.floor(s / 60), r = s % 60; return (m < 10 ? '0' : '') + m + ':' + (r < 10 ? '0' : '') + r; }
  function ring(f) { fg.style.strokeDasharray = (C * f).toFixed(2) + ' 200'; }
  function setSub(s) {
    clearTimeout(swapT);
    sub.classList.add('is-swap');
    swapT = setTimeout(function () { sub.textContent = s; sub.classList.remove('is-swap'); }, rm ? 0 : 200);
  }
  // 1초마다 남은 시간을 줄이고 링을 그만큼 비웁니다. 요소가 화면에서 빠지면 멈춥니다
  function loop() {
    if (!root.isConnected) { clearInterval(tick); return; }
    left = Math.max(0, TOTAL - Math.floor((Date.now() - t0) / 1000));
    tm.textContent = fmt(left);
    ring(left / TOTAL);
    if (!left) set(false);
  }
  function set(on) {
    root.classList.toggle('is-on', on);
    sw.setAttribute('aria-checked', String(on));
    clearInterval(tick);
    if (on) {
      t0 = Date.now(); left = TOTAL;
      tm.textContent = fmt(TOTAL);
      ring(1);
      tick = setInterval(loop, 1000);
      setSub('알림 4개를 숨겼어요');
    } else {
      ring(0);
      setSub('알림을 잠시 멈춰요');
    }
    live.textContent = on ? T.on : T.off;
  }
  sw.addEventListener('click', function () { set(sw.getAttribute('aria-checked') !== 'true'); });
})();
</script>
