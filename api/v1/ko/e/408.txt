<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var av = root.querySelector('.u408-av'), m = root.querySelector('.u408-m'), w = root.querySelector('.u408-w');
  var its = [].slice.call(root.querySelectorAll('.u408-it'));
  var uid = 'u408' + Math.random().toString(36).slice(2, 7);
  m.id = uid; av.setAttribute('aria-controls', uid);
  function isOpen() { return root.classList.contains('is-open'); }
  function open(focusAt) {
    root.classList.add('is-open'); av.setAttribute('aria-expanded', 'true');
    if (focusAt != null) its[(focusAt + its.length) % its.length].focus({ preventScroll: true });
  }
  function close(back) {
    if (!isOpen()) return;
    root.classList.remove('is-open'); av.setAttribute('aria-expanded', 'false');
    if (back) av.focus({ preventScroll: true });
  }
  av.addEventListener('click', function () { if (isOpen()) close(); else open(); });
  av.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowDown' || e.key === 'ArrowUp') { e.preventDefault(); open(e.key === 'ArrowDown' ? 0 : -1); }
    else if (e.key === 'Escape' && isOpen()) { e.stopPropagation(); close(); }
  });
  // 메뉴 안: 위아래 화살표로 돌고, Esc나 Tab은 닫습니다(Esc는 갤러리까지 올라가지 않게 막습니다)
  m.addEventListener('keydown', function (e) {
    var i = its.indexOf(document.activeElement);
    if (e.key === 'ArrowDown') { e.preventDefault(); its[(i + 1) % its.length].focus(); }
    else if (e.key === 'ArrowUp') { e.preventDefault(); its[(i - 1 + its.length) % its.length].focus(); }
    else if (e.key === 'Home') { e.preventDefault(); its[0].focus(); }
    else if (e.key === 'End') { e.preventDefault(); its[its.length - 1].focus(); }
    else if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); close(true); }
    else if (e.key === 'Tab') close();
  });
  its.forEach(function (it) { it.addEventListener('click', function () { close(true); }); });
  // 바깥을 누르면 닫힙니다
  function outside(e) {
    if (!root.isConnected) { document.removeEventListener('pointerdown', outside, true); return; }
    if (isOpen() && !w.contains(e.target)) close();
  }
  document.addEventListener('pointerdown', outside, true);
})();
</script>
