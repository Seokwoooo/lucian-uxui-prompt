<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var dial = root.querySelector('.u463-dial'), kb = root.querySelector('.u463-kb'), vb = root.querySelector('.u463-v');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var N = 10, SPAN = 270, STEP = SPAN / N, A0 = -SPAN / 2;
  var value = 4, ang = A0 + value * STEP, vel = 0, target = ang, raf = 0, drag = null, snappy = false;
  var ticks = [];
  for (var i = 0; i <= N; i++) {
    var t = document.createElement('i');
    t.className = 'u463-tk' + (i % 5 === 0 ? ' is-big' : '');
    t.style.setProperty('--u463-a', (A0 + i * STEP) + 'deg');
    dial.insertBefore(t, dial.firstChild);
    ticks.push(t);
  }
  function lights() {
    ticks.forEach(function (t, i) {
      t.classList.toggle('is-on', i < value);
      t.classList.toggle('is-cur', i === value);
    });
    root.dataset.w = value === 0 ? '0' : value <= 3 ? '1' : value <= 7 ? '2' : '3';
    kb.setAttribute('aria-valuenow', String(value));
    kb.setAttribute('aria-valuetext', '볼륨 ' + value);
  }
  // 숫자는 커질 때 아래에서, 작아질 때 위에서 굴러 들어옵니다
  function roll(dir) {
    // 빠르게 여러 칸을 넘기면 나가던 숫자는 바로 치우고, 지금 숫자만 밀어냅니다
    var all = vb.querySelectorAll('span'), old = all[all.length - 1], s = document.createElement('span');
    for (var i = 0; i < all.length - 1; i++) all[i].remove();
    s.textContent = String(value);
    vb.appendChild(s);
    if (rm || !s.animate) { if (old) old.remove(); return; }
    s.animate([{ transform: 'translateY(' + (dir * 22) + 'px)', opacity: 0 }, { transform: 'none', opacity: 1 }], { duration: 300, easing: 'cubic-bezier(.34,1.5,.64,1)' });
    if (old) old.animate([{ transform: 'none', opacity: 1 }, { transform: 'translateY(' + (-dir * 22) + 'px)', opacity: 0 }], { duration: 200, easing: 'ease-in', fill: 'forwards' }).onfinish = function () { old.remove(); };
  }
  function setValue(v) {
    v = Math.max(0, Math.min(N, v));
    if (v === value) return;
    var dir = v > value ? 1 : -1;
    value = v;
    lights();
    roll(dir);
  }
  function paint() { kb.style.setProperty('--u463-ang', ang.toFixed(2) + 'deg'); }
  function frame() {
    raf = 0;
    if (!root.isConnected) return;
    var k = snappy ? 1100 : 340, c = snappy ? 60 : 19;
    for (var i = 0; i < 4; i++) { var a = k * (target - ang) - c * vel; vel += a / 240; ang += vel / 240; }
    paint();
    if (Math.abs(target - ang) < 0.02 && Math.abs(vel) < 0.05) { ang = target; vel = 0; paint(); return; }
    raf = requestAnimationFrame(frame);
  }
  function go(t, fast) {
    target = t; snappy = !!fast;
    if (rm) { ang = t; vel = 0; paint(); return; }
    if (!raf) raf = requestAnimationFrame(frame);
  }
  // 눈금 사이에서는 가운데에 붙어 있다가 반을 넘으면 다음 눈금으로 빠르게 넘어가는 자석 곡선
  function detent(raw) {
    var n = Math.round((raw - A0) / STEP), c = A0 + n * STEP, off = (raw - c) / (STEP / 2);
    var m = Math.sign(off) * Math.pow(Math.abs(off), 2.4);
    return { n: n, a: c + m * STEP / 2 };
  }
  function angleAt(e) {
    var r = kb.getBoundingClientRect();
    return Math.atan2(e.clientX - (r.left + r.width / 2), -(e.clientY - (r.top + r.height / 2))) * 180 / Math.PI;
  }
  kb.addEventListener('pointerdown', function (e) {
    if (e.button !== 0) return;
    drag = { last: angleAt(e), raw: target };
    try { kb.setPointerCapture(e.pointerId); } catch (er) { /* 무시 */ }
  });
  kb.addEventListener('pointermove', function (e) {
    if (!drag) return;
    var a = angleAt(e), d = a - drag.last;
    if (d > 180) d -= 360; else if (d < -180) d += 360;
    drag.last = a;
    drag.raw = Math.max(A0 - 24, Math.min(-A0 + 24, drag.raw + d));
    // 양 끝을 넘겨 돌리면 고무처럼 조금만 따라갑니다
    var raw = drag.raw < A0 ? A0 + (drag.raw - A0) * 0.3 : drag.raw > -A0 ? -A0 + (drag.raw + A0) * 0.3 : drag.raw;
    var dt = detent(Math.max(A0, Math.min(-A0, raw)));
    setValue(dt.n);
    go(raw < A0 || raw > -A0 ? raw : dt.a, true);
  });
  function up() {
    if (!drag) return;
    drag = null;
    go(A0 + value * STEP, false);
  }
  kb.addEventListener('pointerup', up);
  kb.addEventListener('pointercancel', up);
  kb.addEventListener('keydown', function (e) {
    var k = e.key, v = value;
    if (k === 'ArrowRight' || k === 'ArrowUp') v += 1;
    else if (k === 'ArrowLeft' || k === 'ArrowDown') v -= 1;
    else if (k === 'PageUp') v += 2;
    else if (k === 'PageDown') v -= 2;
    else if (k === 'Home') v = 0;
    else if (k === 'End') v = N;
    else return;
    e.preventDefault();
    setValue(v);
    go(A0 + value * STEP, false);
  });
  lights();
  paint();
})();
</script>
