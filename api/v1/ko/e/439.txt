<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var ls = q('.u439-ls'), rl = q('.u439-rl'), sub = q('.u439-hd small'), live = q('.u439-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { loading: '불러오는 중…', updated: '방금 업데이트', done: '새 글 3개를 불러왔어요' };
  // [분류, 제목, 글쓴이, 읽는 시간, 썸네일 배경]
  var POSTS = [
    ['디자인', '디자인 시스템을 처음 만들 때 정해야 할 다섯 가지', '한서연', '5분', 'linear-gradient(135deg,#8B7CF6,#5B8DEF 60%,#4CC9B0)'],
    ['제품', '작은 팀이 릴리스 노트를 잘 쓰는 법', '박지훈', '7분', 'linear-gradient(135deg,#F7B267,#F4845F 55%,#F25C78)'],
    ['접근성', '다크 모드 색은 대비부터 맞추세요', '이수민', '4분', 'linear-gradient(135deg,#1F2937,#3B4A6B 50%,#8EA4D2)']
  ];
  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  ls.innerHTML = POSTS.map(function (p, i) {
    return '<li class="u439-c" style="--d:' + (i * 0.12) + 's">' +
      '<div class="u439-real"><span class="u439-th" style="background:' + p[4] + '" aria-hidden="true"></span><span class="u439-tx"><span><span class="u439-tg">' + esc(p[0]) + '</span><span class="u439-tt">' + esc(p[1]) + '</span></span>' +
      '<span class="u439-mt"><span class="u439-av" aria-hidden="true">' + esc(p[2].charAt(0)) + '</span>' + esc(p[2]) + ' · ' + esc(p[3]) + '</span></span></div>' +
      '<div class="u439-sk" aria-hidden="true"><i class="u439-th"></i><span class="u439-tx"><i></i><i></i><i></i><span><i></i><i></i></span></span></div></li>';
  }).join('');
  var timer = 0;
  // 뼈대를 보여 주고, 데이터가 오면 카드마다 0.12초씩 늦게 내용을 겹쳐 보입니다
  function load() {
    clearTimeout(timer);
    root.classList.add('is-loading');
    ls.setAttribute('aria-busy', 'true');
    sub.textContent = T.loading;
    live.textContent = T.loading;
    timer = setTimeout(function () {
      if (!root.isConnected) return;
      root.classList.remove('is-loading');
      ls.setAttribute('aria-busy', 'false');
      sub.textContent = T.updated;
      live.textContent = T.done;
    }, rm ? 600 : 1500);
  }
  rl.addEventListener('click', load);
  load();
})();
</script>
