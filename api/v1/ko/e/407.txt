<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { busy: '패스키를 확인하는 중이에요', ok: '로그인했어요' };
  var b = root.querySelector('.u407-b'), t = root.querySelector('.u407-t'), fp = root.querySelector('.u407-fp'), live = root.querySelector('.u407-live');
  var uid = 'u407' + Math.random().toString(36).slice(2, 7), gen = 0;
  // 물드는 무늬: 같은 지문을 한 벌 더 두고, 스캔 빛과 함께 내려오는 사각형으로 잘라 보여 줍니다
  var cp = fp.querySelector('.cp'), lg = fp.querySelector('.lg'), ac = fp.querySelector('.ac');
  cp.id = uid + 'c'; lg.id = uid + 'g';
  ac.innerHTML = fp.querySelector('.bs').innerHTML;
  ac.setAttribute('clip-path', 'url(#' + cp.id + ')');
  fp.querySelector('.sl rect').setAttribute('fill', 'url(#' + lg.id + ')');
  // 처음 자리: 글자 왼쪽 아이콘 칸의 가운데
  function place() { b.style.setProperty('--fx', (12 - t.offsetWidth / 2) + 'px'); }
  place();
  if (document.fonts && document.fonts.ready) document.fonts.ready.then(place);
  function later(fn, ms) { var g = gen; setTimeout(function () { if (root.isConnected && g === gen) fn(); }, ms); }
  b.addEventListener('click', function () {
    if (root.classList.contains('is-scan')) return;
    gen += 1;
    root.classList.add('is-scan'); b.setAttribute('aria-busy', 'true'); live.textContent = T.busy;
    later(function () {
      root.classList.add('is-ok'); b.removeAttribute('aria-busy'); live.textContent = T.ok;
      later(function () { root.classList.remove('is-scan', 'is-ok'); }, rm ? 1200 : 1700);
    }, rm ? 800 : 2150);
  });
})();
</script>
