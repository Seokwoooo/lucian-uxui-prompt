<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = {
    ask: '계속을 누르면 얼굴을 확인해요',
    scan: '얼굴을 확인하는 중…',
    ok: '확인했어요',
    cancel: '패스키 로그인을 취소했어요',
    mail: '메일 앱에서 ‘루시안 로그인’ 메일을 열어 주세요',
    bad: '이메일 주소를 확인해 주세요',
    welcome: '패스키로 로그인했어요',
    sent: '로그인 링크를 보냈어요'
  };
  var panes = [].slice.call(root.querySelectorAll('.u407-pane')), main = panes[0], sentP = panes[1], hiP = panes[2];
  var sys = root.querySelector('.u407-sys'), ss = sys.querySelector('.u407-ss'), cont = sys.querySelector('.u407-cont'), cancel = sys.querySelector('.u407-cancel');
  var pk = root.querySelector('.u407-pk'), form = root.querySelector('.u407-ml'), bx = root.querySelector('.u407-bx'), inp = root.querySelector('.u407-in'), send = root.querySelector('.u407-send');
  var note = root.querySelector('.u407-note'), note0 = note.textContent, to = root.querySelector('.u407-to'), live = root.querySelector('.u407-live');
  var sp = sentP.querySelector('p'), sp0 = sp.textContent, gen = 0;
  function later(fn, ms) { var g = gen; setTimeout(function () { if (root.isConnected && g === gen) fn(); }, ms); }
  function show(p, focusSel) {
    panes.forEach(function (x) { var on = x === p; x.classList.toggle('is-cur', on); x.classList.toggle('is-up', !on && x === main); x.inert = !on; });
    var f = focusSel && p.querySelector(focusSel);
    if (f) later(function () { f.focus({ preventScroll: true }); }, rm ? 0 : 350);
  }
  function setNote(text) { note.textContent = text || note0; note.classList.toggle('is-msg', !!text); }
  // 패스키: 시스템 확인 창처럼 올라오는 시트. 계속 → 얼굴 확인 → 체크 → 환영 화면
  function openSys() {
    sys.setAttribute('data-s', 'ask'); ss.textContent = T.ask;
    cont.disabled = false; cancel.disabled = false;
    root.classList.add('is-sys'); main.inert = true; sys.inert = false;
    later(function () { cont.focus({ preventScroll: true }); }, rm ? 0 : 300);
  }
  function closeSys() { root.classList.remove('is-sys'); sys.inert = true; }
  pk.addEventListener('click', function () { setNote(''); openSys(); });
  cancel.addEventListener('click', function () { gen += 1; closeSys(); main.inert = false; setNote(T.cancel); pk.focus({ preventScroll: true }); });
  cont.addEventListener('click', function () {
    if (sys.getAttribute('data-s') !== 'ask') return;
    sys.setAttribute('data-s', 'scan'); ss.textContent = T.scan; cont.disabled = true;
    later(function () {
      sys.setAttribute('data-s', 'ok'); ss.textContent = T.ok; cancel.disabled = true;
      later(function () { closeSys(); show(hiP, '.u407-out'); live.textContent = T.welcome; }, rm ? 300 : 800);
    }, rm ? 400 : 1300);
  });
  sys.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && !cancel.disabled) { e.preventDefault(); e.stopPropagation(); cancel.click(); }
    if (e.key === 'Tab') { e.preventDefault(); (document.activeElement === cont ? cancel : cont).focus(); }
  });
  // 매직 링크: 이메일 확인 → 보내는 중 → ‘메일함을 확인해 주세요’
  root.addEventListener('pointerdown', function (e) { bx.classList.toggle('is-f', bx.contains(e.target)); });
  inp.addEventListener('blur', function () { bx.classList.remove('is-f'); });
  inp.addEventListener('input', function () { if (bx.classList.contains('is-err')) { bx.classList.remove('is-err'); inp.removeAttribute('aria-invalid'); setNote(''); } });
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    var v = inp.value.trim();
    if (send.classList.contains('is-busy')) return;
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v)) { bx.classList.add('is-err'); inp.setAttribute('aria-invalid', 'true'); setNote(T.bad); inp.focus(); return; }
    send.classList.add('is-busy'); send.setAttribute('aria-busy', 'true'); bx.classList.remove('is-f');
    later(function () {
      send.classList.remove('is-busy'); send.removeAttribute('aria-busy');
      to.textContent = v; sp.textContent = sp0; live.textContent = T.sent;
      show(sentP, '.u407-open');
    }, rm ? 300 : 1000);
  });
  // 메일 앱 열기: 실제 서비스에서는 사용자의 메일 서비스로 보냅니다. 여기서는 다음 할 일을 안내합니다
  root.querySelector('.u407-open').addEventListener('click', function () { live.textContent = T.mail; sp.textContent = T.mail; });
  root.querySelector('.u407-lk').addEventListener('click', function () { show(main, '.u407-in'); });
  root.querySelector('.u407-out').addEventListener('click', function () { gen += 1; setNote(''); inp.value = ''; show(main, '.u407-pk'); });
  sys.inert = true;
  show(main);
})();
</script>
