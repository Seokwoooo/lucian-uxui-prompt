<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var cart = q('.u442-cart'), bd = q('.u442-bd'), pd = q('.u442-pd'), th = q('.u442-th'), add = q('.u442-add'), live = q('.u442-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { cart: '장바구니, 상품 {n}개', added: '장바구니에 담았어요' };
  var ART = '<svg viewBox="0 0 28 28" aria-hidden="true"><rect x="10" y="3" width="8" height="3.5" rx="1.4" fill="#2A2D33"/><rect x="9.2" y="6" width="9.6" height="2" rx=".8" fill="#C4C8CF"/><path d="M9.4 8h9.2l-.8 15.6a1.4 1.4 0 0 1-1.4 1.3h-4.8a1.4 1.4 0 0 1-1.4-1.3z" fill="#9DB4A0"/><path d="M10.3 8h2l-.5 16.9h-.6a1.4 1.4 0 0 1-1.4-1.3z" fill="#fff" opacity=".3"/></svg>';
  th.innerHTML = ART;
  var count = 2;
  function badge() { bd.textContent = count; cart.setAttribute('aria-label', T.cart.replace('{n}', count)); }
  function pos(el) { var x = 0, y = 0; for (var e = el; e && e !== root; e = e.offsetParent) { x += e.offsetLeft; y += e.offsetTop; } return { x: x, y: y }; }
  function hit() {
    count++; badge();
    cart.classList.remove('is-hit'); void cart.offsetWidth; cart.classList.add('is-hit');
    live.textContent = T.added;
  }
  // 사진 복사본이 포물선(2차 베지어)을 따라 장바구니로 날아갑니다
  add.addEventListener('click', function () {
    if (rm || !Element.prototype.animate) { hit(); return; }
    var a = pos(th), b = pos(cart), fly = document.createElement('span');
    fly.className = 'u442-fly'; fly.setAttribute('aria-hidden', 'true'); fly.innerHTML = ART;
    root.appendChild(fly);
    var sx = a.x, sy = a.y, ex = b.x + 29 - 28, ey = b.y + 29 - 28, cx = (sx + ex) / 2 - 20, cy = Math.min(sy, ey) - 90, frames = [];
    for (var i = 0; i <= 12; i++) {
      var t = i / 12, u = 1 - t;
      var x = u * u * sx + 2 * u * t * cx + t * t * ex, y = u * u * sy + 2 * u * t * cy + t * t * ey;
      frames.push({ transform: 'translate(' + x.toFixed(1) + 'px,' + y.toFixed(1) + 'px) scale(' + (1 - t * 0.66).toFixed(3) + ') rotate(' + (t * 24).toFixed(1) + 'deg)', opacity: t > 0.92 ? 0 : 1, offset: t });
    }
    var an = fly.animate(frames, { duration: 720, easing: 'cubic-bezier(.45,0,.55,1)', fill: 'forwards' });
    an.onfinish = function () { fly.remove(); if (root.isConnected) hit(); };
  });
  badge();
})();
</script>
