<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var st = root.querySelector('.u422-st');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  // [후기, 이름, 직함, 이니셜, 색]
  var R = [
    ['회의가 끝나면 할 일이 이미 담당자에게 가 있어요.', '박지민', '프로덕트 매니저', '지민', '#7B61FF'],
    ['도구 다섯 개를 하나로 줄였어요. 온보딩이 하루면 끝나요.', '이도윤', '엔지니어링 리드', '도윤', '#0EA5E9'],
    ['매주 손으로 만들던 보고서가 이제 알아서 만들어져요.', '최서아', '운영 팀장', '서아', '#F97316'],
    ['디자이너와 개발자가 같은 화면을 보니 되묻는 일이 줄었어요.', '정하준', '디자인 리드', '하준', '#14B8A6']
  ];
  var STAR = '<path d="m7 1.2 1.75 3.6 3.95.55-2.86 2.78.68 3.92L7 10.2l-3.52 1.85.68-3.92L1.3 5.35l3.95-.55z"/>';
  var cards = R.map(function (r) {
    var c = document.createElement('figure');
    c.className = 'u422-cd';
    c.style.margin = '0';
    c.innerHTML = '<div class="u422-sr" role="img"></div><blockquote class="u422-q"></blockquote><figcaption class="u422-by"><span class="u422-av" aria-hidden="true"></span><span><b></b><small></small></span></figcaption>';
    var sr = c.querySelector('.u422-sr');
    sr.setAttribute('aria-label', '별 5개 만점에 5개');
    for (var k = 0; k < 5; k++) sr.insertAdjacentHTML('beforeend', '<i style="--i:' + k + '"><svg class="bg" viewBox="0 0 14 14" aria-hidden="true">' + STAR + '</svg><svg class="fg" viewBox="0 0 14 14" aria-hidden="true">' + STAR + '</svg></i>');
    c.querySelector('.u422-q').textContent = r[0];
    c.querySelector('b').textContent = r[1];
    c.querySelector('small').textContent = r[2];
    c.querySelector('.u422-av').textContent = r[3];
    c.querySelector('.u422-av').style.background = r[4];
    st.appendChild(c);
    return c;
  });
  var order = cards.slice(), busy = false, tAuto = 0, held = false;
  // 깊이 0은 맨 위, 1·2는 뒤에서 살짝 보이고, 그 뒤는 숨깁니다
  var DEPTH = [[0, 1, 1], [12, .94, 1], [24, .88, 1], [24, .86, 0]];
  function place(c, d) {
    var p = DEPTH[Math.min(d, 3)];
    c.style.setProperty('--dy', p[0] + 'px');
    c.style.setProperty('--s', p[1]);
    c.style.setProperty('--o', p[2]);
    c.style.zIndex = 10 - d;
    c.classList.toggle('is-top', d === 0);
    c.setAttribute('aria-hidden', String(d !== 0));
  }
  function layout() { order.forEach(place); }
  function stars(c) { c.classList.remove('is-on'); void c.offsetWidth; c.classList.add('is-on'); }
  function clearPose(c) { ['--x', '--y', '--r'].forEach(function (k) { c.style.removeProperty(k); }); }
  // 맨 위 카드를 dir 쪽으로 날려 보내고, 맨 뒤로 돌려놓습니다
  function next(dir) {
    if (busy) return;
    busy = true;
    var top = order[0];
    top.classList.add('is-fly');
    top.style.zIndex = 12;
    top.style.setProperty('--x', (dir * 130) + '%');
    top.style.setProperty('--y', '-14px');
    top.style.setProperty('--r', (dir * 14) + 'deg');
    top.style.setProperty('--o', 0);
    order.push(order.shift());
    order.slice(0, -1).forEach(place);
    stars(order[0]);
    setTimeout(function () {
      if (!root.isConnected) return;
      top.classList.remove('is-fly');
      top.style.transition = 'none';
      clearPose(top); place(top, order.length - 1);
      void top.offsetWidth; top.style.transition = '';
      busy = false; schedule();
    }, rm ? 0 : 460);
  }
  // 맨 뒤 카드를 왼쪽 밖에서 데려와 맨 위에 놓습니다
  function prev() {
    if (busy) return;
    busy = true;
    var c = order.pop();
    c.style.transition = 'none';
    c.style.setProperty('--x', '-130%'); c.style.setProperty('--y', '-14px'); c.style.setProperty('--r', '-14deg'); c.style.setProperty('--o', 0);
    c.style.zIndex = 11;
    void c.offsetWidth; c.style.transition = '';
    order.unshift(c);
    clearPose(c); layout(); stars(c);
    setTimeout(function () { busy = false; schedule(); }, rm ? 0 : 400);
  }
  function schedule() {
    clearTimeout(tAuto);
    if (rm || held) return;
    tAuto = setTimeout(function () { if (root.isConnected && !held) next(-1); }, 3200);
  }
  function hold(v) { held = v; if (v) clearTimeout(tAuto); else schedule(); }
  // 끌기: 카드가 손을 따라 기울고, 충분히 끌거나 빠르게 던지면 넘어갑니다
  var drag = null;
  st.addEventListener('pointerdown', function (e) {
    var top = order[0];
    if (busy || !top.contains(e.target) || (e.button && e.button !== 0)) return;
    drag = { x: e.clientX, y: e.clientY, t: performance.now(), lx: e.clientX, lt: performance.now(), dx: 0, k: root.getBoundingClientRect().width / root.offsetWidth || 1 };
    top.classList.add('is-drag');
    try { st.setPointerCapture(e.pointerId); } catch (err) { /* 시연 포인터는 붙잡을 수 없습니다 */ }
    hold(true);
  });
  st.addEventListener('pointermove', function (e) {
    if (!drag) return;
    var top = order[0];
    drag.dx = (e.clientX - drag.x) / drag.k;
    var dy = (e.clientY - drag.y) / drag.k * 0.25;
    drag.v = (e.clientX - drag.lx) / Math.max(1, performance.now() - drag.lt);
    drag.lx = e.clientX; drag.lt = performance.now();
    top.style.setProperty('--x', drag.dx + 'px');
    top.style.setProperty('--y', dy + 'px');
    top.style.setProperty('--r', (drag.dx / 18) + 'deg');
    // 뒤 카드는 끄는 만큼 조금씩 앞으로 다가옵니다
    var p = Math.min(1, Math.abs(drag.dx) / 140), b = order[1];
    b.style.setProperty('--dy', (12 - 12 * p) + 'px');
    b.style.setProperty('--s', .94 + .06 * p);
  });
  function end() {
    if (!drag) return;
    var top = order[0], d = drag;
    drag = null;
    top.classList.remove('is-drag');
    if (Math.abs(d.dx) > 80 || Math.abs(d.v || 0) > 0.6) next(d.dx < 0 ? -1 : 1);
    else { clearPose(top); place(order[1], 1); }
    hold(st.matches(':hover') || st.matches(':focus-visible'));
  }
  st.addEventListener('pointerup', end);
  st.addEventListener('pointercancel', end);
  st.addEventListener('pointerenter', function () { hold(true); });
  // 키보드 초점일 때만 멈춥니다. 마우스로 끌고 나간 뒤에는 다시 흘러갑니다
  st.addEventListener('pointerleave', function () { if (!drag) hold(st.matches(':focus-visible')); });
  st.addEventListener('focus', function () { if (st.matches(':focus-visible')) hold(true); });
  st.addEventListener('blur', function () { hold(st.matches(':hover')); });
  st.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowRight') { e.preventDefault(); next(-1); }
    else if (e.key === 'ArrowLeft') { e.preventDefault(); prev(); }
  });
  layout();
  stars(order[0]);
  schedule();
})();
</script>
