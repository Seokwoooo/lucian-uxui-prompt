<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var st = root.querySelector('.u444-st'), live = root.querySelector('.u444-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var STEPS = ['장바구니', '배송 정보', '결제', '완료'];
  var T = { done: '완료한 단계', cur: '현재 단계' };
  var CHECK = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="m5.5 12.5 4 4 9-9.5"/></svg>';
  var W = 316 / 4;
  STEPS.forEach(function (s, i) {
    if (i) { var ln = document.createElement('li'); ln.className = 'u444-ln'; ln.setAttribute('aria-hidden', 'true'); ln.style.left = (W * (i - 1) + W / 2 + 22) + 'px'; ln.style.width = (W - 44) + 'px'; ln.innerHTML = '<i></i>'; st.appendChild(ln); }
    var li = document.createElement('li');
    li.className = 'u444-s';
    li.innerHTML = '<button type="button" class="u444-dot"><b>' + (i + 1) + '</b>' + CHECK + '</button><span aria-hidden="true"></span>';
    li.querySelector('span').textContent = s;
    st.appendChild(li);
  });
  var S = st.querySelectorAll('.u444-s'), L = st.querySelectorAll('.u444-ln'), D = st.querySelectorAll('.u444-dot');
  var cur = 0;
  // 앞으로 가면 선이 하나씩 차오르고 체크가 차례로 튀어 오릅니다. 뒤로 가면 거꾸로 걷힙니다
  function go(n, focus) {
    var from = cur, step = rm ? 0 : 0.26;
    cur = n;
    L.forEach(function (ln, k) {
      var on = k < cur;
      var d = on && k >= from ? (k - from) * step : !on && k < from ? (from - 1 - k) * step * 0.6 : 0;
      ln.style.setProperty('--d', d + 's');
      ln.classList.toggle('is-on', on);
    });
    S.forEach(function (li, k) {
      var done = k < cur, isCur = k === cur;
      var d = done && k >= from ? (k - from) * step + step : 0;
      li.style.setProperty('--d', d + 's');
      D[k].style.setProperty('--d', d + 's');
      li.classList.toggle('is-done', done);
      li.classList.toggle('is-cur', isCur);
      D[k].setAttribute('aria-label', STEPS[k] + (done ? ', ' + T.done : isCur ? ', ' + T.cur : ''));
      D[k].tabIndex = isCur ? 0 : -1;
      if (isCur) li.setAttribute('aria-current', 'step'); else li.removeAttribute('aria-current');
    });
    live.textContent = STEPS[cur];
    if (focus) D[cur].focus({ preventScroll: true });
  }
  D.forEach(function (b, i) { b.addEventListener('click', function () { if (i !== cur) go(i, false); }); });
  st.addEventListener('keydown', function (e) {
    var d = e.key === 'ArrowRight' ? 1 : e.key === 'ArrowLeft' ? -1 : 0;
    if (!d) return;
    e.preventDefault(); e.stopPropagation();
    var n = Math.max(0, Math.min(3, cur + d));
    if (n !== cur) go(n, true);
  });
  go(1, false);
})();
</script>
