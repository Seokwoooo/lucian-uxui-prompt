<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var st = q('.u444-st'), ln = q('.u444-ln'), bx = q('.u444-bx'), mh = q('.u444-mh span'), mn = q('.u444-mh small'), segs = root.querySelectorAll('.u444-sg i'), live = q('.u444-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var STEPS = ['장바구니', '배송 정보', '결제', '완료'];
  var T = { back: '{n}단계로 돌아가기: {s}', where: '{a}/{b} 단계, {s}', next: '다음: {s}', last: '마지막 단계', done: '완료한 단계', cur: '현재 단계',
    i1: '스테인리스 텀블러', o1: '세이지 · 1개', i2: '세라믹 머그', o2: '아이보리 · 2개', sub: '상품 금액', toShip: '배송 정보 입력',
    who: '한서연 · 010-2481-3579', addr: '서울 성동구 성수이로 113, 3층', def: '기본 배송지', change: '변경', req: '배송 요청', reqV: '문 앞에 놓아 주세요', toPay: '결제로 넘어가기',
    pm: '결제 수단', toss: '토스페이', tossDc: '첫 결제 할인', total: '총 결제 금액', pay: '{p} 결제하기', paying: '결제하는 중…', ok: '주문을 마쳤어요', okSub: '주문번호 {n} · 내일 도착 예정', again: '쇼핑 계속하기' };
  var CHECK = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="m5.5 12.5 4 4 9-9.5"/></svg>';
  var ARROW = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="M5 12h14M13 6l6 6-6 6"/></svg>';
  var cur = 0, busy = false;
  STEPS.forEach(function (s, i) {
    var li = document.createElement('li');
    li.className = 'u444-s';
    li.innerHTML = '<button type="button" class="u444-dot"><b>' + (i + 1) + '</b></button><span aria-hidden="true"></span>';
    li.querySelector('span').textContent = s;
    st.appendChild(li);
  });
  var S = st.querySelectorAll('.u444-s');
  function panel(i) {
    if (i === 0) return '<div class="u444-row"><span class="u444-th"><i></i></span><span class="u444-tx"><b>' + T.i1 + '</b><small>' + T.o1 + '</small></span><span class="u444-pr">₩26,900</span></div>' +
      '<div class="u444-row"><span class="u444-th"><i class="mug"></i></span><span class="u444-tx"><b>' + T.i2 + '</b><small>' + T.o2 + '</small></span><span class="u444-pr">₩24,000</span></div>' +
      '<div class="u444-kv"><span>' + T.sub + '</span><b>₩50,900</b></div><button type="button" class="u444-next">' + T.toShip + ARROW + '</button>';
    if (i === 1) return '<div class="u444-ad"><div><em>' + T.def + '</em><a href="#addr">' + T.change + '</a></div><b>' + T.who + '</b><small>' + T.addr + '</small></div>' +
      '<div class="u444-nt"><span>' + T.req + '</span><span>' + T.reqV + '</span></div><button type="button" class="u444-next">' + T.toPay + ARROW + '</button>';
    if (i === 2) return '<div class="u444-pm"><i aria-hidden="true">toss</i><span>' + T.toss + '</span><small>−₩3,000</small></div>' +
      '<div class="u444-nt"><span>' + T.tossDc + '</span><span>−₩3,000</span></div><div class="u444-kv"><span>' + T.total + '</span><b>₩47,900</b></div>' +
      '<button type="button" class="u444-next">' + T.pay.replace('{p}', '₩47,900') + '</button>';
    return '<div class="u444-dn"><span>' + CHECK + '</span><b>' + T.ok + '</b><small>' + T.okSub.replace('{n}', '20261123-00481') + '</small></div><button type="button" class="u444-next">' + T.again + '</button>';
  }
  // 단계 표시와 내용을 함께 바꿉니다. 앞으로 가면 오른쪽에서, 뒤로 가면 왼쪽에서 밀려 들어옵니다
  function go(i, focusNext) {
    var dir = i > cur ? 1 : i < cur ? -1 : 0;
    cur = i;
    S.forEach(function (li, k) {
      // 완료 화면에서는 마지막 단계까지 모두 체크합니다
      var done = k < cur || cur === 3, isCur = k === cur && cur < 3, b = li.querySelector('.u444-dot');
      li.classList.toggle('is-done', done);
      li.classList.toggle('is-cur', isCur);
      b.innerHTML = done ? CHECK : '<b>' + (k + 1) + '</b>';
      // 끝낸 단계만 누를 수 있고, 완료 화면에서는 되돌아가지 않습니다
      b.disabled = !done || cur === 3;
      b.setAttribute('aria-label', done ? T.back.replace('{n}', k + 1).replace('{s}', STEPS[k]) : STEPS[k] + (isCur ? ', ' + T.cur : ''));
      if (isCur) li.setAttribute('aria-current', 'step'); else li.removeAttribute('aria-current');
    });
    ln.style.setProperty('--p', (cur / 3).toFixed(3));
    var pn = document.createElement('div');
    pn.className = 'u444-pn' + (dir && !rm ? (dir > 0 ? ' is-in-r' : ' is-in-l') : '');
    pn.innerHTML = panel(cur);
    bx.innerHTML = '';
    bx.appendChild(pn);
    mh.textContent = T.where.replace('{a}', cur + 1).replace('{b}', 4).replace('{s}', STEPS[cur]);
    mn.textContent = cur < 3 ? T.next.replace('{s}', STEPS[cur + 1]) : T.last;
    segs.forEach(function (sg, k) { sg.classList.toggle('is-on', k <= cur); });
    live.textContent = mh.textContent;
    if (focusNext) pn.querySelector('.u444-next').focus({ preventScroll: true });
  }
  bx.addEventListener('click', function (e) {
    if (e.target.closest('a')) { e.preventDefault(); return; }
    var b = e.target.closest('.u444-next');
    if (!b || busy) return;
    var kb = e.detail === 0;
    if (cur === 3) { go(0, kb); return; }
    if (cur === 2) {
      // 결제는 잠깐 진행 표시를 보인 뒤 완료로 넘어갑니다
      busy = true;
      b.setAttribute('aria-disabled', 'true');
      b.innerHTML = '<i aria-hidden="true"></i>' + T.paying;
      setTimeout(function () { if (!root.isConnected) return; busy = false; go(3, kb); }, rm ? 200 : 1000);
      return;
    }
    go(cur + 1, kb);
  });
  st.addEventListener('click', function (e) {
    var b = e.target.closest('.u444-dot');
    if (!b || b.disabled || busy) return;
    go(Array.prototype.indexOf.call(S, b.parentNode), true);
  });
  go(0, false);
})();
</script>
