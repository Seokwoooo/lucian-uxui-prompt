<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var ms = Array.prototype.slice.call(root.querySelectorAll('.u414-m'));
  var dotsBox = root.querySelector('.u414-dots'), x = root.querySelector('.u414-x'), undo = root.querySelector('.u414-undo');
  var bar = root.querySelector('.u414-bar');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var n = ms.length, cur = 0, tToast = 0;
  // 막대 하나가 다 차면(animationend) 다음 공지로 넘어갑니다. 멈춤은 CSS animation-play-state가 맡습니다
  var dots = ms.map(function (m, i) {
    var b = document.createElement('button');
    b.type = 'button'; b.className = 'u414-dot'; b.innerHTML = '<i></i>';
    b.setAttribute('aria-label', '공지 ' + (i + 1) + ' / ' + n);
    b.addEventListener('click', function () { show(i); });
    b.querySelector('i').addEventListener('animationend', function () { if (!rm && root.isConnected && i === cur) show((cur + 1) % n); });
    dotsBox.appendChild(b);
    return b;
  });
  function show(i) {
    ms.forEach(function (m, k) {
      m.classList.toggle('is-out', k === cur && k !== i);
      m.classList.toggle('is-on', k === i);
      m.tabIndex = k === i ? 0 : -1;
      m.setAttribute('aria-hidden', String(k !== i));
    });
    var prev = cur;
    cur = i;
    setTimeout(function () { if (ms[prev] && prev !== cur) ms[prev].classList.remove('is-out'); }, 600);
    dots.forEach(function (d, k) {
      d.classList.remove('is-on');
      d.classList.toggle('is-done', k < i);
      d.setAttribute('aria-current', k === i ? 'true' : 'false');
    });
    void dots[i].offsetWidth;
    if (!rm) dots[i].classList.add('is-on'); else dots[i].classList.add('is-done');
  }
  function gone(v) {
    root.classList.toggle('is-gone', v);
    bar.inert = v;
    clearTimeout(tToast);
    root.classList.toggle('is-toast', v);
    undo.tabIndex = v ? 0 : -1;
    if (v) {
      undo.focus({ preventScroll: true });
      tToast = setTimeout(function () { if (root.isConnected) { root.classList.remove('is-toast'); undo.tabIndex = -1; } }, 5000);
    }
  }
  x.addEventListener('click', function () { gone(true); });
  undo.addEventListener('click', function () { gone(false); show(0); x.focus({ preventScroll: true }); });
  root.querySelectorAll('a').forEach(function (a) { a.addEventListener('click', function (e) { e.preventDefault(); }); });
  show(0);
})();
</script>
