<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var tr = q('.u430-tr'), x = q('.u430-x'), help = q('.u430-help'), go = q('.u430-go'), pop = q('.u430-pop'), input = q('.u430-q'), cnt = q('.u430-s small'), lb = q('.u430-lb'), em = q('.u430-em'), live = q('.u430-live');
  var uid = 'u430' + Math.random().toString(36).slice(2, 7);
  var T = { assign: '배정하기', done: '배정했어요', choose: '담당자 선택', help: '담당자를 정하면 바로 알림이 가요.', notify: '{n} 님에게 알림을 보낼게요.', empty: '“{q}” 검색 결과가 없어요', count: '{n}명', count1: '{n}명 ', sel: '선택됨' };
  var CHECK = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="m5 12.5 4.5 4.5L19 7.5"/></svg>';
  // [팀, [[이름, 역할, 색]]]
  var TEAMS = [
    ['디자인팀', [['한서연', '프로덕트 디자이너', 1], ['정우진', 'UX 리서처', 2]]],
    ['개발팀', [['박지훈', '프론트엔드 개발자', 3], ['김민재', '백엔드 개발자', 4], ['배수아', 'iOS 개발자', 5]]],
    ['마케팅팀', [['이수민', '그로스 마케터', 2], ['최예린', '콘텐츠 디자이너', 4]]]
  ];
  var people = [];
  TEAMS.forEach(function (t, g) { t[1].forEach(function (p) { people.push({ name: p[0], role: p[1], c: p[2], team: t[0], g: g }); }); });
  lb.id = uid + 'lb';
  input.setAttribute('aria-controls', lb.id);
  var sel = people[0], act = -1, shown = [];
  // 초성 검색: 가~힣 글자의 첫소리만 뽑아 ㅂㅈ 같은 입력과 견줍니다
  var CHO = '\u3131\u3132\u3134\u3137\u3138\u3139\u3141\u3142\u3143\u3145\u3146\u3147\u3148\u3149\u314A\u314B\u314C\u314D\u314E';
  function cho(s) { return s.replace(/[\uAC00-\uD7A3]/g, function (c) { return CHO.charAt(Math.floor((c.charCodeAt(0) - 0xAC00) / 588)); }); }
  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function mark(s, i, n) { return i < 0 ? esc(s) : esc(s.slice(0, i)) + '<mark>' + esc(s.slice(i, i + n)) + '</mark>' + esc(s.slice(i + n)); }
  function av(p) { return '<span class="u430-av' + (p.c > 1 ? ' c' + p.c : '') + '" aria-hidden="true">' + esc(p.name.charAt(0)) + '</span>'; }

  function render() {
    var t = input.value.trim().toLowerCase(), isCho = /^[\u3131-\u314E]+$/.test(t);
    shown = [];
    var html = '';
    TEAMS.forEach(function (team, g) {
      var rows = '';
      people.forEach(function (p, k) {
        if (p.g !== g) return;
        var ni = -1, ri = -1;
        if (t) {
          ni = isCho ? cho(p.name).indexOf(t) : p.name.toLowerCase().indexOf(t);
          if (ni < 0 && !isCho) ri = p.role.toLowerCase().indexOf(t);
          if (ni < 0 && ri < 0 && team[0].toLowerCase().indexOf(t) < 0) return;
        }
        var on = p === sel;
        rows += '<div class="u430-opt" role="option" id="' + uid + k + '" data-k="' + k + '" aria-selected="' + on + '">' + av(p) + '<b>' + mark(p.name, ni, t.length) + '</b><small>' + mark(p.role, ri, t.length) + '</small>' + CHECK + '</div>';
        shown.push(k);
      });
      if (rows) html += '<div role="group" aria-labelledby="' + uid + 'g' + g + '"><div class="u430-gh" id="' + uid + 'g' + g + '" role="presentation">' + esc(team[0]) + '</div>' + rows + '</div>';
    });
    lb.innerHTML = html;
    lb.hidden = !shown.length;
    em.hidden = !!shown.length;
    if (!shown.length) { var parts = T.empty.split('{q}'); em.textContent = parts[0] + input.value.trim() + (parts[1] || ''); }
    cnt.textContent = (shown.length === 1 ? T.count1 : T.count).replace('{n}', shown.length).trim();
    live.textContent = shown.length ? cnt.textContent : em.textContent;
    var si = sel ? shown.indexOf(people.indexOf(sel)) : -1;
    act = t ? (shown.length ? 0 : -1) : si;
    paint(true);
  }
  function opts() { return lb.querySelectorAll('.u430-opt'); }
  // 목록 상자 안에서만 스크롤합니다(페이지는 움직이지 않습니다)
  function reveal(el, center) {
    var top = el.offsetTop, bot = top + el.offsetHeight, h = lb.clientHeight;
    if (center) lb.scrollTop = Math.max(0, top - h / 2 + el.offsetHeight / 2);
    else if (top - 30 < lb.scrollTop) lb.scrollTop = Math.max(0, top - 30);
    else if (bot + 6 > lb.scrollTop + h) lb.scrollTop = bot + 6 - h;
  }
  function paint(center) {
    var os = opts();
    Array.prototype.forEach.call(os, function (o, i) { o.classList.toggle('is-act', i === act); });
    if (act >= 0 && os[act]) { input.setAttribute('aria-activedescendant', os[act].id); reveal(os[act], center); }
    else input.removeAttribute('aria-activedescendant');
  }
  function syncTrigger() {
    tr.classList.toggle('is-empty', !sel);
    tr.innerHTML = (sel ? av(sel) + '<span class="u430-v">' + esc(sel.name) + '</span><span class="u430-r">' + esc(sel.role) + '</span>'
      : '<span class="u430-av is-none" aria-hidden="true"><svg viewBox="0 0 24 24"><circle cx="12" cy="9" r="3.5"/><path d="M5.5 19a6.5 6.5 0 0 1 13 0"/></svg></span><span class="u430-v">' + T.choose + '</span><span class="u430-r"></span>') +
      '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="m7 10 5 5 5-5"/></svg>';
    tr.setAttribute('aria-label', (sel ? sel.name + ', ' + T.sel : T.choose));
    x.hidden = !sel;
    help.textContent = sel ? T.notify.replace('{n}', sel.name) : T.help;
    go.disabled = !sel;
    go.textContent = T.assign;
  }
  function setOpen(v, focus) {
    root.classList.toggle('is-open', v);
    tr.setAttribute('aria-expanded', v ? 'true' : 'false');
    input.setAttribute('aria-expanded', v ? 'true' : 'false');
    if (v) { input.value = ''; render(); if (focus) input.focus({ preventScroll: true }); }
    else if (focus) tr.focus({ preventScroll: true });
  }
  function choose(k) {
    sel = people[k];
    syncTrigger();
    setOpen(false, true);
    live.textContent = sel.name + ', ' + T.sel;
  }
  function move(d) {
    var n = opts().length;
    if (!n) return;
    act = act < 0 ? (d > 0 ? 0 : n - 1) : Math.max(0, Math.min(n - 1, act + d));
    paint();
  }

  tr.addEventListener('click', function () { setOpen(!root.classList.contains('is-open'), true); });
  tr.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowDown' || e.key === 'ArrowUp') { e.preventDefault(); setOpen(true, true); }
    // 닫힌 채로 글자를 치면 열면서 그 글자로 바로 찾습니다
    else if (e.key.length === 1 && e.key !== ' ' && !e.metaKey && !e.ctrlKey && !e.altKey && !root.classList.contains('is-open')) {
      e.preventDefault(); setOpen(true, true); input.value = e.key; render();
    }
  });
  x.addEventListener('click', function () {
    sel = null; syncTrigger(); setOpen(false, true);
  });
  input.addEventListener('input', render);
  input.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowDown') { e.preventDefault(); move(1); }
    else if (e.key === 'ArrowUp') { e.preventDefault(); move(-1); }
    else if (e.key === 'Home' && !input.value) { e.preventDefault(); act = 0; paint(); }
    else if (e.key === 'End' && !input.value) { e.preventDefault(); act = opts().length - 1; paint(); }
    else if (e.key === 'Enter') { e.preventDefault(); var o = opts()[act]; if (o) choose(+o.dataset.k); }
    else if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); setOpen(false, true); }
    else if (e.key === 'Tab') setOpen(false);
  });
  pop.addEventListener('mousedown', function (e) { if (e.target !== input) e.preventDefault(); });
  lb.addEventListener('pointermove', function (e) {
    var o = e.target.closest('.u430-opt');
    if (!o) return;
    var i = Array.prototype.indexOf.call(opts(), o);
    if (i !== act) { act = i; Array.prototype.forEach.call(opts(), function (el, j) { el.classList.toggle('is-act', j === act); }); input.setAttribute('aria-activedescendant', o.id); }
  });
  lb.addEventListener('click', function (e) { var o = e.target.closest('.u430-opt'); if (o) choose(+o.dataset.k); });
  root.addEventListener('pointerdown', function (e) {
    if (root.classList.contains('is-open') && !e.target.closest('.u430-pop, .u430-tr')) setOpen(false);
  });
  root.addEventListener('focusout', function (e) {
    if (root.classList.contains('is-open') && e.relatedTarget && !pop.contains(e.relatedTarget) && e.relatedTarget !== tr) setOpen(false);
  });
  root.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && root.classList.contains('is-open')) { e.preventDefault(); e.stopPropagation(); setOpen(false, true); }
  });
  // 배정하기: 잠깐 체크로 바뀌었다가 돌아옵니다
  var goT = 0;
  go.addEventListener('click', function () {
    if (!sel) return;
    go.innerHTML = CHECK + T.done;
    clearTimeout(goT);
    goT = setTimeout(function () { if (root.isConnected) go.textContent = T.assign; }, 1600);
  });
  syncTrigger();
  render();
})();
</script>
