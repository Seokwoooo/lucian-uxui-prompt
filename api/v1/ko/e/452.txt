<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var box = q('.u452-box'), ta = q('.u452-ta'), send = q('.u452-send'), md = q('.u452-md'), mv = q('.u452-mv'), live = q('.u452-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { send: '보내기', stop: '생성 중지', gen: '답변을 만드는 중', done: '답변이 끝났어요', stopped: '생성을 멈췄어요', model: '모델: ' };
  var MODELS = ['Nova 2', 'Nova 2 mini'];
  var LH = 24, MAX = 4 * 24;
  var gen = false, ready = false, tm = 0, mi = 0;

  // 줄 수에 맞춰 높이를 재고, 지금 높이에서 새 높이로 부드럽게 늘이거나 줄입니다(네 줄이 넘으면 안에서 스크롤)
  function grow() {
    var h0 = ta.offsetHeight;
    ta.style.transition = 'none';
    ta.style.height = '0px';
    var sh = ta.scrollHeight, h = Math.max(LH, Math.min(MAX, sh));
    ta.style.height = h0 + 'px';
    void ta.offsetHeight;
    ta.style.transition = '';
    ta.style.height = h + 'px';
    ta.style.overflowY = sh > MAX ? 'auto' : 'hidden';
  }
  function bump(k) {
    if (rm || !send.animate) return;
    send.animate([{ transform: 'scale(1)' }, { transform: 'scale(' + k + ')', offset: .35 }, { transform: 'scale(1.06)', offset: .7 }, { transform: 'scale(1)' }], { duration: 460, easing: 'ease-out' });
  }
  function sync() {
    var can = !!ta.value.trim();
    if (can && !ready && !gen) bump(.86);
    ready = can;
    send.setAttribute('aria-disabled', gen || can ? 'false' : 'true');
    send.setAttribute('aria-label', gen ? T.stop : T.send);
  }
  // 보낸 글은 같은 자리에 복제해 위로 날려 보내고, 입력칸은 비우며 한 줄로 줄입니다
  function fly(text) {
    if (rm) return;
    var f = document.createElement('div');
    f.className = 'u452-fly';
    f.textContent = text;
    f.style.left = (box.offsetLeft + ta.offsetLeft) + 'px';
    f.style.top = (box.offsetTop + ta.offsetTop) + 'px';
    f.style.width = ta.clientWidth + 'px';
    root.appendChild(f);
    var a = f.animate([{ opacity: 1, transform: 'none', filter: 'blur(0)' }, { opacity: 0, transform: 'translateY(-22px) scale(.97)', filter: 'blur(3px)' }], { duration: 420, easing: 'cubic-bezier(.3,.6,.3,1)', fill: 'forwards' });
    a.onfinish = function () { f.remove(); };
  }
  function finish(stopped) {
    clearTimeout(tm);
    if (!gen) return;
    gen = false;
    root.classList.remove('is-gen');
    bump(.9);
    sync();
    live.textContent = stopped ? T.stopped : T.done;
  }
  function submit() {
    if (gen) { finish(true); return; }
    if (!ta.value.trim()) return;
    fly(ta.value);
    ta.value = '';
    grow();
    gen = true;
    root.classList.add('is-gen');
    bump(.82);
    sync();
    live.textContent = T.gen;
    tm = setTimeout(function () { if (root.isConnected) finish(false); }, rm ? 800 : 2600);
  }
  ta.addEventListener('input', function () { grow(); sync(); });
  ta.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) { e.preventDefault(); submit(); }
    else if (e.key === 'Escape' && gen) { e.preventDefault(); e.stopPropagation(); finish(true); }
  });
  send.addEventListener('click', submit);
  box.addEventListener('mousedown', function (e) { if (e.target === box) { e.preventDefault(); ta.focus({ preventScroll: true }); } });
  // 모델 알약: 다음 모델 이름이 아래에서 굴러 올라오고, 알약 폭이 새 이름에 맞춰 늘거나 줍니다
  function fitModel(b) {
    b = b || mv.querySelector('b:not(.is-out)');
    if (b && b.offsetWidth) mv.style.width = b.offsetWidth + 'px';
  }
  md.addEventListener('click', function () {
    mi = (mi + 1) % MODELS.length;
    var old = mv.querySelector('b:not(.is-out)'), nb = document.createElement('b');
    nb.textContent = MODELS[mi];
    nb.className = 'is-in';
    mv.appendChild(nb);
    fitModel(nb);
    void nb.offsetWidth;
    nb.className = '';
    if (old) { old.className = 'is-out'; setTimeout(function () { old.remove(); }, rm ? 0 : 500); }
    md.setAttribute('aria-label', T.model + MODELS[mi]);
  });
  fitModel();
  if (document.fonts && document.fonts.ready) document.fonts.ready.then(function () { if (root.isConnected) fitModel(); });
  sync();
})();
</script>
