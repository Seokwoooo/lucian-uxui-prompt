<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { busy: ' 로그인 창을 여는 중이에요', ok: ' 계정과 연결됐어요' };
  var st = root.querySelector('.u402-st'), live = root.querySelector('.u402-live');
  var bs = [].slice.call(st.querySelectorAll('.u402-b'));
  // 상태 표시(스피너→체크와 문구)를 버튼마다 하나씩 붙입니다
  bs.forEach(function (b) {
    var s = document.createElement('span');
    s.className = 'u402-ss'; s.setAttribute('aria-hidden', 'true');
    s.innerHTML = '<i><svg viewBox="0 0 18 18"><path d="M4 9.4l3.2 3.1L14 5.6"/></svg></i><em><span>연결 중…</span><span>연결됐어요</span></em>';
    b.querySelector('.u402-lb').appendChild(s);
  });
  function later(fn, ms) { setTimeout(function () { if (root.isConnected) fn(); }, ms); }
  // 누른 버튼은 묶음 가운데로 모이고, 나머지는 그 버튼 쪽으로 접혀 사라집니다(transform만 쓰므로 자리 이동이 없습니다)
  bs.forEach(function (b, i) {
    b.addEventListener('click', function () {
      if (st.classList.contains('is-pick')) return;
      var name = b.getAttribute('data-n');
      // offsetTop은 루트 기준이므로 묶음의 위치를 빼서 묶음 안 좌표로 바꿉니다
      var mid = (st.offsetHeight - b.offsetHeight) / 2, top = b.offsetTop - st.offsetTop;
      st.classList.add('is-pick');
      bs.forEach(function (o, k) {
        o.style.setProperty('--y', (k === i ? mid - top : (mid - (o.offsetTop - st.offsetTop)) * 0.55) + 'px');
        o.style.setProperty('--d', Math.abs(k - i));
        o.classList.toggle('is-gone', k !== i);
        if (k !== i) { o.tabIndex = -1; o.setAttribute('aria-hidden', 'true'); }
      });
      b.classList.add('is-pick', 'is-busy'); b.setAttribute('aria-busy', 'true');
      live.textContent = name + T.busy;
      later(function () {
        b.classList.remove('is-busy'); b.removeAttribute('aria-busy'); b.classList.add('is-ok');
        live.textContent = name + T.ok;
        later(function () {
          b.classList.remove('is-ok', 'is-pick');
          bs.forEach(function (o, k) {
            o.style.setProperty('--y', '0px');
            o.style.setProperty('--d', Math.max(0, Math.abs(k - i) - 1));
            o.classList.remove('is-gone'); o.removeAttribute('tabindex'); o.removeAttribute('aria-hidden');
          });
          later(function () { st.classList.remove('is-pick'); bs.forEach(function (o) { o.style.removeProperty('--y'); o.style.removeProperty('--d'); }); }, rm ? 0 : 800);
        }, rm ? 800 : 1200);
      }, rm ? 500 : 1500);
    });
  });
})();
</script>
