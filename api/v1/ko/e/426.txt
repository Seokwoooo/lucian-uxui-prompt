<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var gr = root.querySelector('.u426-gr'), th = root.querySelector('.u426-th');
  var seg = Array.prototype.slice.call(root.querySelectorAll('.u426-seg button'));
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  // [전체 값, 올해 값, 전체 소수 자리, 올해 소수 자리, 전체 접미사, 올해 접미사, 단위]
  var S = [
    [3200, 1150, 0, 0, '+', '+', ' 팀'],
    [99.99, 99.98, 2, 2, '%', '%', ''],
    [4.9, 4.8, 1, 1, '', '', '★'],
    [120, 48, 0, 0, '만', '만', ' 건']
  ];
  var LB = [['Haru를 쓰는 팀', '올해 새로 합류한 팀'], ['최근 12개월 가동률', '올해 가동률'], ['앱 평균 평점', '올해 받은 평점'], ['AI가 정리한 회의록', '올해 정리한 회의록']];
  // 추세선 점(0~1). 전체와 올해가 같은 개수라 모양이 부드럽게 바뀝니다
  var SP = [
    [[.1, .16, .22, .3, .42, .55, .7, .88], [.2, .3, .26, .42, .5, .62, .74, .9]],
    [[.86, .9, .84, .92, .88, .94, .9, .93], [.9, .86, .92, .88, .93, .9, .94, .92]],
    [[.6, .66, .7, .68, .76, .8, .84, .88], [.7, .74, .68, .78, .76, .82, .8, .84]],
    [[.05, .1, .18, .28, .4, .56, .72, .92], [.3, .36, .44, .4, .55, .62, .7, .86]]
  ];
  var per = 0, shown = S.map(function () { return 0; }), raf = 0;
  function fmt(v, d) { var s = v.toFixed(d).split('.'); s[0] = s[0].replace(/\B(?=(\d{3})+(?!\d))/g, ','); return s.join('.'); }
  function pts(a) { return a.map(function (y, i) { return [2 + i * 128 / (a.length - 1), 22 - y * 20]; }); }
  function line(a) { return pts(a).map(function (p, i) { return (i ? 'L' : 'M') + p[0].toFixed(1) + ' ' + p[1].toFixed(1); }).join(' '); }
  var cells = S.map(function (s, i) {
    var li = document.createElement('li');
    li.className = 'u426-c';
    li.innerHTML = '<span class="u426-n"><span class="v">0</span><small></small></span><p></p><svg class="u426-sk" viewBox="0 0 132 24" aria-hidden="true"><path class="ar"/><path pathLength="1"/><circle r="2.2"/></svg>';
    gr.appendChild(li);
    return li;
  });
  function label() {
    cells.forEach(function (li, i) {
      var s = S[i];
      li.querySelector('small').textContent = s[4 + per] + s[6];
      li.querySelector('small').classList.toggle('st', s[6] === '★');
      li.querySelector('p').textContent = LB[i][per];
      var a = SP[i][per], d = line(a), last = pts(a)[a.length - 1];
      var paths = li.querySelectorAll('path');
      paths[1].style.d = 'path("' + d + '")';
      paths[0].style.d = 'path("' + d + ' L130 24 L2 24 Z")';
      li.querySelector('circle').style.cx = last[0] + 'px';
      li.querySelector('circle').style.cy = last[1] + 'px';
      li.querySelector('.v').setAttribute('aria-hidden', 'true');
      li.setAttribute('aria-label', fmt(s[per], s[2 + per]) + s[4 + per] + s[6] + ' ' + LB[i][per]);
    });
  }
  // 현재 보이는 값에서 목표 값까지 숫자를 셉니다
  function count(ms) {
    cancelAnimationFrame(raf);
    var from = shown.slice(), to = S.map(function (s) { return s[per]; }), t0 = performance.now();
    var tick = function (now) {
      if (!root.isConnected) return;
      var k = rm ? 1 : Math.min(1, (now - t0) / ms), e = 1 - Math.pow(1 - k, 3);
      cells.forEach(function (li, i) {
        shown[i] = from[i] + (to[i] - from[i]) * e;
        li.querySelector('.v').textContent = fmt(shown[i], S[i][2 + per]);
      });
      if (k < 1) raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);
  }
  function thumb() { var b = seg[per]; th.style.setProperty('--tw', b.offsetWidth + 'px'); th.style.setProperty('--tx', (b.offsetLeft - 2) + 'px'); }
  function set(p, focus) {
    if (p === per) return;
    per = p;
    seg.forEach(function (b, i) { b.setAttribute('aria-checked', String(i === p)); b.tabIndex = i === p ? 0 : -1; });
    thumb(); label(); count(900);
    if (focus) seg[p].focus({ preventScroll: true });
  }
  seg.forEach(function (b, i) {
    b.addEventListener('click', function () { set(i); });
    b.addEventListener('keydown', function (e) {
      if (['ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown'].indexOf(e.key) < 0) return;
      e.preventDefault(); set(1 - per, true);
    });
  });
  label();
  th.style.transition = 'none'; thumb(); void th.offsetWidth; th.style.transition = '';
  // 화면에 들어오면 한 번 세고 추세선을 그립니다
  function reveal() { if (root.classList.contains('is-in')) return; root.classList.add('is-in'); count(1500); }
  if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (en) { if (en[0].isIntersecting) { reveal(); io.disconnect(); } }, { threshold: 0.4 });
    io.observe(root);
  } else reveal();
})();
</script>
