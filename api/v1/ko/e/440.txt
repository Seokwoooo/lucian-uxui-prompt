<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var h = q('.u440-h'), p = q('.u440-p'), go = q('.u440-go'), tp = q('.u440-tp'), cnt = q('.u440-ch small'), bar = q('.u440-bar'), rows = q('.u440-rows'), live = q('.u440-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = {
    h0: '아직 프로젝트가 없어요', p0: '첫 프로젝트를 만들면 파일과 할 일, 팀원을 한곳에서 관리할 수 있어요.',
    h1: '첫 프로젝트가 생겼어요', p1: '이제 팀원을 초대해서 함께 써 보세요.',
    h2: '팀이 모였어요', p2: '템플릿으로 첫 문서를 빠르게 채워 보세요.',
    h3: '준비가 끝났어요', p3: '이제 워크스페이스를 마음껏 써 보세요.',
    done: '완료', count: '{a}/{b} 완료', open: '워크스페이스로 가기', busy0: '만드는 중…', busy1: '보내는 중…'
  };
  // [이름, 보조 설명]
  var STEPS = [['첫 프로젝트 만들기', '1분이면 돼요'], ['팀원 초대하기', '링크나 이메일로'], ['템플릿 둘러보기', '회의록, 기획서, 회고']];
  var CHECK = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="m5.5 12.5 4 4 9-9.5"/></svg>';
  var PLUS = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="M12 5v14M5 12h14"/></svg>';
  var done = [false, false, false], busy = false;
  rows.innerHTML = STEPS.map(function (s) { return '<button type="button" class="u440-row"><span class="u440-cc" aria-hidden="true">' + CHECK + '</span><span class="u440-rt"><b></b></span><span class="u440-rs"></span></button>'; }).join('');
  var R = rows.querySelectorAll('.u440-row');
  R.forEach(function (r, i) { r.querySelector('b').textContent = STEPS[i][0]; });

  function next() { for (var i = 0; i < 2; i++) if (!done[i]) return i; return done[2] ? -1 : 2; }
  function render(animate) {
    var n = done.filter(Boolean).length, nx = next();
    // 머리말은 앞 단계부터 차례로 채워진 만큼을 따릅니다
    var lvl = done[0] ? (done[1] ? (done[2] ? 3 : 2) : 1) : 0;
    var nh = T['h' + lvl], np = T['p' + lvl];
    if (h.textContent !== nh) { h.textContent = nh; p.textContent = np; if (animate) [h, p].forEach(function (el) { el.classList.remove('is-new'); void el.offsetWidth; el.classList.add('is-new'); }); }
    root.classList.toggle('s1', done[0]);
    root.classList.toggle('s2', done[1]);
    root.classList.toggle('s3', n === 3);
    if (!busy) go.innerHTML = (nx < 0 ? '' : PLUS) + '<span>' + (nx < 0 ? T.open : STEPS[nx][0]) + '</span>';
    // 다음 할 일이 템플릿이거나 모두 끝나면 보조 버튼을 숨기고 주 버튼이 폭을 다 씁니다
    tp.hidden = nx === 2 || nx < 0;
    cnt.textContent = T.count.replace('{a}', n).replace('{b}', 3);
    bar.style.setProperty('--p', n / 3);
    bar.setAttribute('aria-valuenow', n);
    R.forEach(function (r, i) {
      r.classList.toggle('is-done', done[i]);
      r.classList.toggle('is-next', i === nx);
      r.setAttribute('aria-disabled', done[i] ? 'true' : 'false');
      r.setAttribute('aria-label', STEPS[i][0] + (done[i] ? ', ' + T.done : ''));
      r.querySelector('.u440-rs').textContent = done[i] ? T.done : STEPS[i][1];
    });
  }
  // 만들기와 초대는 잠깐 진행 표시를 보인 뒤 체크하고, 템플릿은 바로 체크합니다
  function act(i) {
    if (busy || i < 0 || done[i]) return;
    if (i === 2) { done[2] = true; render(true); live.textContent = STEPS[2][0] + ', ' + T.done; return; }
    busy = true;
    go.innerHTML = '<i aria-hidden="true"></i><span>' + (i === 0 ? T.busy0 : T.busy1) + '</span>';
    go.setAttribute('aria-busy', 'true');
    setTimeout(function () {
      if (!root.isConnected) return;
      busy = false;
      go.removeAttribute('aria-busy');
      done[i] = true;
      render(true);
      live.textContent = STEPS[i][0] + ', ' + T.done;
    }, rm ? 150 : 650);
  }
  go.addEventListener('click', function () { var nx = next(); if (nx >= 0) act(nx); });
  tp.addEventListener('click', function () { act(2); });
  R.forEach(function (r, i) { r.addEventListener('click', function () { act(i); }); });
  render(false);
})();
</script>
