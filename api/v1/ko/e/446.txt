<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { avg: '5점 만점에 4.6점', row: function (s, n, p) { return s + '점 ' + n + '개, ' + p + '%'; }, cnt: function (n) { return n.toLocaleString('ko-KR') + '개'; } };
  var AVG = 4.6, DIST = [980, 190, 62, 28, 24];
  var total = DIST.reduce(function (a, b) { return a + b; }, 0), max = Math.max.apply(null, DIST);
  var num = root.querySelector('.u446-n'), st = root.querySelector('.u446-st'), ls = root.querySelector('.u446-ls');
  // 별: 18px 별 다섯 개를 3px 간격으로. 채움은 반 개 단위로 반올림해 잘라 보여 줍니다
  var STAR = 'M9 1.3l2.3 4.8 5.2.7-3.8 3.6.9 5.2L9 13.1l-4.6 2.5.9-5.2L1.5 6.8l5.2-.7z';
  var stars = [0, 1, 2, 3, 4].map(function (k) { return '<path transform="translate(' + k * 20.75 + ' 0)" d="' + STAR + '"/>'; }).join('');
  st.querySelector('.bs').innerHTML = stars; st.querySelector('.fl').innerHTML = stars;
  var half = Math.round(AVG * 2) / 2, whole = Math.floor(half);
  var w = whole * 20.75 + (half - whole) * 18;
  st.style.setProperty('--cut', (101 - w).toFixed(2) + 'px');
  st.setAttribute('aria-label', T.avg);
  // 분포 막대
  var STAR_S = '<svg viewBox="0 0 18 18" aria-hidden="true"><path d="' + STAR + '"/></svg>';
  ls.innerHTML = DIST.map(function (n, k) {
    var p = Math.round(n / total * 100);
    return '<li class="u446-row" tabindex="0" aria-label="' + T.row(5 - k, n, p) + '"><span aria-hidden="true">' + (5 - k) + STAR_S + '</span>' +
      '<span class="u446-bar" aria-hidden="true"><i style="--k:' + k + ';--p:' + (n / max).toFixed(3) + '"></i></span>' +
      '<span class="u446-v" aria-hidden="true"><b>' + p + '%</b><em>' + T.cnt(n) + '</em></span></li>';
  }).join('');
  var rows = [].slice.call(ls.children);
  function on(r) { rows.forEach(function (x) { x.classList.toggle('is-on', x === r); }); ls.classList.toggle('is-hov', !!r); }
  rows.forEach(function (r) {
    r.addEventListener('pointerenter', function () { on(r); });
    r.addEventListener('focus', function () { on(r); });
    r.addEventListener('blur', function () { on(null); });
  });
  ls.addEventListener('pointerleave', function () { on(null); });
  // 화면에 들어오면 숫자가 세어 올라가고 별과 막대가 찹니다
  function play() {
    root.classList.add('is-in');
    if (rm) { num.textContent = AVG.toFixed(1); return; }
    var t0 = performance.now(), dur = 1000;
    (function step(t) {
      if (!root.isConnected) return;
      var k = Math.min(1, (t - t0) / dur), e = 1 - Math.pow(1 - k, 3);
      num.textContent = (AVG * e).toFixed(1);
      if (k < 1) requestAnimationFrame(step);
    })(t0);
  }
  if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (es) { if (es.some(function (e) { return e.isIntersecting; })) { io.disconnect(); play(); } }, { threshold: .4 });
    io.observe(root);
  } else play();
})();
</script>
