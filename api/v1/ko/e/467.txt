<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var cd = root.querySelector('.u467-cd'), ta = root.querySelector('.u467-ta'), mr = root.querySelector('.u467-mr'), rg = root.querySelector('.u467-rg'), fg = root.querySelector('.u467-rg .fg'), num = root.querySelector('.u467-rg b'), sv = root.querySelector('.u467-sv'), lb = root.querySelector('.u467-lb'), live = root.querySelector('.u467-live');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = {
    left: function (n) { return n + '자 남았어요'; },
    over: function (n) { return n + '자 넘었어요'; }
  };
  var MAX = 40, WARN = 8, C = 2 * Math.PI * 12, state = '', count = 0;
  // 같은 요소가 여러 번 놓여도 라벨이 제 입력칸을 가리키게 합니다
  var uid = 'u467' + Math.random().toString(36).slice(2, 7);
  ta.id = uid; lb.setAttribute('for', uid);
  function esc(s) { return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }
  function again(el, c) { if (rm) return; el.classList.remove(c); void el.offsetWidth; el.classList.add(c); }
  function update(typed) {
    var chars = Array.from(ta.value), n = chars.length, left = MAX - n;
    // 넘친 글자는 뒤에 깔린 거울 층에서 형광펜으로 칠합니다(입력칸 글자 자리와 똑같이 줄바꿈됩니다)
    var head = chars.slice(0, MAX).join(''), tail = chars.slice(MAX).join('');
    mr.innerHTML = esc(head) + (tail ? '<mark>' + esc(tail) + '</mark>' : '') + '​';
    mr.scrollTop = ta.scrollTop;
    fg.style.strokeDasharray = (C * Math.min(1, n / MAX)).toFixed(2) + ' ' + C.toFixed(2);
    num.textContent = left < 0 ? '−' + (-left) : String(left);
    var next = left < 0 ? 'over' : left <= WARN ? 'warn' : 'ok';
    root.classList.toggle('is-warn', next === 'warn');
    root.classList.toggle('is-over', next === 'over');
    sv.disabled = n === 0 || left < 0;
    ta.setAttribute('aria-invalid', String(left < 0));
    if (typed && next === 'over') {
      if (state !== 'over') again(cd, 'is-shake');
      else if (n > count) again(rg, 'is-nudge');
    }
    if (next !== state || (next !== 'ok' && typed && (left === 0 || left === WARN || left === -1))) live.textContent = next === 'over' ? T.over(-left) : next === 'warn' ? T.left(left) : '';
    state = next; count = n;
  }
  ta.addEventListener('input', function () { update(true); });
  ta.addEventListener('scroll', function () { mr.scrollTop = ta.scrollTop; });
  cd.addEventListener('animationend', function (e) { if (e.target === cd) cd.classList.remove('is-shake'); });
  update(false);
})();
</script>
