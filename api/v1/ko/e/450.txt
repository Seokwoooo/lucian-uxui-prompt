<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var T = { openL: '상담 창 열기', closeL: '상담 창 닫기', joined: '상담원이 연결됐어요' };
  var sc = root.querySelector('.u450-sc'), top = root.querySelector('.u450-top'), fab = root.querySelector('.u450-fab'), pn = root.querySelector('.u450-pn');
  var start = root.querySelector('.u450-start'), live = root.querySelector('.u450-live'), raf = 0, gen = 0;
  var uid = 'u450' + Math.random().toString(36).slice(2, 8);
  pn.id = uid + 'p'; fab.setAttribute('aria-controls', pn.id); fab.setAttribute('aria-label', T.openL);
  function later(fn, ms) { var g = gen; setTimeout(function () { if (root.isConnected && g === gen) fn(); }, ms); }
  // 스크롤: 맨 위로 버튼은 120px 넘게 내려가면 나타나고, 링과 위쪽 막대가 읽은 비율만큼 찹니다
  function onScroll() {
    raf = 0;
    var max = sc.scrollHeight - sc.clientHeight, p = max > 0 ? Math.min(1, sc.scrollTop / max) : 0;
    root.style.setProperty('--p', p.toFixed(4));
    var show = sc.scrollTop > 120;
    if (show !== top.classList.contains('is-show')) { top.classList.toggle('is-show', show); top.tabIndex = show ? 0 : -1; }
  }
  sc.addEventListener('scroll', function () { if (!raf) raf = requestAnimationFrame(onScroll); }, { passive: true });
  top.addEventListener('click', function () {
    sc.scrollTo({ top: 0, behavior: rm ? 'auto' : 'smooth' });
    later(function () { sc.focus({ preventScroll: true }); }, rm ? 0 : 500);
  });
  function setOpen(on, refocus) {
    root.classList.toggle('is-open', on);
    fab.setAttribute('aria-expanded', String(on));
    fab.setAttribute('aria-label', on ? T.closeL : T.openL);
    pn.inert = !on;
    if (on) later(function () { pn.focus({ preventScroll: true }); }, rm ? 0 : 120);
    else if (refocus) fab.focus({ preventScroll: true });
  }
  fab.addEventListener('click', function () { setOpen(!root.classList.contains('is-open')); });
  pn.addEventListener('keydown', function (e) { if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); setOpen(false, true); } });
  [].slice.call(pn.querySelectorAll('.u450-fq a')).forEach(function (a) { a.addEventListener('click', function (e) { e.preventDefault(); }); });
  // 상담 시작: 연결 중 표시 뒤 상담원 첫 메시지와 입력칸으로 바뀝니다
  start.addEventListener('click', function () {
    if (start.classList.contains('is-busy')) return;
    start.classList.add('is-busy'); start.setAttribute('aria-busy', 'true');
    later(function () {
      start.classList.remove('is-busy'); start.removeAttribute('aria-busy');
      pn.classList.add('is-chat'); live.textContent = T.joined;
      // 시작 버튼이 사라지므로 초점을 메시지 입력칸으로 옮겨 상담 창 안에 둡니다
      pn.querySelector('.u450-in input').focus({ preventScroll: true });
    }, rm ? 300 : 1100);
  });
  pn.inert = true;
  onScroll();
})();
</script>
