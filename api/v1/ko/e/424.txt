<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var its = Array.prototype.slice.call(root.querySelectorAll('.u424-it'));
  var qs = its.map(function (it) { return it.querySelector('.u424-q'); });
  var uid = 'u424-' + Math.random().toString(36).slice(2, 7);
  its.forEach(function (it, i) {
    var a = it.querySelector('.u424-a');
    qs[i].id = uid + '-q' + i; a.id = uid + '-a' + i;
    qs[i].setAttribute('aria-controls', a.id); a.setAttribute('aria-labelledby', qs[i].id);
    a.inert = true;
  });
  // 한 번에 하나만 열립니다
  function toggle(i) {
    its.forEach(function (it, k) {
      var v = k === i && !it.classList.contains('open');
      it.classList.toggle('open', v);
      qs[k].setAttribute('aria-expanded', String(v));
      it.querySelector('.u424-a').inert = !v;
    });
  }
  qs.forEach(function (q, i) {
    q.addEventListener('click', function () { toggle(i); });
    q.addEventListener('keydown', function (e) {
      var n = qs.length, to = e.key === 'ArrowDown' ? i + 1 : e.key === 'ArrowUp' ? i - 1 : e.key === 'Home' ? 0 : e.key === 'End' ? n - 1 : null;
      if (to === null) return;
      e.preventDefault(); qs[(to + n) % n].focus();
    });
  });
})();
</script>
