<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = function (s) { return root.querySelector(s); };
  var box = q('.u451-box'), ed = q('.u451-ed'), ta = q('.u451-ta'), hl = q('.u451-hl'), mz = q('.u451-mz');
  var pop = q('.u451-pop'), pill = q('.u451-pill'), lb = q('.u451-lb'), live = q('.u451-live'), send = q('.u451-send'), atb = q('.u451-at');
  var uid = 'u451' + Math.random().toString(36).slice(2, 7);
  lb.id = uid + 'l';
  ta.setAttribute('aria-controls', lb.id);
  var T = { count: '멤버 {n}명', none: '일치하는 멤버가 없어요', added: '{name} 님을 멘션했어요', sent: '댓글을 남겼어요' };
  // [이름, 팀, 아바타 그라디언트 두 색]
  var PEOPLE = [['서지훈', '개발', '#7CCBFF', '#3B6CF0'], ['한서연', '디자인', '#FFB48C', '#F2557A'], ['김민준', 'PM', '#8EE3BE', '#139A72'], ['이수아', '마케팅', '#D2C2FF', '#7458F0']];
  var names = PEOPLE.map(function (p) { return p[0]; });
  var LH = 22, ROW = 38;
  var open = false, anchor = -1, vis = [], act = 0, popAt = -1, dismissed = -1;

  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  lb.innerHTML = PEOPLE.map(function (p, i) {
    return '<div class="u451-o" role="option" id="' + uid + i + '" aria-selected="false"><span class="u451-av" aria-hidden="true" style="--u451-a:' + p[2] + ';--u451-b:' + p[3] + '">' + esc(p[0].charAt(0)) + '</span><span class="u451-nm">' + esc(p[0]) + '</span><span class="u451-rl">' + esc(p[1]) + '</span></div>';
  }).join('');
  var rows = [].slice.call(lb.children);
  var reM = new RegExp('@(' + names.slice().sort(function (a, b) { return b.length - a.length; }).map(function (n) { return n.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); }).join('|') + ')', 'g');

  // 글자는 투명한 입력칸 뒤의 거울 층이 그립니다. 멤버 이름은 이 층에서만 칩 모양이 되어, 커서 위치는 입력칸과 똑같이 맞습니다
  function paintText() {
    var v = ta.value, out = '', last = 0, m;
    reM.lastIndex = 0;
    while ((m = reM.exec(v))) {
      out += esc(v.slice(last, m.index)) + '<span class="u451-m' + (m.index === popAt ? ' is-pop' : '') + '">' + esc(m[0]) + '</span>';
      last = m.index + m[0].length;
    }
    hl.innerHTML = out + esc(v.slice(last)) + '​';
    popAt = -1;
    var pp = hl.querySelector('.is-pop');
    if (pp) pp.addEventListener('animationend', function () { pp.classList.remove('is-pop'); });
    hl.scrollTop = ta.scrollTop;
    send.classList.toggle('is-on', !!v.trim());
  }
  // @가 놓인 글자 위치를 거울 층으로 재서, 목록을 그 글자 바로 아래(공간이 모자라면 위)에 붙입니다
  function place() {
    mz.textContent = ta.value.slice(0, anchor);
    var mk = document.createElement('span');
    mk.textContent = '@';
    mz.appendChild(mk);
    var ax = box.offsetLeft + ed.offsetLeft + mk.offsetLeft;
    var ay = box.offsetTop + ed.offsetTop + mk.offsetTop - ta.scrollTop;
    var x = Math.max(10, Math.min(root.offsetWidth - pop.offsetWidth - 10, ax - 12));
    var h = 10 + vis.length * ROW, y = ay + LH + 6, up = y + h > root.offsetHeight - 6;
    if (up) y = ay - h - 6;
    pop.style.left = x + 'px';
    pop.style.top = y + 'px';
    pop.style.transformOrigin = (ax - x + 6) + 'px ' + (up ? '100%' : '0');
  }
  function paint(snap) {
    rows.forEach(function (r, i) { r.setAttribute('aria-selected', vis[act] === i ? 'true' : 'false'); });
    if (snap) pill.style.transition = 'none';
    pill.style.setProperty('--u451-py', act * ROW + 'px');
    if (snap) { void pill.offsetWidth; pill.style.transition = ''; }
    ta.setAttribute('aria-activedescendant', uid + vis[act]);
  }
  function filter(query) {
    var t = query.toLowerCase();
    vis = [];
    rows.forEach(function (r, i) {
      var n = names[i], k = n.toLowerCase().indexOf(t), on = k >= 0;
      if (on) vis.push(i);
      r.classList.toggle('is-out', !on);
      r.setAttribute('aria-hidden', on ? 'false' : 'true');
      r.querySelector('.u451-nm').innerHTML = on && t ? esc(n.slice(0, k)) + '<mark>' + esc(n.slice(k, k + t.length)) + '</mark>' + esc(n.slice(k + t.length)) : esc(n);
    });
    return vis.length;
  }
  function setOpen(v) {
    open = v;
    root.classList.toggle('is-open', v);
    ta.setAttribute('aria-expanded', v ? 'true' : 'false');
    if (!v) ta.removeAttribute('aria-activedescendant');
  }
  function detect() {
    var c = ta.selectionStart, m = /(^|\s)@([^\s@]{0,16})$/.exec(ta.value.slice(0, c));
    if (!m) { dismissed = -1; setOpen(false); return; }
    var a = c - m[2].length - 1;
    if (a === dismissed) return;
    var fresh = !open || a !== anchor;
    anchor = a;
    if (fresh) pop.classList.add('is-snap');
    var n = filter(m[2]);
    if (!n) { setOpen(false); live.textContent = T.none; pop.classList.remove('is-snap'); return; }
    if (fresh) act = 0; else act = Math.min(act, n - 1);
    place();
    paint(fresh);
    if (fresh) { void pop.offsetWidth; pop.classList.remove('is-snap'); }
    setOpen(true);
    live.textContent = T.count.replace('{n}', n);
  }
  function pick(i) {
    var v = ta.value, c = ta.selectionStart, ins = '@' + names[i] + ' ';
    ta.value = v.slice(0, anchor) + ins + v.slice(c);
    var pos = anchor + ins.length;
    ta.setSelectionRange(pos, pos);
    popAt = anchor;
    setOpen(false);
    paintText();
    live.textContent = T.added.replace('{name}', names[i]);
  }

  ta.addEventListener('input', function () { paintText(); detect(); });
  ta.addEventListener('scroll', function () { hl.scrollTop = ta.scrollTop; if (open) place(); });
  ta.addEventListener('click', detect);
  ta.addEventListener('keyup', function (e) { if (/^(ArrowLeft|ArrowRight|Home|End)$/.test(e.key)) detect(); });
  ta.addEventListener('blur', function () { setOpen(false); });
  ta.addEventListener('keydown', function (e) {
    if (open && (e.key === 'ArrowDown' || e.key === 'ArrowUp')) {
      e.preventDefault();
      act = (act + (e.key === 'ArrowDown' ? 1 : -1) + vis.length) % vis.length;
      paint(false);
      return;
    }
    if (open && (e.key === 'Enter' || e.key === 'Tab')) { e.preventDefault(); pick(vis[act]); return; }
    if (open && e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); dismissed = anchor; setOpen(false); return; }
    // 칩 바로 뒤에서 지우면 이름 전체를 한 번에 지웁니다
    if (e.key === 'Backspace' && ta.selectionStart === ta.selectionEnd) {
      var c = ta.selectionStart, before = ta.value.slice(0, c);
      for (var i = 0; i < names.length; i++) {
        var tok = '@' + names[i];
        if (before.slice(-tok.length) === tok) {
          e.preventDefault();
          ta.value = before.slice(0, -tok.length) + ta.value.slice(c);
          ta.setSelectionRange(c - tok.length, c - tok.length);
          paintText();
          detect();
          return;
        }
      }
    }
  });
  pop.addEventListener('mousedown', function (e) { e.preventDefault(); });
  lb.addEventListener('pointermove', function (e) {
    var o = e.target.closest('.u451-o');
    if (!o) return;
    var k = vis.indexOf(rows.indexOf(o));
    if (k >= 0 && k !== act) { act = k; paint(false); }
  });
  lb.addEventListener('click', function (e) {
    var o = e.target.closest('.u451-o');
    if (o && !o.classList.contains('is-out')) pick(rows.indexOf(o));
  });
  box.addEventListener('mousedown', function (e) { if (e.target === box || e.target === ed) { e.preventDefault(); ta.focus({ preventScroll: true }); } });
  atb.addEventListener('mousedown', function (e) { e.preventDefault(); });
  // @ 버튼: 커서 자리(입력칸에 초점이 없으면 글 끝)에 @를 넣고 목록을 엽니다
  atb.addEventListener('click', function () {
    var v = ta.value, focused = document.activeElement === ta;
    var s = focused ? ta.selectionStart : v.length, en = focused ? ta.selectionEnd : v.length;
    var pre = v.slice(0, s), ins = (pre && !/\s$/.test(pre) ? ' ' : '') + '@';
    ta.value = pre + ins + v.slice(en);
    ta.focus({ preventScroll: true });
    ta.setSelectionRange(s + ins.length, s + ins.length);
    dismissed = -1;
    paintText();
    detect();
  });
  send.addEventListener('click', function () {
    if (!ta.value.trim()) return;
    setOpen(false);
    var done = function () { if (!root.isConnected) return; ta.value = ''; paintText(); live.textContent = T.sent; };
    if (hl.animate && !window.matchMedia('(prefers-reduced-motion: reduce)').matches) hl.animate([{ opacity: 1, transform: 'none' }, { opacity: 0, transform: 'translateY(-10px)' }], { duration: 240, easing: 'cubic-bezier(.4,0,.6,1)' }).onfinish = done;
    else done();
  });

  paintText();
  ta.setSelectionRange(ta.value.length, ta.value.length);
  detect();
})();
</script>
