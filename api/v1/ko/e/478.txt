<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var url = root.dataset.url, box = root.querySelector('.u478-q'), cp = root.querySelector('.u478-cp'), live = root.querySelector('.u478-live');
  root.querySelector('.u478-u').textContent = url.replace(/^https?:\/\//, '');

  // ── 작은 QR 인코더: 바이트 모드, 오류 정정 M, 버전 1~6 ──
  var EC = [null, [10, 1, 16], [16, 1, 28], [26, 1, 44], [18, 2, 32], [24, 2, 43], [16, 4, 27]]; // [블록당 정정 바이트, 블록 수, 블록당 데이터 바이트]
  var AL = [null, [], [6, 18], [6, 22], [6, 26], [6, 30], [6, 34]];
  function mul(x, y) { var z = 0; for (var i = 7; i >= 0; i--) { z = (z << 1) ^ ((z >>> 7) * 0x11D); z ^= ((y >>> i) & 1) * x; } return z; }
  function rsDiv(n) {
    var r = []; for (var i = 0; i < n - 1; i++) r.push(0); r.push(1);
    var g = 1;
    for (i = 0; i < n; i++) { for (var j = 0; j < r.length; j++) { r[j] = mul(r[j], g); if (j + 1 < r.length) r[j] ^= r[j + 1]; } g = mul(g, 2); }
    return r;
  }
  function rsRem(data, div) {
    var r = div.map(function () { return 0; });
    data.forEach(function (b) { var f = b ^ r.shift(); r.push(0); div.forEach(function (c, i) { r[i] ^= mul(c, f); }); });
    return r;
  }
  function qr(text) {
    var bytes = [].slice.call(new TextEncoder().encode(text)), v = 1;
    while (v < 6 && 12 + bytes.length * 8 > EC[v][1] * EC[v][2] * 8) v++;
    var ecn = EC[v][0], nb = EC[v][1], dpb = EC[v][2], cap = nb * dpb * 8;
    var bits = [];
    function put(val, len) { for (var i = len - 1; i >= 0; i--) bits.push((val >>> i) & 1); }
    put(4, 4); put(bytes.length, 8); bytes.forEach(function (b) { put(b, 8); });
    put(0, Math.min(4, cap - bits.length));
    while (bits.length % 8) bits.push(0);
    for (var pad = 0xEC; bits.length < cap; pad ^= 0xEC ^ 0x11) put(pad, 8);
    var data = []; for (var i = 0; i < bits.length; i += 8) { var b = 0; for (var k = 0; k < 8; k++) b = b << 1 | bits[i + k]; data.push(b); }
    // 블록마다 정정 바이트를 붙이고 번갈아 엮습니다
    var div = rsDiv(ecn), blocks = [], ecs = [];
    for (i = 0; i < nb; i++) { var blk = data.slice(i * dpb, (i + 1) * dpb); blocks.push(blk); ecs.push(rsRem(blk, div)); }
    var all = [];
    for (i = 0; i < dpb; i++) blocks.forEach(function (bk) { all.push(bk[i]); });
    for (i = 0; i < ecn; i++) ecs.forEach(function (e) { all.push(e[i]); });
    var n = 17 + 4 * v, M = [], F = [];
    for (i = 0; i < n; i++) { M.push(new Array(n).fill(false)); F.push(new Array(n).fill(false)); }
    function set(x, y, d) { M[y][x] = d; F[y][x] = true; }
    for (i = 0; i < n; i++) { set(6, i, i % 2 === 0); set(i, 6, i % 2 === 0); }
    [[3, 3], [n - 4, 3], [3, n - 4]].forEach(function (c) {
      for (var dy = -4; dy <= 4; dy++) for (var dx = -4; dx <= 4; dx++) {
        var x = c[0] + dx, y = c[1] + dy, d = Math.max(Math.abs(dx), Math.abs(dy));
        if (x >= 0 && x < n && y >= 0 && y < n) set(x, y, d !== 2 && d !== 4);
      }
    });
    var al = AL[v], last = al.length - 1;
    al.forEach(function (ax, ii) { al.forEach(function (ay, jj) {
      if ((ii === 0 && jj === 0) || (ii === 0 && jj === last) || (ii === last && jj === 0)) return;
      for (var dy = -2; dy <= 2; dy++) for (var dx = -2; dx <= 2; dx++) set(ax + dx, ay + dy, Math.max(Math.abs(dx), Math.abs(dy)) !== 1);
    }); });
    function format(mask) {
      var d = (0 << 3) | mask, r = d;
      for (var i = 0; i < 10; i++) r = (r << 1) ^ ((r >>> 9) * 0x537);
      var fb = ((d << 10) | r) ^ 0x5412, bit = function (i) { return ((fb >>> i) & 1) === 1; };
      for (i = 0; i <= 5; i++) set(8, i, bit(i));
      set(8, 7, bit(6)); set(8, 8, bit(7)); set(7, 8, bit(8));
      for (i = 9; i < 15; i++) set(14 - i, 8, bit(i));
      for (i = 0; i < 8; i++) set(n - 1 - i, 8, bit(i));
      for (i = 8; i < 15; i++) set(8, n - 15 + i, bit(i));
      set(8, n - 8, true);
    }
    format(0);
    // 데이터 비트를 오른쪽 아래부터 두 줄씩 지그재그로 채웁니다
    var bi = 0, total = all.length * 8;
    for (var right = n - 1; right >= 1; right -= 2) {
      if (right === 6) right = 5;
      for (var vert = 0; vert < n; vert++) for (var j = 0; j < 2; j++) {
        var x = right - j, up = ((right + 1) & 2) === 0, y = up ? n - 1 - vert : vert;
        if (!F[y][x] && bi < total) { M[y][x] = ((all[bi >>> 3] >>> (7 - (bi & 7))) & 1) === 1; bi++; }
      }
    }
    var MASKS = [
      function (x, y) { return (x + y) % 2 === 0; }, function (x, y) { return y % 2 === 0; },
      function (x) { return x % 3 === 0; }, function (x, y) { return (x + y) % 3 === 0; },
      function (x, y) { return (Math.floor(x / 3) + Math.floor(y / 2)) % 2 === 0; }, function (x, y) { return x * y % 2 + x * y % 3 === 0; },
      function (x, y) { return (x * y % 2 + x * y % 3) % 2 === 0; }, function (x, y) { return ((x + y) % 2 + x * y % 3) % 2 === 0; }
    ];
    function apply(m) { for (var y = 0; y < n; y++) for (var x = 0; x < n; x++) if (!F[y][x] && MASKS[m](x, y)) M[y][x] = !M[y][x]; }
    // 점수가 가장 낮은(읽기 쉬운) 마스크를 고릅니다
    function penalty() {
      var p = 0, dark = 0, y, x, lines = [];
      for (y = 0; y < n; y++) { var row = '', col = ''; for (x = 0; x < n; x++) { row += M[y][x] ? 1 : 0; col += M[x][y] ? 1 : 0; if (M[y][x]) dark++; } lines.push(row, col); }
      lines.forEach(function (s) {
        (s.match(/0{5,}|1{5,}/g) || []).forEach(function (r) { p += r.length - 2; });
        p += 40 * ((s.match(/(?=10111010000|00001011101)/g) || []).length);
      });
      for (y = 0; y < n - 1; y++) for (x = 0; x < n - 1; x++) { var c = M[y][x]; if (c === M[y][x + 1] && c === M[y + 1][x] && c === M[y + 1][x + 1]) p += 3; }
      return p + 10 * Math.floor(Math.abs(dark * 20 / (n * n) - 10));
    }
    var best = 0, bestP = Infinity;
    for (var m = 0; m < 8; m++) { apply(m); format(m); var sc = penalty(); if (sc < bestP) { bestP = sc; best = m; } apply(m); }
    apply(best); format(best);
    return { n: n, M: M, F: F };
  }

  // SVG로 그리기: 데이터 점은 살짝 둥근 사각형, 위치 표시 세 개는 둥근 고리와 보라색 눈동자
  var Q = qr(url), n = Q.n, pad = 3, W = n + pad * 2, d = '';
  function inFinder(x, y) { return (x < 7 && y < 7) || (x >= n - 7 && y < 7) || (x < 7 && y >= n - 7); }
  for (var y = 0; y < n; y++) for (var x = 0; x < n; x++) if (Q.M[y][x] && !inFinder(x, y)) d += 'M' + (x + pad + 0.06) + ' ' + (y + pad + 0.06) + 'h.88v.88h-.88z';
  var eyes = '';
  [[0, 0], [n - 7, 0], [0, n - 7]].forEach(function (c) {
    var X = c[0] + pad, Y = c[1] + pad;
    eyes += '<rect class="f" x="' + (X + 0.5) + '" y="' + (Y + 0.5) + '" width="6" height="6" rx="1.7" stroke-width="1"/>' +
      '<rect class="e" x="' + (X + 2) + '" y="' + (Y + 2) + '" width="3" height="3" rx=".9"/>';
  });
  var svg = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
  svg.setAttribute('viewBox', '0 0 ' + W + ' ' + W);
  svg.setAttribute('role', 'img');
  svg.setAttribute('aria-label', 'QR 코드: ' + url);
  svg.setAttribute('shape-rendering', 'geometricPrecision');
  svg.innerHTML = '<path class="m" d="' + d + '"/>' + eyes;
  box.insertBefore(svg, box.firstChild);

  // 링크 복사: 클립보드에 넣고, 버튼이 ‘복사됨’으로 바뀌었다가 돌아옵니다
  var t1 = 0;
  cp.addEventListener('click', function () {
    try { if (navigator.clipboard && window.isSecureContext) navigator.clipboard.writeText(url).catch(function () {}); } catch (e) { /* 복사가 막힌 환경에서도 표시는 바뀝니다 */ }
    cp.classList.remove('is-ok'); void cp.offsetWidth; cp.classList.add('is-ok');
    live.textContent = '링크를 복사했어요';
    clearTimeout(t1);
    t1 = setTimeout(function () { if (root.isConnected) { cp.classList.remove('is-ok'); live.textContent = ''; } }, 2200);
  });
  if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (es) { if (es.some(function (e) { return e.isIntersecting; })) { io.disconnect(); root.classList.add('is-in'); } }, { threshold: .4 });
    io.observe(root);
  } else root.classList.add('is-in');
})();
</script>
